86: compute_hash() {
87:   local all_files
88:   all_files=$(
89:     for dir in "${SOURCE_DIRS[@]}"; do
90:       find "$dir" -type f
91:     done
92:     for file in "${SOURCE_FILES[@]}"; do
93:       [[ -f "$file" ]] && echo "$file"
94:     done
95:   )
96:   # Force C locale so sort order is consistent regardless of the environment.
97:   # Xcode build phases run without locale variables, which changes sort ordering.
98:   (
99:     # Include PODS_ROOT and RN_ROOT so switching between worktrees or RN
100:     # sources invalidates the cache.
101:     echo "PODS_ROOT=${PODS_ROOT:-}"
102:     echo "RN_ROOT=${RN_ROOT:-}"
103:     # Include the Swift toolchain version so upgrading Xcode invalidates the
104:     # cache. A slice's .swiftmodule and Clang module cache are tied to the
105:     # compiler that produced them; reusing a slice built by an older toolchain
106:     # after an Xcode upgrade surfaces as spurious compile errors that look like
107:     # source problems. swiftc --version reports the swiftlang/clang tags and the
108:     # target triple, so any toolchain change forces a clean rebuild.
109:     echo "TOOLCHAIN_VERSION=$(xcrun swiftc --version 2>/dev/null || true)"
110:     echo "$all_files" | LC_ALL=C sort | while IFS= read -r file; do
111:       echo "$file"
112:       cat "$file"
113:     done
114:   ) | shasum -a 256 | cut -d' ' -f1
115: }
