async function nativeHost(verb: "dev" | "serve", prefix: string[], options: ProofOptions): Promise<ProofHost> {
  const env = { ...process.env, ...options.env }, id = randomUUID();
  const ports = portRange(env), timeout = deadline(options.timeoutMs, 30_000);
  const args = [...(options.args ?? [])];
  if (!args.includes("--port")) args.push("--port", ports);
  const run = logged(resolveBinary(), [verb, ...prefix, ...args], { ...options, env: { ...env, SNAPBACK4_PROOF_ID: id, SNAPBACK4_SUPERVISOR_PID: String(process.pid) } });
  try {
    const url = await ready(run, line => {
      const match = line.match(/^SNAPBACK4_READY verb=(dev|serve) generation=\d+ url=(http:\/\/[^ ]+) pid=(\d+)(?: |$)/);
      if (!match) return;
      if (match[1] !== verb || Number(match[3]) !== run.child.pid) throw Error("proof ready identity mismatch");
      return match[2];
    }, timeout);
    const { response, body } = await fetchIdentity(`${url}/auth/config`, timeout);
    if (!response.ok || body.proof_id !== id) throw Error("proof served identity mismatch");

      if (status.body.pid !== run.child.pid) throw Error("proof dev PID mismatch");

    return { url, pid: run.child.pid!, logs: run.logs, stop: () => stopChild(run.child, deadline(options.stopTimeoutMs, 5000)) };

export async function withBrowser<T>(options: LaunchOptions & Pick<ProofOptions, "cwd" | "logsDir"> & { chromium?: BrowserType }, proof: (context: BrowserContext) => Promise<T>): Promise<T> {
  const directory = logsDir(options); mkdirSync(directory, { recursive: true, mode: 0o700 });
  const err = openSync(join(directory, `${randomUUID()}.browser.stderr.log`), "wx", 0o600);
  const { chromium: suppliedChromium, executablePath, cwd: _cwd, logsDir: _logs, ...other } = options;
  const caller = suppliedChromium ?? (await import("playwright-core")).chromium;
  const launch = { ...other, ...(executablePath === undefined ? {} : { executablePath }), timeout: deadline(options.timeout, 30_000), logger: {
    isEnabled: () => true,
    log: (name: string, severity: string, message: string | Error) => { writeSync(err, `${name} ${severity}: ${String(message)}\n`); },
  } };
  const args = [...(options.args ?? [])];
  if (process.getuid?.() === 0 && !args.includes("--no-sandbox")) args.push("--no-sandbox");
  // Playwright owns its browser child over a pipe, including parent-exit cleanup.
  // No WebSocket server or TCP debugging port is opened.
  let browser;
  try {
    try { browser = await caller.launch({ ...launch, args, chromiumSandbox: !args.includes("--no-sandbox") }); }
    catch (error) {
      if (args.includes("--no-sandbox") || !sandboxUnavailable(error)) throw error;
      browser = await caller.launch({ ...launch, args: [...args, "--no-sandbox"], chromiumSandbox: false });
    }
    let context: BrowserContext | undefined;
    try {
      context = await browser.newContext();
      // Init scripts run on each document, including HTTP redirect destinations,
      // before any application script samples its URL.
      await context.addInitScript(() => {
        if (window.top !== window || !/^https?:$/.test(location.protocol)) return;
        const url = new URL(location.href);
        url.searchParams.set("witness", "web");
        history.replaceState(history.state, "", url.href);
      });
      context.setDefaultTimeout(30_000); context.setDefaultNavigationTimeout(30_000);
      return await proof(context);
    } finally {
      try { if (context) await bounded(context.close(), 5000, "browser context close"); }
      finally { await bounded(browser.close(), 5000, "browser close"); }
    }
  } finally { closeSync(err); }
}

export async function waitForFresh(locator: Locator, previous: string | null, options: number | { within?: Locator; timeoutMs?: number } = 30_000): Promise<string> {
  const timeoutMs = deadline(typeof options === "number" ? options : options.timeoutMs, 30_000);
  if (typeof options !== "number" && options.within) locator = options.within.locator(locator);
  const end = performance.now() + timeoutMs;
  let count = 0, text: string | null = null, detail = "";
  while (performance.now() < end) {
    try {
      count = await bounded(locator.count(), Math.max(1, end - performance.now()), "locator count");
      text = count ? await bounded(locator.textContent({ timeout: Math.max(1, Math.min(1000, end - performance.now())) }), Math.max(1, end - performance.now()), "locator text") : null;
      if (text && text !== previous) return text;
    } catch (error) {
      detail = String(error);
      // A strictness/selector error needs its original repair, not a timeout.
      if (error instanceof Error && error.name !== "TimeoutError" && !/timed out after/.test(error.message)) throw error;
    }
    await new Promise(resolve => setTimeout(resolve, Math.min(25, Math.max(1, end - performance.now()))));
  }
  throw Error(`no fresh signal after ${timeoutMs} ms: ${String(locator)}; last count=${count}, text=${JSON.stringify(text)}; ${count ? "present but unchanged" : "absent"}; previous=${JSON.stringify(previous)}${detail ? `; ${detail}` : ""}`);
}
/** Poll authoritative server truth after a receipt; a pending prediction is not a commit. */
export async function waitForServer<T>(read: () => Promise<T>, accepts: (value: T) => boolean, options: { timeoutMs?: number; label?: string } = {}): Promise<T> {
  const timeoutMs = deadline(options.timeoutMs, 30_000), end = performance.now() + timeoutMs;
  let last: T | undefined;
  while (performance.now() < end) {
    let timer: ReturnType<typeof setTimeout> | undefined;
    try {
      const value = await Promise.race([Promise.resolve().then(read), new Promise<never>((_, reject) => {
        timer = setTimeout(() => reject(Error(`${options.label ?? "server truth"} timed out after ${timeoutMs} ms; last=${JSON.stringify(last)}`)), Math.max(1, end - performance.now()));
      })]);
      last = value;
      if (accepts(value)) return value;
    } finally { clearTimeout(timer); }
    await new Promise(resolve => setTimeout(resolve, Math.min(25, Math.max(1, end - performance.now()))));
  }
  throw Error(`${options.label ?? "server truth"} timed out after ${timeoutMs} ms; last=${JSON.stringify(last)}`);
}

export async function bounded<T>(work: Promise<T>, ms: number, label: string): Promise<T> {
  let timer: ReturnType<typeof setTimeout> | undefined;
  try {
    return await Promise.race([work, new Promise<never>((_, reject) => {
      timer = setTimeout(() => reject(Error(`${label} timed out after ${ms} ms`)), ms);
    })]);
  } finally { clearTimeout(timer); }
}
/** Attempts every cleanup, even if a preceding step fails or stalls. */
export async function cleanup(...steps: Array<(() => unknown | Promise<unknown>) | undefined>): Promise<void> {
  const errors: unknown[] = [];
  for (const step of steps) if (step) {
    try { await bounded(Promise.resolve().then(step), 10_000, "proof cleanup"); }
    catch (error) { errors.push(error); }
  }
  if (errors.length) throw new AggregateError(errors, "proof cleanup failed");
}
export async function stopChild(child: ChildProcess | Pick<ProofHost, "url" | "stop">, graceMs = 5000): Promise<void> {
  deadline(graceMs, 5000);
  if (child && "url" in child && typeof child.url === "string" && typeof child.stop === "function") {
    await bounded(Promise.resolve().then(() => child.stop()), graceMs + 5000, "host stop"); return;
  }
  if (!child || !("kill" in child) || typeof child.kill !== "function" || typeof child.once !== "function"
      || !(child.exitCode === null || typeof child.exitCode === "number")
      || !(child.signalCode === null || typeof child.signalCode === "string")) {
    throw new TypeError("stopChild needs a ChildProcess or { url, stop }");
  }
  if (child.exitCode !== null || child.signalCode !== null) return;
  let done!: () => void;
  const exited = new Promise<void>(resolve => { done = resolve; });
  child.once("exit", done);
  child.once("error", done);
  try {
    child.kill("SIGTERM");
    try { await bounded(exited, graceMs, "SIGTERM"); }
    catch {
      child.kill("SIGKILL");
      await bounded(exited, 5000, "SIGKILL reap");
    }
  } finally { child.removeListener("exit", done); child.removeListener("error", done); }
}
export function portRange(env: NodeJS.ProcessEnv = process.env): string {
  const low = env.PORT_LO, high = env.PORT_HI;
  if (low === undefined && high === undefined) return "0";
  if (!low || !high || !/^\d+$/.test(low) || !/^\d+$/.test(high)
      || Number(low) < 1 || Number(high) > 65535 || Number(low) > Number(high)) {
    throw Error("PORT_LO and PORT_HI must both specify an inclusive range, 1 <= low <= high <= 65535");
  }
  return `${Number(low)}-${Number(high)}`;
}

export function logged(command: string, args: readonly string[], options: ProofOptions = {}) {
  const cwd = resolve(options.cwd ?? "."), id = randomUUID();
  const directory = logsDir(options); mkdirSync(directory, { recursive: true, mode: 0o700 });
  const logs = { stdout: join(directory, `${id}.stdout.log`), stderr: join(directory, `${id}.stderr.log`) };
  const out = openSync(logs.stdout, "wx", 0o600), err = openSync(logs.stderr, "wx", 0o600);
  let child: ChildProcess;
  try { child = spawn(command, [...args], { cwd, detached: options.detached ?? false, env: { ...(options.env ?? process.env), SNAPBACK4_PROOF_LAUNCH: id }, stdio: [options.livenessPipe ? "pipe" : "ignore", options.secret ? "pipe" : out, err] }); }
  finally { closeSync(out); closeSync(err); }
  // Secret stdout never reaches a file, even briefly. Callers still receive it.
  let stdout = "";
  child.stdout?.setEncoding("utf8");
  child.stdout?.on("data", chunk => { stdout += chunk; });
  register(child);
  child.stdin?.on("error", () => {});
  let error: Error | undefined;
  child.on("error", value => { error = value; });
  const closed = new Promise<void>(resolve => child.once("close", () => resolve()));
  return { child, logs, id, closed, stdout: () => options.secret ? stdout : logText(logs.stdout), error: () => error };
}

export async function ready(run: ReturnType<typeof logged>, parse: (line: string) => string | undefined, timeoutMs: number): Promise<string> {
  const end = performance.now() + timeoutMs;
  let outAt = 0, errAt = 0;
  while (performance.now() < end) {
    if (run.error()) throw run.error();
    if (run.child.exitCode !== null || run.child.signalCode !== null) {
      throw Error(`proof child exited before ready (${run.child.exitCode ?? run.child.signalCode}); ${run.logs.stderr}\n${logText(run.logs.stderr).slice(-4000)}`);
    }
    for (const [path, offset, stream] of [[run.logs.stdout, outAt, "out"], [run.logs.stderr, errAt, "err"]] as const) {
      const text = logText(path);
      const complete = text.lastIndexOf("\n") + 1;
      for (const line of text.slice(offset, complete).split("\n")) {
        const url = parse(stripVTControlCharacters(line).trim());
        if (url) return url;
      }
      if (stream === "out") outAt = complete; else errAt = complete;
    }
    await sleep(25);
  }
  throw Error(`proof child ready timed out after ${timeoutMs} ms; logs: ${run.logs.stdout}, ${run.logs.stderr}`);
}

// @ref LLP 3000.011.003.000#8 — a browser proof calls the app's own authenticated client.
import type { App } from "./app.js";
import type { Mutation, Write } from "./contract.js";

type Operations = Record<string, unknown>;
export function registerWebWitness(app: App, api: Operations): void {
  if (typeof window === "undefined" || typeof window.location?.search !== "string") return;
  const enabled = new URLSearchParams(window.location.search).get("witness") === "web";
  const armed = () => { if (!enabled) throw Error("E_WITNESS_DISABLED: add &witness=web"); };
  const operations: unknown[] = [];
  const walk = (value: Operations) => { for (const entry of Object.values(value)) {
    if (entry && typeof entry === "object") {
      if ("kind" in entry && entry.kind === "mutation" && !("ephemeral" in entry && entry.ephemeral === true)) operations.push(entry);
      else if (!("kind" in entry)) walk(entry as Operations);
    }
  } };
  walk(api);
  let form: { terminal: Promise<unknown>; cancel: () => void } | undefined;
  Object.defineProperty(window, "__snapback4Witness", { configurable: true, value: {
    ready: () => app.ready,
    viewer: () => app.client?.viewer(),
    beginForm: (name: string | undefined, expected: Record<string, unknown> | null | undefined, button: Element) => {
      armed();
      if (!app.client) throw Error("E_WITNESS_AUTH: no signed-in client");
      form?.cancel();
      const client = app.client, originalClient = client.mutate, originalApp = app.mutate;
      let click: Event | undefined, submit: Event | undefined, clickTimer: ReturnType<typeof setTimeout> | undefined, claimed = false, cancelObserver = () => {};
      const outsideClick = Symbol("witness outside clicked dispatch");
      let resolve!: (write: unknown) => void;
      const terminal = new Promise(answer => { resolve = answer; });
      let settled = false, noWriteTimer: ReturnType<typeof setTimeout> | undefined;
      const noWrite = () => finish({ state: "failed", mutation: name, why: { code: "E_WITNESS_NO_WRITE", message: "armed click produced no durable mutation (form handler returned early, or the app was not ready)" } });
      const finish = (write: unknown) => {
        if (settled) return;
        settled = true; clearTimeout(expiry); clearTimeout(noWriteTimer); restore(); cancelObserver(); resolve(write);
      };
      // Bound even an abandoned arm or a claimed write that never settles.
      const receiptExpired = () => finish({ state: "failed", mutation: name, why: { code: "E_WITNESS_SENT", message: "claimed write did not settle within 30000 ms; inspect its original intent id" } });
      let expiry = setTimeout(noWrite, 30_000);
      // Chromium's active event ties admission to this actual control's handler.
      // Timers, input autosaves and previously-started writes cannot claim it.
      const capture = (event: Event) => {
        if (!event.isTrusted) { click = undefined; submit = undefined; clearTimeout(clickTimer); return; }
        if (event.isTrusted && event.composedPath().includes(button)) {
          click = event; submit = undefined;
          clearTimeout(noWriteTimer);
          noWriteTimer = setTimeout(() => { if (!claimed) noWrite(); }, 1000);
          clearTimeout(clickTimer);
          clickTimer = setTimeout(() => { click = undefined; submit = undefined; }, 0);
        }
      };
      const captureSubmit = (event: Event) => {
        if (click && event.isTrusted && (event as SubmitEvent).submitter === button) submit = event;
      };
      document.addEventListener("click", capture, true);
      document.addEventListener("submit", captureSubmit, true);
      const restore = () => {
        if (client.mutate === clientWrapper) client.mutate = originalClient;
        if (app.mutate === appWrapper) app.mutate = originalApp;
        document.removeEventListener("click", capture, true);
        document.removeEventListener("submit", captureSubmit, true); clearTimeout(clickTimer);
      };
      const canonical = (value: unknown): unknown => Array.isArray(value) ? value.map(canonical) : value && typeof value === "object"
        ? Object.fromEntries(Object.keys(value).sort().map(key => [key, canonical((value as Record<string, unknown>)[key])])) : value;
      const execute = async (original: Function, receiver: unknown, op: Mutation<any, any>, args: Record<string, unknown>, options?: any): Promise<Write> => {
        const durable = operations.some(value => (value as Mutation<any, any>).name === op.name);
        const eventMatches = !!click && (window.event === click || (!!submit && window.event === submit));
        const argsMatch = expected == null || JSON.stringify(canonical(expected)) === JSON.stringify(canonical(args));
        const eligible = !claimed && eventMatches && !options?.[outsideClick] && durable && (!name || name === op.name) && argsMatch;
        let forwarded = options;
        if (!eligible && receiver === app && durable) {
          const { id, waitSeconds, ...other } = options ?? {};
          forwarded = { ...other, id, waitSeconds, [outsideClick]: true };
        }
        const invoke = () => original.call(receiver, op, args, forwarded) as Promise<Write>;
        if (!eligible) {
          if (!claimed && durable) {
            claimed = true; restore();
            const reason = !eventMatches || options?.[outsideClick] ? "event: expected the armed click or its synchronous submit"
              : name && name !== op.name ? `operation: expected ${name}, got ${op.name}`
              : `arguments: expected ${JSON.stringify(canonical(expected))}, got ${JSON.stringify(canonical(args))}`;
            finish({ state: "failed", mutation: op.name, why: { code: "E_WITNESS_UNCLAIMED", message: reason } });
          }
          return invoke();
        }
        claimed = true; restore(); clearTimeout(expiry); clearTimeout(noWriteTimer);
        // A late click receives its own complete settlement allowance.
        expiry = setTimeout(receiptExpired, 30_000);
        let id: string | undefined, early: Write[] = [];
        const stop = client.onWrite(write => {
          // Native clients publish terminal receipts; foreign adapters must not
          // turn a pending notification into terminal proof evidence either.
          if ((write as Write).state === "pending") return;
          if (id === undefined) { early.push(write); if (early.length > 256) early.shift(); }
          else if (write.id === id) { stop(); finish({ ...write, mutation: op.name }); }
        });
        cancelObserver = stop;
        try {
          const write = await invoke();
          id = write.id;
          const receipt = write.state === "pending" ? early.find(value => value.id === id && value.state !== "pending") : write;
          early = [];
          if (receipt) { stop(); finish({ ...receipt, mutation: op.name }); }
          return write;
        } catch (error) { stop(); finish({ state: "failed", why: { code: "E_WITNESS_WEB" } }); throw error; }
      };
      // Hooks submit through client; forms using app.mutate enter before its readiness await.
      const clientWrapper = ((op: Mutation<any, any>, args: Record<string, unknown>, options?: any) => execute(originalClient, client, op, args, options)) as typeof client.mutate;
      const appWrapper = ((op: Mutation<any, any>, args: Record<string, unknown>, options?: any) => execute(originalApp, app, op, args, options)) as typeof app.mutate;
      client.mutate = clientWrapper; app.mutate = appWrapper;
      form = { terminal, cancel: () => { if (!claimed) noWrite(); else finish({ state: "failed", mutation: name, why: { code: "E_WITNESS_UNCLAIMED", message: "rearmed before the claimed write settled; inspect its original intent id" } }); } };
    },
    formResult: () => form?.terminal,
    mutate: async (name: string | undefined, args: Record<string, unknown>, id: string) => {
      armed();
      const operation = operations.find(value => !name || (value as { name: string }).name === name);
      if (!operation) throw Error("E_WITNESS_MUTATION: no matching durable mutation; pass --mutation with a generated API name (ephemeral signals do not prove stored writes)");
      if (!app.client) throw Error("E_WITNESS_AUTH: the app has no signed-in client");
      let observed: unknown;
      let settle: ((write: unknown) => void) | undefined;
      const stop = app.client.onWrite(write => { if (write.id === id) { observed = write; settle?.(write); } });
      try {
        const write = await app.mutate(operation as Mutation<any, any>, args, { id, waitSeconds: 0 });
        if (write.state !== "pending") return write;
        return observed ?? await new Promise(resolve => { settle = resolve; });
      } finally { stop(); }
    },
  } });
}

export function isReady<R extends Read<unknown>>(read: R): read is Extract<R, { readonly data: unknown }> {
  return "data" in read;
}
/** Narrow to disclosed non-null data; isReady alone still includes absent/partial null. */
export function isPresent<R extends Read<unknown>>(read: R): read is Extract<R, { readonly existence: "present" }> {
  return read != null && "data" in read && read.existence === "present";
}
export interface ReadMatchHandlers<T, R = unknown> {
  loading: (read: Extract<Read<T>, { loading: true }>) => R;
  denied: (why: Refusal, read: Extract<Read<T>, { denied: Refusal }>) => R;
  absent: (read: Absent<T>) => R;
  partial: (data: T, read: Partial<T>) => R;
  present: (data: Present<T>["data"], read: Present<T>) => R;
}

/** `otherwise` covers omitted loading/absent/present handlers only.
 * Denial and uncertain emptiness always require an explicit decision. */
export type ReadMatchFallbackHandlers<T, R = unknown> =
  Pick<ReadMatchHandlers<T, R>, "denied" | "partial"> & {
    [K in "loading" | "absent" | "present"]?: ReadMatchHandlers<T, R>[K];
  } & { otherwise: (read: Extract<Read<T>, { loading: true }> | Absent<T> | Present<T>) => R };

/** A handler set with `otherwise` while it still lacks `denied` or `partial`:
 * accepted for inference only, so the missing decision is named below. */
type ReadMatchIncomplete<T> = { [K in keyof ReadMatchHandlers<T>]?: ReadMatchHandlers<T>[K] } & Pick<ReadMatchFallbackHandlers<T>, "otherwise">;
type MissingDenied<T> = Omit<ReadMatchHandlers<T>, "denied"> & { denied?: never };
/** The error a fallback without `denied`/`partial` reports: the property's
 * type is the fix. LLP 3000.031 §8 keeps both decisions explicit; an
 * `otherwise` built by `otherwiseAll` is that decision, so it needs neither. */
type FallbackDecisions<H> = H extends { otherwise: OtherwiseAll<unknown> } ? unknown
  : H extends { otherwise: unknown }
  ? ("denied" extends keyof H ? unknown : { denied: "otherwise skips denied: add denied: why => …, or replace otherwise with ...otherwiseAll(fallback) to send denied and partial there too" })
    & ("partial" extends keyof H ? unknown : { partial: "otherwise skips partial: add partial: data => …, or replace otherwise with ...otherwiseAll(fallback) to send denied and partial there too" })
  : "denied" extends keyof H ? unknown : { denied: "matchRead requires denied: why => … (or ...otherwiseAll(fallback))" };

/** What `matchRead` accepts: a read, or a `usePage` result (it reads `page.read`). */
type Matchable<T> = Read<T> | { readonly read: Read<T>; readonly loadMore: () => void };

/** Render the current read, preserving its narrowed data and metadata.
 * Pass no type arguments: T comes from the read and the result is the union of
 * the handlers' returns. `present`/`partial` receive data; `otherwise` receives
 * the read itself (narrow with `isReady`). Denied and partial stay explicit:
 * to route them to the fallback too, say so with one spread, in any position,
 * `matchRead(read, { ...otherwiseAll(() => "…"), present: rows => … })`; a
 * handler the call names always wins over it.
 * Pure dispatch: no reclassification, subscription, retention or permission.
 * A `usePage` result may be passed whole (its `read` is matched); present
 * pages may still be incomplete.
 * @ref LLP 3000.031#8 — explicit denied/partial decisions even with a fallback.
 */
// @ref LLP 3000.013#6.6 — an omitted denied/partial with `otherwise` fails naming the fix, not a Pick<> dump.
// @ref LLP 3000.013#6.7 — NoInfer wraps the whole handler set, never T inside it: `NoInfer<T> & ({} | undefined)` never simplified, so present rows refused `rows[0]` and `for…of`.
// @ref LLP 3000.013#6.8 — a page hook is matched by its read: `matchRead(page, …)` was the most common miss.
export function matchRead<T, H extends NoInfer<ReadMatchHandlers<T> | ReadMatchFallbackHandlers<T> | ReadMatchIncomplete<T> | MissingDenied<T>>>(
  read: Matchable<T>, handlers: H & FallbackDecisions<H>,
): ReturnType<Extract<H[keyof H], (...args: never[]) => unknown>>;

## React hooks and auth decisions
Render editable sign-in controls only while signed-out, Restoring… otherwise. AppProvider resets on app/viewer changes; restoring → signed-out preserves the null-viewer form. `useAppReady()` waits for initial app.ready and current signed-in auth without an admission refusal; gate writers/proof input. Successful later sign-in recovers from initial failure. Readiness is neither connectivity nor permission; surface auth.why.
`useAuth()` members: `state`, `viewer`, `why`, `signup(email, password)`, `login(email, password)`, `guest()`, `logout()`, `logoutEverywhere()`. `signOut()` belongs to the client. Expo session store: secureStoreSessionStore(SecureStore); password form: [reference](CLIENT-AND-OPERATIONS.md#password-react-and-expo).
`matchRead` handlers: `loading(read)`, `denied(why,read)`, `absent(read)`, `partial(data,read)`, `present(data,read)`. `otherwise` covers omitted loading/absent/present; otherwiseAll also takes denied/partial. A complete empty list is absent; `absent` can be an authorised empty list. `matchReadList` handles lists; `isPresent(read)` excludes null, while `isReady(read)` still permits null; `existence(read)` and `read.existence` classify.
`loading`: keep shells; `denied`: show refusal; `present`: render; `absent`: proven empty; `partial`: uncertain. Offline absent is absence at the acquired watermark. Hidden/missing rows are indistinguishable. `readPresentation` notices never grant permission. Render page.items, follow hasMore/loadMore. Unknown counts never hide conversations.
`query(op,args)` settles ready/denied within 30 seconds. Unavailable offline reads give immediate `E_OFFLINE`. After four rounds, exhaustion refuses retryable `E_QUERY_RETRY`; permanently unavailable held reads refuse non-retryable `E_QUERY_HELD`. Retention, RowOf/ArgsOf/ResultOf and {skip}: [reference](CLIENT-AND-OPERATIONS.md). Share/paginate: eight active, eight queued observation attempts; direct queries and timed presence refreshes use separate paths.
Browser createApp: no sync → journal; sync → webDevice(); online writes: writes:"online". Native: nativeDevice; Expo Go online. Every milestone: npm run export:ios; proof in the background: witness status --wait. Rebuild npm run native:smoke. Acquisition/offline reopen/reconnect/write replay: guide witness.
useOnline: server reads/ephemeral/actions; durable writes: app.mutate. SDK owns delivery reporting; screen owns visible `seen` reporting. useSeen owns receipt retries (guide receipts). Read-only clients: guide read-only. Media: OFFLINE-UPLOADS.md. Never wait for networkidle; assert fresh sent before rendered output (guide testing).
Keyed counts: hidden key → null; authorized missing count group → 0; public count → number. Choose count ?? 0 explicitly.
