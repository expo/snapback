#!/usr/bin/env bash
# Host-only bridge: build in the commissioned shared target, then copy artifacts
# for repository scripts that use literal target paths. No target symlinks.
/home/ccheever/.cargo/bin/cargo "$@"
status=$?
if [ "${CARGO_TARGET_DIR:-}" = /home/ccheever/sb5-lanes/target ] && [ "$status" = 0 ] && [ "${1:-}" = build ] && [ "$PWD" = /home/ccheever/sb5-lanes/lane-a/snapback4 ]; then
  if [[ " $* " == *" wasm32-unknown-unknown "* ]]; then
    mkdir -p target/wasm32-unknown-unknown/wasm
    cp --reflink=auto "$CARGO_TARGET_DIR/wasm32-unknown-unknown/wasm/snapback4_device.wasm" target/wasm32-unknown-unknown/wasm/snapback4_device.wasm || exit 1
  else
    mkdir -p target/debug
    cp --reflink=auto "$CARGO_TARGET_DIR/debug/snapback4" target/debug/snapback4 || exit 1
  fi
fi
exit "$status"
