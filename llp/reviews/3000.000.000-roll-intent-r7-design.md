pub(super) fn prepare(target: &[Source], current: &[Source]) -> Bridge {
    let old = crate::compile(target);
    let next = crate::compile(current);
    if !old.diagnostics.is_empty() {
        return Bridge {
            sources: target.to_vec(),
            changes: vec![],
            diagnostics: old.diagnostics,
        };
    }
    if !next.diagnostics.is_empty() {
        return Bridge {
            sources: current.to_vec(),
            changes: vec![],
            diagnostics: next.diagnostics,
        };
    }
    if contracts(&old, &next) {
        return retained(current.to_vec(), false);
    }
    match alpha(target, current) {
        Ok(Some(sources)) => {

            if !bound.diagnostics.is_empty() {

                    sources,
                    changes: vec![],

            }

                return retained(sources, true);
            }
        }
        Err(reason) => return refused(current, &reason),
        Ok(None) => {}
    }
    refused(current,"the complete current workflow does not retain every predecessor table, field, operation and accepted input")
}

    if !old.schema.tables.iter().all(|(name, table)| {

            table
                .columns
                .keys()

        })
    }) {

            .iter()

    if old == new {
        return true;
    }
    match (old, new) {
        (ArgType::Optional(a), ArgType::Optional(b)) => accepts(a, b),
        (_, ArgType::Optional(b)) => accepts(old, b),
        (ArgType::Optional(_), _) => false,
        (
            ArgType::Text {
                min: a,
                max: b,
                format: f,
            },
            ArgType::Text {
                min: x,
                max: y,
                format: g,
            },
        ) => f == g && x <= a && b <= y,
        (ArgType::Enum { variants: a }, ArgType::Enum { variants: b }) => {
            a.iter().all(|v| b.contains(v))
        }
        (
            ArgType::List { .. } | ArgType::ListRange { .. },
            ArgType::List { .. } | ArgType::ListRange { .. },
        ) => {
            let parts = |kind: &ArgType| match kind {
                ArgType::List { of, max } => (0, *max, of.clone()),
                ArgType::ListRange { of, min, max } => (*min, *max, of.clone()),
                _ => unreachable!(),
            };
            let (a, b, element) = parts(old);
            let (x, y, next) = parts(new);
            x <= a
                && b <= y
                && match (element, next) {
                    (None, None) => true,
                    (Some(a), Some(b)) => accepts(&a, &b),
                    _ => false,
                }
        }
        (ArgType::IntRange { .. }, ArgType::Int) => true,
        (ArgType::IntRange { min: a, max: b }, ArgType::IntRange { min: x, max: y }) => {
            x <= a && b <= y
        }
        _ => false,
    }
}
fn word(name: &str) -> bool {
    !name.is_empty() && name.chars().all(|c| c.is_ascii_alphanumeric() || c == '_')
}
fn offset(text: &str, line: u32, col: u32) -> usize {
    let start = text
        .split_inclusive('\n')
        .take(line.saturating_sub(1) as usize)
        .map(str::len)
        .sum::<usize>();
    start
        + text[start..]
            .chars()
            .take(col.saturating_sub(1) as usize)
            .map(char::len_utf8)
            .sum::<usize>()
}
fn declaration(text: &str, line: u32) -> (usize, usize) {
    let start = offset(text, line, 1);
    let end = text
        .split_inclusive('\n')
        .enumerate()
        .skip(line as usize)
        .find(|(_, s)| {
            !s.trim().is_empty() && !s.starts_with(char::is_whitespace) && !s.starts_with("--")
        })
        .map(|(i, _)| offset(text, i as u32 + 1, 1))
        .unwrap_or(text.len());
    (start, end)
}

    }

    if old.tables.iter().chain(&doc.tables).any(|t| !word(&t.name)) {
        return Ok(None);
    }

        if old.tables.iter().any(|t| t.name == next.name) {
            continue;
        }

            next.renamed_from.as_ref() == Some(&t.name)

    // identity in an argument/column must never turn a nominal Ref into Int,
    // Principal, etc. Validate the parser's actual type interpretation.

                    .into(),
            );
        }

        .tables
        .iter()

        .collect::<BTreeSet<_>>();

            "the declared table mapping is ambiguous or captures a current table name".into(),
        );
    }

        || !doc.effects.is_empty()
        || !doc.maintains.is_empty()
        || !doc.timelines.is_empty()
        || !doc.views.is_empty()
        || !doc.feed_views.is_empty()
        || !doc.backfills.is_empty()
        || !doc.downfills.is_empty()
        || !doc.extractions.is_empty()

            .tables
            .iter()
            .any(|t| t.receipts.is_some() || t.columns.iter().any(|c| c.maintained.is_some()))

    });

            ast::Stmt::Bind { names, .. } => names.iter().any(|n| reserved.contains(n)),

                reserved.contains(name) || bindings(body, reserved)
            }

                then_body,
                else_body,
                ..

            ast::Stmt::As { body, .. } => bindings(body, reserved),
            _ => false,
        })

        .iter()

        capture |= args.iter().any(|a| reserved.contains(&a.name));

        ast::Expr::Walk { .. } | ast::Expr::Merge(_) => unsupported = true,
        _ => (),

        crate::bind::intent_walk::expression(&mut inv.value, &mut check);

            .rules

                .flatten()

            }
        }

            crate::bind::intent_walk::expression(value, &mut check);
        }
    }

        return Err(
            "a recursive or merge expression needs an authored compatible table mapping".into(),
        );
    }
    if capture {
        return Err("a table identity mapping captures a local, argument or shape name".into());
    }

    let mut edits = Vec::<(usize, usize, String)>::new();

                .tables
                .iter()

        }
        let (a, b) = declaration(text, table.site.line);
        edits.push((a, b, source));
    }
    for inv in &doc.invariants {
        let table = names.get(&inv.table).unwrap_or(&inv.table);
        let mut value = inv.value.clone();
        crate::bind::intent_walk::expression(&mut value, &mut |e| match e {
            ast::Expr::Read(r)
            | ast::Expr::Exists { read: r, .. }
            | ast::Expr::Created { read: r, .. } => {
                if let Some(name) = names.get(&r.table) {
                    r.table = name.clone()
                }
            }
            ast::Expr::Write { table, .. } => {
                if let Some(name) = names.get(table) {
                    *table = name.clone()
                }
            }
            _ => (),
        });
        let target = if let Some(column) = &inv.column {
            format!("{table}.{column} {} only", inv.target)
        } else if inv.target == "always" {
            format!("{table} always")
        } else {
            format!("{table}.{} only", inv.target)
        };

        edits.push((
            a,
            b,
            format!("invariant {target} {}\n", print::expression(&value)),
        ));
    }
    let mut other = doc.clone();
    other.tables.clear();
    other.invariants.clear();

    crate::bind::intent_walk::document(&mut other, &mut |e| {
        let (table, site) = match e {
            ast::Expr::Read(r)
            | ast::Expr::Exists { read: r, .. }
            | ast::Expr::Created { read: r, .. } => (&r.table, &r.site),
            ast::Expr::Write { table, site, .. } => (&*table, &*site),
            _ => return,
        };
        let Some(name) = names.get(table) else { return };
        if let Some(token) = tokens.iter().find(|t| {
            t.site.line == site.line
                && t.site.col >= site.col
                && t.kind == Kind::Word
                && t.value == *table
        }) {
            let a = offset(text, token.site.line, token.site.col);
            edits.push((a, a + token.value.len(), name.clone()));
        } else {
            reference_error = Some("a table reference has no unambiguous source span".to_owned());
        }
    });
    if let Some(error) = reference_error {
        return Err(error);
    }
    for arg in doc
        .programs
        .iter()
        .flat_map(|p| &p.args)
        .chain(doc.functions.iter().flat_map(|p| &p.args))
        .chain(doc.actions.iter().flat_map(|p| &p.args))
        .chain(doc.actions.iter().flat_map(|p| p.returns.iter().flatten()))
    {
        let mut kind = arg.type_.clone();
        remap_type(&mut kind, &names);
        if kind != arg.type_ {
            let (a, b) = arg_type(text, &tokens, arg)?;
            edits.push((a, b, print::scalar_type(&kind)));
        }
    }
    edits.sort_by_key(|(a, b, _)| (*a, *b));
    edits.dedup();
    if edits.iter().any(|(a, b, _)| {
        a > b || *b > text.len() || !text.is_char_boundary(*a) || !text.is_char_boundary(*b)
    }) {
        return Err("an unsupported source span needs an authored compatible mapping".into());
    }
    if edits.windows(2).any(|p| p[0].1 > p[1].0) {
        return Err("overlapping source spans in table identity mapping".into());
    }
    let mut result = text.clone();
    for (a, b, value) in edits.into_iter().rev() {
        result.replace_range(a..b, &value);
    }
    Ok(Some(vec![(file.clone(), result)]))
}
fn header(
    file: &str,
    source: &str,
    from: &str,
    to: &str,
    prior: Option<&str>,
) -> Result<String, String> {
    if prior.is_some_and(|p| !word(p)) {
        return Err("qualified predecessor table origin needs an authored source".into());
    }
    let line = source.lines().next().ok_or("missing table header")?;
    let (tokens, _) = lexer::lex(file, line);
    let name = tokens
        .get(1)
        .filter(|t| t.kind == Kind::Word && t.value == from)
        .ok_or("missing table name span")?;
    let mut edits = vec![(
        offset(line, name.site.line, name.site.col),
        offset(line, name.site.line, name.site.col) + from.len(),
        to.to_owned(),
    )];
    if let Some(at) = tokens.iter().enumerate().skip(2).find_map(|(i,t)| (t.value=="renamed").then_some(i)) {
        let last = tokens
            .get(at + 2)
            .filter(|_| tokens.get(at + 1).is_some_and(|t| t.value == "from"))
            .ok_or("invalid table origin span")?;
        edits.push((
            offset(line, tokens[at].site.line, tokens[at].site.col),
            offset(line, last.site.line, last.site.col) + last.value.len(),
            prior
                .map(|p| format!("renamed from {p}"))
                .unwrap_or_default(),
        ));
    } else if let Some(prior) = prior {
        let colon = tokens
            .iter()
            .find(|t| t.value == ":")
            .ok_or("missing table header colon")?;
        let a = offset(line, colon.site.line, colon.site.col);
        edits.push((a, a, format!(" renamed from {prior}")));
    }
    let mut out = source.to_owned();
    edits.sort_by_key(|(a, _, _)| std::cmp::Reverse(*a));
    for (a, b, value) in edits {
        out.replace_range(a..b, &value);
    }
    Ok(out)
}
fn arg_type(
    text: &str,
    tokens: &[lexer::Token],
    arg: &ast::ArgDecl,
) -> Result<(usize, usize), String> {
    let at = tokens
        .iter()
        .position(|t| t.site == arg.site)
        .ok_or("missing argument source span")?;
    let colon = (at..tokens.len())
        .find(|i| tokens[*i].value == ":")
        .ok_or("missing argument type colon")?;
    let start = tokens.get(colon + 1).ok_or("missing argument type")?;
    let mut depth = 0_i32;
    for (ordinal, t) in tokens.iter().skip(colon + 1).enumerate() {
        // `alias` and `default` are legal first nominal type tokens too.
        if ordinal > 0
            && depth == 0
            && (matches!(
                t.value.as_str(),
                "," | ")" | "=" | "default" | "alias" | "renamed" | "deprecated"
            ) || t.kind == Kind::Newline)
        {
            return Ok((
                offset(text, start.site.line, start.site.col),
                text[..offset(text, t.site.line, t.site.col)]
                    .trim_end()
                    .len(),
            ));
        }
        match t.value.as_str() {
            "(" | "[" | "<" => depth += 1,
            ")" | "]" | ">" => depth -= 1,
            _ => (),
        }
    }
    Err("unterminated argument type source span".into())
}

        match &mut t.kind {TypeKind::Ref{table}=>{if let Some(name)=aliases.get(table){*table=name.clone();}},TypeKind::List{item,..}=>remap_type(item,aliases),_=>()}
    }

        if type_!=column.type_ {if let Some((start,end))=type_span(file,&source,column){edits.push((start,end,print::scalar_type(&type_)));}}
    }
    let mut visit=|expr:&mut ast::Expr| {
        if let ast::Expr::Read(read)|ast::Expr::Exists{read,..}|ast::Expr::Created{read,..}=expr {
            if let Some(name)=aliases.get(&read.table) {let start=offset(&source,read.site.line,read.site.col);edits.push((start,start+read.table.len(),name.clone()));}
        }
    };
    for rule in decl.rules.iter_mut().chain(decl.column_reads.iter_mut().map(|(_,r)|r)) {crate::bind::intent_walk::expression(&mut rule.value,&mut visit);}
    for column in &mut decl.columns {
        for expr in [&mut column.read,&mut column.default].into_iter().flatten(){crate::bind::intent_walk::expression(expr,&mut visit);}
        if let Some(expr)=column.maintained.as_mut().and_then(|a|a.filter.as_mut()){crate::bind::intent_walk::expression(expr,&mut visit);}
    }
    for (expr,_) in &mut decl.final_exists {crate::bind::intent_walk::expression(expr,&mut visit);}
    // Sync targets are structural table probes outside the expression AST.
    if let Some(sync)=&decl.sync {
        let (tokens,_)=lexer::lex(file,&source);
        // Lexer continuation rules keep indented `or` arms in this logical
        // declaration even when their physical line differs from `sync`.
        if let Some(start)=tokens.iter().position(|t|t.site==sync.site) {
            let end=tokens[start..].iter().position(|t|matches!(t.kind,Kind::Newline|Kind::Eof)).map_or(tokens.len(),|n|start+n);
            for pair in tokens[start..end].windows(2) {if pair[0].kind==Kind::Word && pair[1].value=="[" {if let Some(name)=aliases.get(&pair[0].value){let start=offset(&source,pair[0].site.line,pair[0].site.col);edits.push((start,start+pair[0].value.len(),name.clone()));}}}
        }
    }

    }

    for (start,end,value) in edits {source.replace_range(start..end,&value);}
    Ok(source)
}
fn join(old:&ast::ScalarType,new:&ast::ScalarType,tables:&BTreeMap<String,(&String,&String,ast::TableDecl)>)->Option<ast::ScalarType> {
    let kind=match (&old.kind,&new.kind) {
        (TypeKind::Int|TypeKind::IntRange{..},TypeKind::Int|TypeKind::IntRange{..})=>{
            let bounds=|k:&TypeKind|match k{TypeKind::IntRange{min,max}=>(min.unwrap_or(i64::MIN),max.unwrap_or(i64::MAX)),_=>(i64::MIN,i64::MAX)};
            let(a,b)=bounds(&old.kind);let(x,y)=bounds(&new.kind);let(min,max)=(a.min(x),b.max(y));
            if min==i64::MIN && max==i64::MAX {TypeKind::Int}else{TypeKind::IntRange{min:(min!=i64::MIN).then_some(min),max:(max!=i64::MAX).then_some(max)}}
        }
        (TypeKind::Image{max:a,once:b},TypeKind::Image{max:x,once:y}) if b==y=>TypeKind::Image{max:(*a).max(*x),once:*b},
        (TypeKind::Video{max:a},TypeKind::Video{max:b})=>TypeKind::Video{max:(*a).max(*b)},
        (TypeKind::File{max:a,types:b},TypeKind::File{max:x,types:y}) if b==y=>TypeKind::File{max:(*a).max(*x),types:b.clone()},
        (TypeKind::Text{min:a,max:b,format:c},TypeKind::Text{min:x,max:y,format:z}) if c==z=>TypeKind::Text{min:Some(a.unwrap_or(0).min(x.unwrap_or(0))).filter(|v|*v>0),max:(*b).max(*y),format:c.clone()},
        (TypeKind::Enum{variants:a},TypeKind::Enum{variants:b})=>{let mut v=a.clone();for x in b {if !v.contains(x){v.push(x.clone());}}TypeKind::Enum{variants:v}},
        (TypeKind::List{item:a,min:b,max:c},TypeKind::List{item:x,min:y,max:z})=>TypeKind::List{item:Box::new(join(a,x,tables)?),min:(*b).min(*y),max:(*c).max(*z)},
        // Only an explicit rename provides nominal identity; structural similarity
        // or independently declared replacement tables cannot justify a bridge.
        (TypeKind::Ref{table:a},TypeKind::Ref{table:b}) if a==b || tables.get(a).is_some_and(|(_,_,t)| t.name==*b)=>TypeKind::Ref{table:a.clone()},
        (a,b) if a==b=>a.clone(),
        // Other conversions require an explicit typed migration. A bridge never
        // guesses a nominal reference, money currency or asset policy.
        _=>return None,
    };
    Some(ast::ScalarType{kind,optional:old.optional||new.optional})
}

            for(expr,_)in &next.final_exists {edits.push((at,at,format!("  after insert <- {}\n",print::expression(&remap(expr)))));}
            for prior in [&old.expire,&old.not_before].into_iter().flatten(){if let Some((a,b))=line_span(file,text,&prior.1){edits.push((a,b,String::new()));}}
            if let Some((field,_))=&next.expire {edits.push((at,at,format!("  expire at .{field}\n")));}
            if let Some((field,_))=&next.not_before {edits.push((at,at,format!("  not before .{field}\n")));}
            for index in next.indexes.iter().filter(|i|i.unique&&!old.indexes.iter().any(|o|o.unique&&o.columns==i.columns)) {
                if let Some((_,current_text,_))=docs.iter().find(|(_,_,d)|d.tables.iter().any(|t|t.name==next.name)) {
                    if let Some((a,b))=line_span(&index.site.file,current_text,&index.site){edits.push((at,at,format!("{}\n",&current_text[a..b])));}
                }
            }
            let mut replaced = BTreeSet::new();
            for rule in &old.rules {
                let Some((start, end)) = line_span(file, text, &rule.site) else {
                    continue;
                };
                if !replaced.insert((start, end)) {
                    continue;
                }
                let rules = old
                    .rules
                    .iter()
                    .filter(|r| r.site == rule.site)
                    .map(|old_rule| {
                        let effect = old_rule.effect.word();
                        match next.rules.iter().find(|r| r.effect == old_rule.effect) {
                            Some(current) => {
                                if let Some(raw)=authored_read(current){return raw}
                                let bindings = if current.bindings.is_empty() {
                                    String::new()
                                } else {
                                    format!(" ({})", current.bindings.join(", "))
                                };
                                format!(
                                    "  {effect}{bindings} <- {}",
                                    print::expression(&remap(&current.value))
                                )
                            }
                            None if old_rule.effect==ast::Effect::Read&&next.public.is_some()=>String::new(),
                            None => format!("  {effect} <- deny"),
                        }
                    })
                    .collect::<Vec<_>>()
                    .join("\n");
                if text[start..end] != rules {
                    edits.push((start, end, rules));
                    changes.push(format!(
                        "{}: current authored {} authority retained",
                        old.name,
                        rule.effect.word()
                    ));
                }
            }
            for col in &old.columns {
                let Some(next_col) = next
                    .columns
                    .iter()
                    .find(|c| c.name == col.name || c.renamed_from.as_ref() == Some(&col.name))
                else {
                    continue;
                };
                let grouped = next
                    .column_reads
                    .iter()
                    .find(|(fields, _)| fields.contains(&next_col.name))
                    .map(|(_, r)| &r.value);
                let Some(gate) = next_col.read.as_ref().or(grouped) else {
                    continue;
                };
                if col.read.is_none()
                    && !old
                        .column_reads
                        .iter()
                        .any(|(fields, _)| fields.contains(&col.name))
                {
                    let at = declaration(text, old.site.line).1;
                    edits.push((
                        at,
                        at,
                        format!(
                            "  read({}) <- {}\n",
                            col.name,
                            print::expression(&remap(gate))
                        ),
                    ));
                    changes.push(format!(
                        "{}.{}: retained current payload read guard",
                        old.name, col.name
                    ));
                }
            }
        }
        for old in &doc.programs {
            let mut reads = BTreeSet::new();
            let mut d = ast::Document::default();
            d.programs.push(old.clone());
            crate::bind::intent_walk::document(&mut d, &mut |e| match e {
                ast::Expr::Read(r)
                | ast::Expr::Exists { read: r, .. }
                | ast::Expr::Created { read: r, .. } => {
                    reads.insert(r.table.clone());
                }
                ast::Expr::Write { table, .. } => {
                    reads.insert(table.clone());
                }
                _ => (),
            });
            if reads.is_disjoint(&relaxed)
                && reads.is_disjoint(&expanded)
                && !forced_programs.contains(&old.name)
            {
                continue;
            }
            let Some((current_file, current_text, next)) = programs.get(&old.name) else {
                continue;
            };
            if old.kind != next.kind
                || old.args.len() != next.args.len()
                || old
                    .args
                    .iter()
                    .zip(&next.args)
                    .any(|(a, b)| a.name != b.name || !input_accepts(&a.type_, &b.type_))
            {
                continue;
            }
            let (_, _, mut source) = declaration(current_text, next.site.line);
            let (tokens, _) = lexer::lex(current_file, &source);
            let mut replacements = Vec::new();
            if next.name!=old.name {if let Some(name)=tokens.iter().find(|t|t.value==next.name){let at=offset(&source,name.site.line,name.site.col);replacements.push((at,at+name.value.len(),old.name.clone()));}}
            let parsed = parser::parse_document(current_file, &source).0;
            let mut parsed = parsed;
            crate::bind::intent_walk::document(&mut parsed, &mut |e| {
                let (table, site) = match e {
                    ast::Expr::Read(r)
                    | ast::Expr::Exists { read: r, .. }
                    | ast::Expr::Created { read: r, .. } => (&r.table, &r.site),
                    ast::Expr::Write { table, site, .. } => (&*table, &*site),
                    _ => return,
                };
                if let Some(old) = aliases.get(table) {
                    if let Some(token) = tokens.iter().find(|t| {
                        t.site.line == site.line && t.site.col >= site.col && t.value == *table
                    }) {
                        let at = offset(&source, token.site.line, token.site.col);
                        replacements.push((at, at + table.len(), old.clone()));
                    }
                }
            });
            replacements.sort();
            replacements.dedup();
            replacements.sort_by_key(|r| std::cmp::Reverse(r.0));
            for (a, b, value) in replacements {
                source.replace_range(a..b, &value);
            }
            let (a, b, _) = declaration(text, old.site.line);
            edits.push((a, b, source));
            changes.push(format!("operation {}: current authored compatibility endpoint preserves its argument contract and guards",old.name));
        }
        for fill in &doc.backfills {
            let Some(current) = tables.get(&fill.table) else {
                continue;
            };
            if let ast::Expr::Object { fields, .. } = &fill.value {
                if fields
                    .iter()
                    .all(|f| current.columns.iter().any(|c| c.name == f.name))
                {
                    let (a, b, _) = declaration(text, fill.site.line);
                    edits.push((a, b, String::new()));
                    changes.push(format!(
                        "{}: retained current forward migration custody instead of historical {}",
                        fill.table, fill.name
                    ));
                }
            }
        }
        for down in &doc.downfills {
            if docs.iter().any(|(_, _, d)| {
                d.downfills
                    .iter()
                    .any(|f| f.table == down.table && f.name == down.name)
            }) {
                continue;
            }
            let Some(current) = tables.get(&down.table) else {
                continue;
            };
            if let ast::Expr::Object { fields, .. } = &down.value {
                if fields
                    .iter()
                    .all(|f| current.columns.iter().any(|c| c.name == f.name))
                {
                    let (a, b, _) = declaration(text, down.site.line);
                    edits.push((a, b, String::new()));
                    changes.push(format!("{}: obsolete down {} stays historical; visible view keeps current native values",down.table,down.name));
                }
            }
        }
        for (current_file, current_text, current_doc) in &docs {
            for backfill in &current_doc.backfills {
                if !doc.tables.iter().any(|t| t.name == backfill.table) {
                    continue;
                }
                let (_, _, source) = declaration(current_text, backfill.site.line);
                edits.push((text.len(), text.len(), format!("\n{source}")));
                changes.push(format!(
                    "{}: retained current migration {}",
                    backfill.table, backfill.name
                ));
                let _ = current_file;
            }
        }
        edits.sort_by_key(|e| std::cmp::Reverse((e.0, e.1)));
        for (a, b, value) in edits {
            text.replace_range(a..b, &value);
        }
    }
    relaxed.extend(expanded);
    relaxed
}

pub(super) fn input_accepts(old: &ast::ScalarType, new: &ast::ScalarType) -> bool {
    if old == new {
        return true;
    }
    if old.optional && !new.optional {
        return false;
    }
    use ast::TypeKind::*;
    let bounds = |k: &ast::TypeKind| match k {
        Int => Some((i64::MIN, i64::MAX)),
        IntRange { min, max } => Some((min.unwrap_or(i64::MIN), max.unwrap_or(i64::MAX))),
        _ => None,
    };
    match (bounds(&old.kind), bounds(&new.kind)) {
        (Some((a, b)), Some((x, y))) => x <= a && y >= b,
        _ => match (&old.kind,&new.kind){(Enum{variants:a},Enum{variants:b})=>a.iter().all(|v|b.contains(v)),_=>old.kind==new.kind},
    }
}

//! Carry authored guards; never invent a value for a newly optional field.
// @ref LLP 3000.000.000#20-visible-rollback-views-roll — compilation gates automatic registration.
use crate::{ast::Stmt, parser, print};
use std::collections::BTreeSet;

pub(super) fn carry(
    sources: &mut [(String, String)],
    current: &[(String, String)],
    _affected: &BTreeSet<String>,
    changes: &mut Vec<String>,
) -> (Vec<crate::Diagnostic>,BTreeSet<String>) {
    let mut diagnostics = Vec::new();
    let mut fallback=BTreeSet::new();
    let current_documents = current
        .iter()
        .map(|(file, text)| parser::parse_document(file, text).0)
        .collect::<Vec<_>>();
    let target_documents = sources
        .iter()
        .map(|(file, text)| parser::parse_document(file, text).0)
        .collect::<Vec<_>>();
    let helper_signatures = |docs: &[crate::ast::Document]| {
        docs.iter()
            .flat_map(|d| {
                d.functions.iter().map(move |f| {
                    let text = d
                        .source
                        .declarations
                        .get(&f.site.to_string())
                        .cloned()
                        .unwrap_or_default();
                    let (tokens, _) = crate::lexer::lex(&d.file, &text);
                    let calls = tokens
                        .windows(2)
                        .filter(|p| p[0].kind == crate::lexer::Kind::Word && p[1].value == "(")
                        .map(|p| p[0].value.clone())
                        .collect::<Vec<_>>();
                    (
                        f.name.clone(),
                        (
                            tokens
                                .into_iter()
                                .map(|t| (t.kind, t.value))
                                .collect::<Vec<_>>(),
                            calls,
                        ),
                    )
                })
            })
            .collect::<std::collections::BTreeMap<_, _>>()
    };
    let current_helpers = helper_signatures(&current_documents);
    let target_helpers = helper_signatures(&target_documents);
    let mut next=std::collections::BTreeMap::new();
    for doc in &current_documents {
        for p in &doc.programs {next.insert(p.name.clone(),p.clone());}
        for(name,target,_)in &doc.aliases {if let Some(p)=doc.programs.iter().find(|p|p.name==*target){let mut exported=p.clone();exported.name=name.clone();next.insert(name.clone(),exported);}}
    }
    for (file, text) in sources {
        let doc = parser::parse_document(file, text).0;
        let mut edits = Vec::new();
        for old in doc.programs {
            let Some(new) = next.get(&old.name) else {
                continue;
            };
            // The whole selected source must retain current matching guards,
            // including changes unrelated to the column that triggered a view.
            let guards = old
                .prelude
                .iter()
                .chain(&old.body)
                .filter_map(guard)
                .collect::<BTreeSet<_>>();
            let added = new
                .prelude
                .iter()
                .chain(&new.body)
                .filter_map(guard)
                .filter(|g| !guards.contains(g))
                .collect::<BTreeSet<_>>();
            let bindings = |p: &crate::ast::ProgramDecl| {
                p.prelude
                    .iter()
                    .chain(&p.body)
                    .filter_map(|s| {
                        if let Stmt::Bind { names, value, .. } = s {
                            Some((names, value))
                        } else {
                            None
                        }
                    })
                    .flat_map(|(names, value)| {
                        names.iter().enumerate().map(move |(at, name)| {
                            (
                                name.clone(),
                                (format!("{at}:{}", print::expression(value)), value.clone()),
                            )
                        })
                    })
                    .collect::<std::collections::BTreeMap<_, _>>()
            };
            let old_bindings = bindings(&old);
            let new_bindings = bindings(new);
            let mut dependent = BTreeSet::new();
            let mut helpers = BTreeSet::new();
            let collect = |expr: &crate::ast::Expr,
                           names: &mut BTreeSet<String>,
                           helpers: &mut BTreeSet<String>| {
                let mut expr = expr.clone();
                crate::bind::intent_walk::expression(&mut expr, &mut |e| match e {
                    crate::ast::Expr::Name { name, .. } => {
                        names.insert(name.clone());
                    }
                    crate::ast::Expr::Call { name, .. } => {
                        helpers.insert(name.clone());
                    }
                    _ => (),
                });
            };
            for s in new.prelude.iter().chain(&new.body) {
                if let Stmt::Require { condition, .. } = s {
                    collect(condition, &mut dependent, &mut helpers);
                }
            }
            let mut visited = BTreeSet::new();
            let mut unsafe_binding = false;
            while let Some(name) = dependent.pop_first() {
                if !visited.insert(name.clone()) {
                    continue;
                }
                if let Some((signature, value)) = new_bindings.get(&name) {
                    if old_bindings.get(&name).map(|(s, _)| s) != Some(signature)
                        || has_write(value)
                    {
                        unsafe_binding = true;
                    }
                    collect(value, &mut dependent, &mut helpers);
                } else if old_bindings.contains_key(&name) {
                    unsafe_binding = true;
                }
            }
            // Reused authored workflows retain their complete helper meaning,
            // including helpers feeding writes, returns and system scopes.
            let mut workflow = crate::ast::Document::default();
            workflow.programs.push(new.clone());
            crate::bind::intent_walk::document(&mut workflow, &mut |e| {
                if let crate::ast::Expr::Call { name, .. } = e {
                    helpers.insert(name.clone());
                }
            });
            let mut visited = BTreeSet::new();
            let mut unsafe_helper = false;
            while let Some(name) = helpers.pop_first() {
                if !visited.insert(name.clone()) {
                    continue;
                }
                if current_helpers.get(&name).map(|s| &s.0)
                    != target_helpers.get(&name).map(|s| &s.0)
                {
                    unsafe_helper = true;
                }
                if let Some((_, calls)) = current_helpers.get(&name) {
                    helpers.extend(calls.iter().cloned());
                }
            }
            let nested = |p: &crate::ast::ProgramDecl| {
                p.body
                    .iter()
                    .filter(|s| matches!(s, Stmt::If { .. } | Stmt::For { .. } | Stmt::As { .. }))
                    .map(|s| normalized_statement(s))
                    .collect::<Vec<_>>()
            };
            if unsafe_binding || unsafe_helper || nested(&old) != nested(new) {
                fallback.insert(old.name.clone());
                diagnostics.push(crate::diag::type_error(&old.site,format!("rollback guard dependency closure for {} differs; an old binding/helper or control-flow path cannot establish the current guard",old.name),"retain a compatible guarded old workflow or declare intentional hiding of the unrepresentable values").guide("deploy"));
                continue;
            }
            if added.is_empty() {
                continue;
            }
            // Keep the old workflow; add current requirements after its reads,
            // immediately before the first effect/return. Undefined or changed
            // bindings fail ordinary compilation rather than guessing policy.
            let at = old
                .body
                .iter()
                .find(|s| {
                    !matches!(s, Stmt::Require { .. })
                        && !matches!(s,Stmt::Bind{value,..} if !has_write(value))
                })
                .map(stmt_site)
                .unwrap_or(old.site.line + 1);
            let offset = text
                .split_inclusive('\n')
                .take(at.saturating_sub(1) as usize)
                .map(str::len)
                .sum::<usize>();
            edits.push((
                offset,
                added.iter().map(|g| format!("  {g}\n")).collect::<String>(),
            ));
            for g in added {
                changes.push(format!("operation {}: carried current {g}", old.name));
            }
        }
        edits.sort_by_key(|(at, _)| std::cmp::Reverse(*at));
        for (at, guard) in edits {
            text.insert_str(at, &guard);
        }
    }
    (diagnostics,fallback)
}
fn guard(s: &Stmt) -> Option<String> {
    match s {
        Stmt::Require {
            condition,
            code,
            message,
            ..
        } => Some(format!(
            "require {} else {}{}",
            print::expression(condition),
            code,
            message
                .as_ref()
                .map(|m| format!(" '{}'", m.replace('\\', "\\\\").replace('\'', "\\'")))
                .unwrap_or_default()
        )),
        _ => None,
    }
}
fn stmt_site(s: &Stmt) -> u32 {
    match s {
        Stmt::As { site, .. }
        | Stmt::Bind { site, .. }
        | Stmt::Expr { site, .. }
        | Stmt::Require { site, .. }
        | Stmt::Return { site, .. }
        | Stmt::Emit { site, .. }
        | Stmt::If { site, .. }
        | Stmt::For { site, .. }
        | Stmt::Break { site }
        | Stmt::Continue { site } => site.line,
    }
}

fn has_write(e: &crate::ast::Expr) -> bool {
    let mut e = e.clone();
    let mut writes = false;
    crate::bind::intent_walk::expression(&mut e, &mut |e| {
        if matches!(e, crate::ast::Expr::Write { .. }) {
            writes = true;
        }
    });
    writes
}

fn normalized_statement(s: &Stmt) -> String {
    match s {
        Stmt::As {
            principal, body, ..
        } => format!(
            "as {} {:?}",
            print::expression(principal),
            body.iter().map(normalized_statement).collect::<Vec<_>>()
        ),
        Stmt::Bind { names, value, .. } => format!("bind {names:?} {}", print::expression(value)),
        Stmt::Expr { value, .. } => format!("expr {}", print::expression(value)),
        Stmt::Require {
            condition,
            code,
            message,
            ..
        } => format!(
            "require {} {code:?} {message:?}",
            print::expression(condition)
        ),
        Stmt::Return { value, .. } => format!("return {}", print::expression(value)),
        Stmt::Emit {
            effect,
            payload,
            once,
            ..
        } => format!(
            "emit {effect} {} {:?}",
            print::expression(payload),
            once.as_ref().map(print::expression)
        ),
        Stmt::If {
            condition,
            then_body,
            else_body,
            ..
        } => format!(
            "if {} {:?} {:?}",
            print::expression(condition),
            then_body
                .iter()
                .map(normalized_statement)
                .collect::<Vec<_>>(),
            else_body
                .iter()
                .map(normalized_statement)
                .collect::<Vec<_>>()
        ),
        Stmt::For {
            name, domain, body, ..
        } => format!(
            "for {name} {} {:?}",
            print::expression(domain),
            body.iter().map(normalized_statement).collect::<Vec<_>>()
        ),
        Stmt::Break { .. } => "break".into(),
        Stmt::Continue { .. } => "continue".into(),
    }
}

pub(super) fn verify(dir: &Path, view: &View) -> Result<(), String> {
    if view.parent == view.release || is_view(dir, view.parent) {

        .sources()

        let risks = super::rollback_foresight::risks(&bound.schema, &old.schema, target, dir);
        let widening=snapback4_lang::release_bridge::requires_view(&original,&current);
        let defaulting=super::rollback_foresight::missing_defaults(&bound.schema,&old.schema)?;
        if !defaulting.is_empty(){return Err(format!("E_ADOPT: rollback to release {target} would misrepresent a removed defaulted column {}; declare a truthful cross-table down before deploy: {}",defaulting.join(", "),risks.join("; ")))}

        // or native-only repair must not restore an older executable behind
        // unchanged intent declarations. Metadata-only registration is excluded
        // by the shared executable digest, keeping equal-source no-ops finite.
        let closure_changed = !bound.invariants.is_empty()
            && crate::release_files::executable_digest(&old_root)?
                != crate::release_files::executable_digest(stage)?;
        if !widening && !closure_changed {
            continue;
        }
        if crate::dev_worker::source(&Project {root:stage.into()}).is_some()
            || !crate::dev_worker::declared(&Project {root:stage.into()})?.is_empty()
            || crate::dev_worker::source(&Project {
            root: old_root.clone(),
        })
        .is_some()
            || !crate::dev_worker::declared(&Project {
                root: old_root.clone(),
            })?
            .is_empty()
        {
            return Err("E_ADOPT: a generated rollback view needs an authored worker bundle; retain a compatible guarded source".into());
        }
        let hides = bound
            .rollback_hides
            .iter()
            .filter(|h| h.release == target)
            .collect::<Vec<_>>();
        let mut source = current.clone();
        // A typed null acknowledgment retains this one old nonnullable domain;
        // all other widened columns still receive the visible bridge union.
        for hide in &hides {
            let old_kind = old
                .schema
                .table(&hide.table)
                .and_then(|t| t.columns.get(&hide.column))
                .ok_or("rollback hides names no predecessor column")?;
            if bound.schema.table_renames.contains_key(&hide.table)
                || bound
                    .schema
                    .asset_renames
                    .get(&hide.table)
                    .is_some_and(|m| m.contains_key(&hide.column))
            {
                return Err("rollback hides requires the same table/column spelling in the original release".into());
            }
            if old_kind.is_optional() {
                return Err("rollback hides requires a nonnullable predecessor column".into());
            }
            for (file, text) in &mut source {
                let doc = snapback4_lang::parse_document(file, text).0;
                if let Some(column) = doc
                    .tables
                    .iter()
                    .find(|t| t.name == hide.table)
                    .and_then(|t| t.columns.iter().find(|c| c.name == hide.column))
                {
                    if let Some((start, end)) =
                        snapback4_lang::release_bridge::type_span(file, text, column)
                    {
                        let scalar = column.type_.required();
                        text.replace_range(
                            start..end,
                            &snapback4_lang::print::scalar_type(&scalar),
                        );
                    }
                }
            }
        }
        let bridge = snapback4_lang::release_bridge::loosen(
            &original,
            &source,
        );
        if !bridge.diagnostics.is_empty() {
            return Err(format!(
                "E_ADOPT: rollback view for release {target} does not compile; retain a truthful guarded old workflow or declare its intentional hiding: {}",
                bridge
                    .diagnostics
                    .iter()
                    .map(ToString::to_string)
                    .collect::<Vec<_>>()
                    .join("; ")
            ));
        }
        let compiled = snapback4_lang::compile(&bridge.sources);
        let mut aliases:std::collections::BTreeMap<String,String> = bound
            .schema
            .table_renames
            .iter()
            .filter(|(_, old)| compiled.schema.tables.contains_key(*old))
            .map(|(new, old)| (new.clone(), old.clone()))
            .collect();
        for(new,prior)in &compiled.schema.table_renames{if bound.schema.tables.contains_key(prior)&&!bound.schema.tables.contains_key(new){aliases.insert(prior.clone(),new.clone());}}
        let current = bound
            .schema
            .with_table_names(&aliases)
            .map_err(|e| e.to_string())?;
        let mut columns=std::collections::BTreeMap::new();
        for(name,table)in &compiled.schema.tables{if let Some(next)=current.table(name){
            let mut map=std::collections::BTreeMap::new();
            for(new,prior)in current.asset_renames.get(name).into_iter().flatten(){if next.columns.contains_key(new)&&table.columns.contains_key(prior)&&!table.columns.contains_key(new){map.insert(new.clone(),prior.clone());}}
            for(new,prior)in compiled.schema.asset_renames.get(name).into_iter().flatten(){if next.columns.contains_key(prior)&&table.columns.contains_key(new)&&!next.columns.contains_key(new){map.insert(prior.clone(),new.clone());}}
            if !map.is_empty(){columns.insert(name.clone(),map);}
        }}
        let current=current.with_column_names(&columns);
        for (name, table) in &compiled.schema.tables {
            if let Some(next) = current.table(name) {
                use snapback4_core::schema::Constraint;
                let hard=|c:&Constraint|!matches!(c,Constraint::Text{..}|Constraint::Unique{..}|Constraint::Capabilities(_)|Constraint::Recorded);
                let column=|field:&String|field.clone();
                for c in next.constraints.iter().filter(|c|hard(c)) {
                    let mut constraint=c.clone();
                    match &mut constraint {Constraint::Cap{by,..}|Constraint::Immutable{fields:by}=>for f in by{*f=column(f)},_=>()}
                    if !table.constraints.contains(&constraint){return Err(format!("E_ADOPT: rollback view for release {target} cannot preserve current constraint on {name}; author a compatible guarded rollback workflow"))}
                }
                for index in next.indexes.values().filter(|i|i.unique){
                    let mut components=index.components.clone();for part in &mut components{if let snapback4_core::schema::IndexComponent::Column(c)=part{*c=column(c);}}
                    if !table.indexes.values().any(|i|i.unique&&i.components==components){return Err(format!("E_ADOPT: rollback view for release {target} cannot preserve a current unique key on {name}; author a compatible guarded rollback workflow"))}
                }
                let on_delete=next.on_delete.iter().map(|(field,policy)|(column(field),policy.clone())).collect::<std::collections::BTreeMap<_,_>>();
                if table.expire!=next.expire||table.not_before!=next.not_before||table.through!=next.through||table.on_delete!=on_delete||table.sync!=next.sync||table.leave!=next.leave {
                    return Err(format!("E_ADOPT: rollback view for release {target} cannot preserve current lifetime or sync audience on {name}; author a compatible guarded rollback workflow"))
                }
                if ![
                    (&table.rules.read, &next.rules.read),
                    (&table.rules.insert, &next.rules.insert),
                    (&table.rules.update, &next.rules.update),
                    (&table.rules.delete, &next.rules.delete),
                ]
                .iter()
                .all(|(view, current)| view == current)
                    || table.columns.keys().any(|c| {
                        table.column_reads.get(c) != next.column_reads.get(c)
                            || table.written_by.get(c) != next.written_by.get(c)
                    })
                {
                    return Err(format!(
                        "E_ADOPT: rollback view for release {target} cannot preserve current authority on {name}; retain an authored compatible guarded old workflow before deploying this widening"
                    ));
                }
            }
        }
        if compiled.programs.iter().any(|p| p.program.native) {
            return Err("E_ADOPT: a generated rollback view needs an authored native bundle; retain a compatible guarded source".into());
        }
        id = id.checked_add(1).ok_or("release identity exhausted")?;
        let root = stage.join("rollback-views").join(id.to_string());
        fs::create_dir_all(root.join("snapback")).map_err(|e| e.to_string())?;
        for (name, text) in &bridge.sources {
            let path = root.join("snapback").join(name);
            if let Some(parent) = path.parent() {
                fs::create_dir_all(parent).map_err(|e| e.to_string())?;
            }
            fs::write(path, text).map_err(|e| e.to_string())?;
        }
        let mut view = View {
            target,
            release: id,
            parent,
            digest: String::new(),
            changes: bridge.changes,
            hides: hides
                .iter()
                .map(|h| format!("{} where {} = null because {}", h.table, h.column, h.reason))
                .collect(),
        };
        fs::write(
            root.join("device-compatibility.json"),
            serde_json::to_vec_pretty(&snapback4_core::device_protocol::compatibility(
                &compiled.schema.public_projection(),
            ))
            .unwrap(),
        )
        .map_err(|e| e.to_string())?;
        view.digest = crate::release_files::executable_digest(&root)?;
        fs::write(
            root.join("rollback-view.json"),
            serde_json::to_vec_pretty(&view).unwrap(),
        )
        .map_err(|e| e.to_string())?;
        views.push(view);
    }
    fs::write(
        stage.join("rollback-views.json"),
        serde_json::to_vec_pretty(&views).unwrap(),
    )
    .map_err(|e| e.to_string())?;
    Ok(views)
}
pub(super) struct Installed {
    dir: PathBuf,
    parent: u64,
    ids: Vec<u64>,
    pub pending: bool,
}
impl Drop for Installed {
    fn drop(&mut self) {
        if !self.pending && super::releases::selectable(&self.dir, self.parent).is_err() {
            for id in &self.ids {
                let _ = fs::remove_dir_all(self.dir.join("releases").join(id.to_string()));
            }
        }
    }
}
pub(super) fn install(
    dir: &Path,
    stage: &Path,
    parent: u64,
    views: &[View],
) -> Result<Installed, String> {
    let mut installed = Installed {
        dir: dir.into(),
        parent,
        ids: Vec::new(),
        pending: false,
    };
    for v in views {
        let from = stage.join("rollback-views").join(v.release.to_string());
        crate::release_files::durable(&from)?;
        fs::rename(&from, dir.join("releases").join(v.release.to_string()))
            .map_err(|e| e.to_string())?;
        installed.ids.push(v.release);
    }
    if stage.join("rollback-views").exists() {
        fs::remove_dir(stage.join("rollback-views")).map_err(|e| e.to_string())?;
    }
    Ok(installed)
}

pub(super) fn needs_registration(dir:&Path,matched:Option<u64>,prepared:&[View])->Result<bool,String>{
    let held=matched.map(|n|manifest(&dir.join("releases").join(n.to_string()))).transpose()?.unwrap_or_default();
    for view in &held{verify(dir,view)?;}
    Ok(held.len()!=prepared.len()||prepared.iter().any(|v|!held.iter().any(|old|old.target==v.target&&old.digest==v.digest&&old.hides==v.hides)))
}

//! @ref LLP 3000.000.000.000#11-old-client-operation-contracts-rel5 — operation evolution lowers into the existing shared IR.
use crate::{ast::*, diag::{Diagnostic, Fallible, type_error, unknown}};
use snapback4_core::{Schema, ir::ArgType, value::Value};

fn modified(args: &[ArgDecl]) -> bool { args.iter().any(|a|a.default.is_some() || a.alias.is_some() || a.deprecated) }
fn invalid(site: &crate::Site) -> Diagnostic {
    type_error(site,"argument aliases/defaults/deprecation are supported on non-native queries and mutations only","keep ordinary parameters here; use a query/mutation for a release-compatible wrapper").guide("deploy")
}
pub(super) fn expand(documents: &mut [Document], diagnostics: &mut Vec<Diagnostic>) {
    for doc in documents {
        for p in &doc.programs {
            if (p.native || p.kind==ProgramKind::Job) && modified(&p.args) {diagnostics.push(invalid(&p.site));}
        }
        for a in &doc.actions {if modified(&a.args) {diagnostics.push(invalid(&a.site));}}
        for f in &doc.functions {if modified(&f.args) {diagnostics.push(invalid(&f.site));}}
        let authored=doc.programs.clone();
        for (name,target,site) in &doc.aliases {
            let Some(p)=authored.iter().find(|p|p.name==*target && !p.native && p.kind!=ProgramKind::Job) else {
                diagnostics.push(unknown(site,format!("alias {name} needs an authored query/mutation target {target} in this module"),"target a non-native query or mutation; aliases cannot chain").guide("deploy"));continue;
            };
            let mut alias=p.clone();alias.name=name.clone();alias.site=site.clone();doc.programs.push(alias);
        }
    }
}

pub(super) fn argument(arg: &ArgDecl, schema: &Schema) -> Fallible<(ArgType,Option<Value>)> {
    if arg.alias.is_some() && arg.deprecated {
        return Err(type_error(&arg.site,format!("argument {} cannot be both an alias target and deprecated",arg.name),"keep the canonical argument with `alias oldName`; remove deprecated").guide("deploy"));
    }
    let kind=crate::types::scalar_to_arg(&arg.type_);
    let bad=||type_error(&arg.site,format!("default for {} must be a fitting non-null scalar literal and cannot combine with deprecated",arg.name),"use default <literal>, or mark the unused parameter deprecated").guide("deploy");
    let value=if let Some(expr)=&arg.default {
        if arg.deprecated {return Err(bad());}
        let value=match expr {
            Expr::Literal {value:Literal::Bool(v),..}=>Value::Bool(*v),
            Expr::Literal {value:Literal::Str(v),..}=>match arg.type_.kind {TypeKind::Ref{..}|TypeKind::Id|TypeKind::Principal=>Value::Id(v.clone()),_=>Value::String(v.clone())},
            Expr::Literal {value:Literal::Number(v),..}=>Value::Int(v.parse().map_err(|_|bad())?),
            Expr::Unary {op,value,..} if op=="-" => match value.as_ref() {Expr::Literal{value:Literal::Number(v),..}=>Value::Int(format!("-{v}").parse().map_err(|_|bad())?),_=>return Err(bad())},
            _=>return Err(bad()),
        };
        snapback4_core::engine::validate::validate_arg(schema,&kind,&value,&arg.name).map_err(|_|bad())?;
        Some(value)
    } else {None};
    let wire=if arg.default.is_some() || arg.deprecated {
        match kind {ArgType::Optional(_)=>kind,_=>ArgType::Optional(Box::new(kind))}
    } else {kind};
    Ok((wire,value))
}

//! Column and parameter types, including precise unsupported-default repairs.
use super::Parser;
use crate::{ast::*, diag::{Fallible, Site, syntax, syntax_with}, lexer::Kind, print};
impl Parser {
    pub(super) fn parse_args(&mut self) -> Fallible<Vec<ArgDecl>> {
        self.take("(")?;
        let mut args = Vec::new();
        while !self.at(")") {
            let name = self.peek().clone();
            let binding = self.local_name()?;
            self.take(":")?;
            let type_site = self.peek().site.clone();
            let type_ = self.parse_type(false).map_err(|mut d| {
                if d.mechanical && d.message.starts_with("once is a column modifier") {
                    // Nested element types need a whole-list rewrite, not an argument fragment.
                    if d.site == type_site { d.site = name.site.clone(); d.rewrite = format!("{binding}: {}", d.rewrite); }
                    else { d.mechanical = false; }
                }
                d
            })?;
            let mut default = None;
            let mut alias = None;
            let mut deprecated = false;
            loop {
                if self.maybe("default") {
                    if default.is_some() { return Err(syntax(&self.peek().site, "an argument has at most one default")); }
                    default = Some(self.expression(0, &[",", ")", "alias", "deprecated"])?);
                } else if self.maybe("alias") {
                    if alias.is_some() { return Err(syntax(&self.peek().site, "an argument has at most one alias")); }
                    alias = Some(self.local_name()?);
                } else if self.maybe("deprecated") {
                    if deprecated { return Err(syntax(&self.peek().site, "an argument is marked deprecated at most once")); }
                    deprecated = true;
                } else { break; }
            }
            if self.at("=") { return Err(syntax(&self.peek().site,"argument defaults use `default <literal>`")); }
            args.push(ArgDecl { name: binding, type_, default, alias, deprecated, site: name.site });
            if !self.maybe(",") {
                break;
            }
        }
        self.take(")")?;
        Ok(args)
    }

    pub(super) fn parse_type(&mut self, column: bool) -> Fallible<ScalarType> {
        let site = self.peek().site.clone();
        let mut kind;
        if self.maybe("[") {
            let item = self.parse_type(column)?;
            self.take("]")?;
            // @ref LLP 3000.007#5.4 — list ranges share text's inclusive spelling.
            let bounds_error = |site: &Site| syntax_with(site, "list bounds require <=N or an inclusive a..b range", "write [T] <=N or [T] a..b, for example [int] 1..8").guide("types");
            let number = |parser: &mut Self| -> Fallible<u32> {
                if parser.at("-") {
                    return Err(syntax(&parser.peek().site, "list bounds must be nonnegative integers"));
                }
                if !parser.at_kind(Kind::Number) { return Err(bounds_error(&parser.peek().site)); }
                let token = parser.take_kind(Kind::Number)?;
                if token.value.contains('.') {
                    return Err(syntax(&token.site, "list bounds must be integers; fractional bounds are not allowed"));
                }
                token.value.parse::<u32>().map_err(|_| syntax(&token.site, "list bounds must be integers from 0 through 4294967295"))
            };
            let (min, max) = if self.maybe("<=") {
                (0, number(self)?)
            } else {
                let min = number(self)?;
                if !self.maybe(".") || !self.maybe(".") { return Err(bounds_error(&self.peek().site)); }
                (min, number(self)?)
            };
            if min > max {
                return Err(syntax(&site, "list minimum must not exceed its maximum"));
            }
            if item.optional {
                return Err(syntax(&site, "list element types cannot be optional"));
            }
            kind = TypeKind::List { item: Box::new(item), min, max };
        } else {
            let name = self.word()?;
            kind = match name.value.as_str() {
                "text" => {
                    let (min, max) = if self.maybe("<=") {
                        (None, self.number()?)
                    } else if self.at_kind(Kind::Number) {
                        let min = self.number()?;
                        self.take(".")?;
                        self.take(".")?;
                        (Some(min), self.number()?)
                    } else {
                        // The default is normalized here; every downstream
                        // validator and resource calculation sees the same bound.
                        (None, 10_000)
                    };
                    let mut format = None;
                    if self.maybe("format") {
                        let value = self.word()?;
                        if !matches!(value.value.as_str(), "handle" | "url" | "email") {
                            let bound = match min {
                                Some(min) => format!("{min}..{max}"),
                                None => format!("<={max}"),
                            };
                            return Err(syntax_with(&value.site, format!("unknown text format {}", value.value), format!("use text {bound} format handle, text {bound} format url, or text {bound} format email")));
                        }
                        format = Some(value.value);
                    }
                    TypeKind::Text { min, max, format }
                }
                "decimal" => TypeKind::Decimal { scale: self.scale("decimal 12")? },
                "money" => {
                    let currency = self.word()?;
                    if currency.value.len() != 3 || !currency.value.bytes().all(|b| b.is_ascii_uppercase()) {
                        return Err(syntax_with(&currency.site, "money currency must be exactly three uppercase letters", "use money USD or money USD 2"));
                    }
                    let scale = if self.at_kind(Kind::Number) { self.scale(&format!("money {} 12", currency.value))? } else { 2 };
                    TypeKind::Money { currency: currency.value, scale }
                }
                "json" => {
                    self.take("<=")?;
                    TypeKind::Json { max: self.number()? }
                }
                "bytes" => {
                    self.take("<=")?;
                    TypeKind::Bytes { max: self.number()? }
                }
                "image" => {
                    if !self.maybe("<=") { return Err(syntax_with(&site, "image needs a maximum upload size in bytes", "write `image <=5MB` (5,000,000 bytes), optionally followed by once or ?").guide("types")); }
                    TypeKind::Image { max: self.media_size()?, once: false }
                }
                "video" => {
                    self.take("<=")?;
                    TypeKind::Video { max: self.media_size()? }
                }
                // `file` is a type only before `<=`: a table named `file` stays a reference.
                // @ref LLP 3000.038#1 — the allow-list names only types the server can verify against the bytes.
                "file" if self.at("<=") => {
                    self.take("<=")?;
                    let max = self.number()?;
                    let mut types = Vec::new();
                    if self.maybe("(") {
                        while !self.at(")") {
                            let value = self.take_kind(Kind::Str)?;
                            if let Some(why) = snapback4_core::assets::file_type_refusal(&value.value) {
                                return Err(syntax_with(&value.site, why, format!("file <={max} ('application/pdf', 'text/plain')")).guide("assets"));
                            }
                            if types.contains(&value.value) {
                                return Err(syntax_with(&value.site, format!("'{}' is listed twice", value.value), "list each media type once"));
                            }
                            types.push(value.value);
                            if !self.maybe(",") { break; }
                        }
                        self.take(")")?;
                        if types.is_empty() {
                            return Err(syntax_with(&site, "a file allow-list needs at least one media type", format!("file <={max} ('application/pdf'), or file <={max} for every verifiable type")).guide("assets"));
                        }
                    }
                    TypeKind::File { max, types }
                }
                "cursor" => TypeKind::Cursor,
                "id" => TypeKind::Id,
                "principal" => TypeKind::Principal,
                "int" if self.at_kind(Kind::Number) || matches!(self.peek().value.as_str(), "<=" | ">=" | "<" | ">" | "-" | ".") => {
                    let (min, max) = self.int_range(&name.site)?;
                    TypeKind::IntRange { min, max }
                }
                "int" => TypeKind::Int,
                "time" => TypeKind::Time,
                "bool" => TypeKind::Bool,
                "enum" => {
                    self.take("(")?;
                    let mut variants = Vec::new();
                    while !self.at(")") {
                        variants.push(self.take_kind(Kind::Str)?.value);
                        if !self.maybe(",") {
                            break;
                        }
                    }
                    self.take(")")?;
                    if variants.is_empty() {
                        return Err(syntax_with(&site, "enum needs at least one variant", "enum('draft', 'published')"));
                    }
                    TypeKind::Enum { variants }
                }
                other => TypeKind::Ref { table: other.to_owned() },
            };
        }
        if self.maybe("once") {
            if !column {
                let optional = self.maybe("?");
                return Err(syntax_with(&site, "once is a column modifier, not an argument constraint; declare it on the owning table column", print::scalar_type(&ScalarType { kind, optional })).mechanical());
            }
            match &mut kind {
                TypeKind::Image { once, .. } => *once = true,
                TypeKind::Video { .. } => return Err(syntax_with(&site, "once is only specified for image columns; video ranges are unspecified", "use photo: image <=10000000 once, or remove once from the video column")),
                TypeKind::File { .. } => return Err(syntax_with(&site, "once is only specified for image columns; a downloaded file is saved by the browser, so one view cannot be kept", "remove once from the file column; use image <=10000000 once for one-view photos").guide("assets")),
                _ => return Err(syntax_with(&site, "once is only specified for image columns", "remove once from this column; use image <=10000000 once for one-use images")),
            }
        }
        let optional = self.maybe("?");
        Ok(ScalarType { kind, optional })
    }

    pub(super) fn media_size(&mut self) -> Fallible<u32> {
        let site = self.peek().site.clone();
        let size = self.number()?;
        let multiplier = match self.peek().value.as_str() { "KB" => 1_000, "MB" => 1_000_000, "GB" => 1_000_000_000, _ => return Ok(size) };
        self.position += 1;
        size.checked_mul(multiplier).ok_or_else(|| syntax_with(&site, "upload bound exceeds 4,294,967,295 bytes", "use image <=5MB or a byte count fitting u32").guide("types"))
    }
}

//! Immutable release custody: digest the required tree and flush it before DB selection.
// @ref LLP 3000.000.000#5-hot-activation-and-ownership — source durability precedes the database's release pointer.
use sha2::{Digest, Sha256};
use std::{
    fs,
    io::{Read, Write},
    path::{Path, PathBuf},
};

fn files(root: &Path, path: &Path, out: &mut Vec<PathBuf>) -> Result<(), String> {
    let mut entries = fs::read_dir(path)
        .map_err(|e| e.to_string())?
        .map(|e| e.map(|e| e.path()).map_err(|e| e.to_string()))
        .collect::<Result<Vec<_>, _>>()?;
    entries.sort();
    for path in entries {
        let relative = path.strip_prefix(root).map_err(|e| e.to_string())?;
        if relative
            .components()
            .next()
            .is_some_and(|c| c.as_os_str() == ".snapback4")
        {
            continue;
        }
        let m = fs::symlink_metadata(&path).map_err(|e| e.to_string())?;
        if m.is_dir() {
            files(root, &path, out)?;
        } else if m.is_file() {
            out.push(path);
        } else {
            return Err(format!(
                "immutable release artifact must be a real directory or regular file: {}",
                path.display()
            ));
        }
    }
    Ok(())
}

    bundle_digest(root, false)

pub(crate) fn executable_digest(root: &Path) -> Result<String, String> {
    bundle_digest(root, true)
}
fn bundle_digest(root: &Path, executable: bool) -> Result<String, String> {
    let mut paths = Vec::new();
    files(root, root, &mut paths)?;
    let mut hash = Sha256::new();
    for path in paths {
        let name = path.strip_prefix(root).unwrap().to_string_lossy();
        // @ref LLP 3000.000.000.000.000#4-cli-surface-contracts — metadata does not change executable release identity.
        if name == ".pending-adoption" || name == "provenance.json" {continue;}
        // @ref LLP 3000.000.000#16-release-reactivation-and-rollback-foresight-rel5 — compiler and host-location reports are not executable content.
        // Keep both reports in control-request integrity; only release identity
        // ignores them. Activation recompiles and checks the retained source normally.
        if executable && (matches!(name.as_ref(), "device-compatibility.json" | "worker-package.json" | "rollback-view.json" | "rollback-views.json") || name.starts_with("rollback-views/")) {continue;}
        hash.update((name.len() as u64).to_le_bytes());
        hash.update(name.as_bytes());
        #[cfg(unix)]
        use std::os::unix::fs::OpenOptionsExt;
        let mut options = fs::OpenOptions::new();
        options.read(true);
        #[cfg(unix)]
        options.custom_flags(libc::O_NOFOLLOW);
        let mut file = options.open(&path).map_err(|e| e.to_string())?;
        hash.update(
            file.metadata()
                .map_err(|e| e.to_string())?
                .len()
                .to_le_bytes(),
        );
        let mut buffer = [0u8; 16384];
        loop {
            let n = file.read(&mut buffer).map_err(|e| e.to_string())?;
            if n == 0 {
                break;
            }
            hash.update(&buffer[..n]);
        }
    }
    Ok(format!("{:x}", hash.finalize()))
}
pub(crate) fn durable(root: &Path) -> Result<(), String> {
    let mut paths = Vec::new();
    files(root, root, &mut paths)?;
    for path in paths {
        fs::File::open(path)
            .and_then(|f| f.sync_all())
            .map_err(|e| e.to_string())?;
    }
    fn dirs(path: &Path) -> Result<(), String> {
        for e in fs::read_dir(path).map_err(|e| e.to_string())? {
            let e = e.map_err(|e| e.to_string())?;
            if e.file_type().map_err(|e| e.to_string())?.is_dir() {
                dirs(&e.path())?;
            }
        }
        fs::File::open(path)
            .and_then(|f| f.sync_all())
            .map_err(|e| e.to_string())
    }
    dirs(root)
}

pub(crate) fn link_store(root: &Path) -> Result<(), String> {
    let project = crate::project::Project { root: root.into() };
    snapback4_core::private_fs::directory(&project.data_dir()).map_err(|e| e.to_string())?;
    let path = project.store_path();
    #[cfg(unix)]
    {
        let target = Path::new("../../../store.sqlite");
        match fs::symlink_metadata(&path) {
            Ok(m) if m.file_type().is_symlink() => {
                if fs::read_link(&path).map_err(|e| e.to_string())? != target {
                    fs::remove_file(&path).map_err(|e| e.to_string())?;
                    std::os::unix::fs::symlink(target, &path).map_err(|e| e.to_string())?;
                }
            }
            Ok(_) => {
                return Err(format!(
                    "release store path must be a symbolic link: {}",
                    path.display()
                ));
            }
            Err(e) if e.kind() == std::io::ErrorKind::NotFound => {
                std::os::unix::fs::symlink(target, &path).map_err(|e| e.to_string())?
            }
            Err(e) => return Err(e.to_string()),
        }
        Ok(())
    }
    #[cfg(not(unix))]
    Err("serving shared releases requires Unix symbolic links".into())
}

pub(crate) struct Pointer {
    staged: PathBuf,
    dir: PathBuf,
    release: u64,
    previous: Option<u64>,
    rollback: bool,
    ledger: fs::File,
}
impl Pointer {
    pub(crate) fn prepare(
        dir: &Path,
        release: u64,
        previous: Option<u64>,
        rollback: bool,
    ) -> Result<Self, String> {
        let pointer = dir.join("current");
        if pointer.exists() && !pointer.is_file() {
            return Err("current must be a regular release pointer file".into());
        }
        let staged = dir.join(format!(".current-{}-{release}", std::process::id()));
        let file = snapback4_core::private_fs::file(&staged, true).map_err(|e| e.to_string())?;
        let mut prepared = Self {
            staged,
            dir: dir.into(),
            release,
            previous,
            rollback,
            ledger: file,
        };
        writeln!(prepared.ledger, "{release}")
            .and_then(|_| prepared.ledger.sync_all())
            .map_err(|e| e.to_string())?;
        prepared.ledger = fs::OpenOptions::new()
            .create(true)
            .append(true)
            .open(dir.join("releases.jsonl"))
            .map_err(|e| e.to_string())?;
        writeln!(prepared.ledger,"{}",serde_json::json!({"at":snapback4_core::now_ms(),"release":release,"previous":previous,"rollback":rollback,"state":"prepared"})).and_then(|_|prepared.ledger.sync_all()).map_err(|e|e.to_string())?;
        Ok(prepared)
    }
    pub(crate) fn finish(&mut self, generation: u64) -> Result<(), String> {
        fs::rename(&self.staged,self.dir.join("current")).and_then(|_|fs::File::open(&self.dir)?.sync_all()).map_err(|e|format!("release {} committed in database, but current copy failed: {e}; inspect `snapback4 releases {}`",self.release,self.dir.display()))?;
        writeln!(self.ledger,"{}",serde_json::json!({"at":snapback4_core::now_ms(),"release":self.release,"previous":self.previous,"rollback":self.rollback,"generation":generation,"state":"activated"})).and_then(|_|self.ledger.sync_all()).map_err(|e|format!("release committed, but ledger completion failed: {e}"))
    }
}
impl Drop for Pointer {
    fn drop(&mut self) {
        if self.staged.exists() {
            let _ = fs::remove_file(&self.staged);
        }
    }
}

// @ref LLP 3000.000.000#16-release-reactivation-and-rollback-foresight-rel5 — deletion requires a definitive non-commit, never a lost reply.
pub(crate) struct Candidate { dir: PathBuf, release: u64, pub committed: bool, pub pending: bool }
impl Candidate {
    pub(crate) fn new(dir: &Path, release: u64) -> Self { Self {dir:dir.into(),release,committed:false,pending:false} }
}
impl Drop for Candidate {
    fn drop(&mut self) {
        if self.committed { return; }
        if self.pending {eprintln!("release {} result remains unknown; retained source while the server may still commit; inspect releases",self.release);return;}
        let probe=(|| -> Result<bool,String> {
            let store=snapback4_core::store_port::AnyStore::open_read_only(&self.dir.join("store.sqlite")).map_err(|e|e.to_string())?;
            let history:bool=store.has_release_history()?;
            if !history {return Ok(false);}
            store.release_referenced(self.release as i64)
        })();
        match probe {
            Ok(false)=>if let Err(e)=fs::remove_dir_all(self.dir.join("releases").join(self.release.to_string())) {eprintln!("uncommitted release {} cleanup failed: {e}",self.release);},
            Ok(true)=>{},
            Err(e)=>eprintln!("release {} outcome unknown; retained source for recovery; inspect releases: {e}",self.release),
        }
    }
}
