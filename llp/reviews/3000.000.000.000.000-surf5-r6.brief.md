// SQLite READONLY can create WAL sidecars. A stopped store's existing owner
// lock makes immutable inspection safe; a live host supplies its own status.
fn inspect_current(dir: &Path) -> Result<u64, String> {
    use rusqlite::{Connection, OpenFlags, OptionalExtension};
    let database = dir.join("store.sqlite").canonicalize().map_err(|e| e.to_string())?;
    let missing = || format!("no current release in {}; run `snapback4 deploy {}` first", dir.display(), dir.display());
    let owner = match snapback4_core::private_fs::auxiliary_file(&database,
        &snapback4_core::private_fs::owner_path(&database), false, 0) {
        Ok(owner) => owner,
        Err(error) if error.kind() == std::io::ErrorKind::NotFound => {
            // Relocated/legacy deployments may not yet have a sidecar beside
            // the resolved DB. Establish a complete current source first; only
            // a known deployment may let SQLite create its ordinary WAL files.
            return pointer_current(dir, &database);
        },
        Err(error) => return Err(format!("{}; existing ownership sidecar unavailable: {error}", missing())),
    };
    match owner.try_lock() {
        Ok(()) => {},
        Err(std::fs::TryLockError::WouldBlock) => {
            match crate::release_control::request(dir, crate::release_control::Command::Status {after:0, items_after:None, event:None}) {
                Ok(status) => return status["release"].as_u64().filter(|n|*n>0).ok_or_else(missing),
                Err(error) => {
                    // A killed host's watchdogs retain ownership during their
                    // bounded cleanup. Preserve serve's existing restart wait.
                    if !crate::store_owner::stopped_host(&database) {
                        // A second source directory may alias the same DB. Its
                        // source cannot control this host, but its token still
                        // belongs to the resolved database. Establish current
                        // source before permitting ordinary read-only WAL work.
                        return pointer_current(dir, &database).map_err(|_| format!("{}; {error}",missing()));
                    }
                    let deadline=std::time::Instant::now()+std::time::Duration::from_secs(5);
                    loop {
                        match owner.try_lock() {
                            Ok(()) => break,
                            Err(std::fs::TryLockError::WouldBlock) if std::time::Instant::now()<deadline => std::thread::sleep(std::time::Duration::from_millis(25)),
                            _ => return Err(format!("{}; {error}",missing())),
                        }
                    }
                }
            }
        },
        Err(error) => return Err(error.to_string()),
    }
    if !snapback4_core::private_fs::owner_is_named(&database, &owner) {
        return Err("store ownership changed during release inspection; retry".into());
    }
    #[cfg(unix)]
    let bytes = { use std::os::unix::ffi::OsStrExt; database.as_os_str().as_bytes() };
    #[cfg(not(unix))]
    let bytes = database.to_str().ok_or("invalid store path")?.as_bytes();
    let mut uri = String::from("file:");
    for byte in bytes {
        if byte.is_ascii_alphanumeric() || b"/._-".contains(byte) { uri.push(*byte as char); }
        else { uri.push_str(&format!("%{byte:02X}")); }
    }
    let mut wal = database.as_os_str().to_os_string(); wal.push("-wal"); let wal = PathBuf::from(wal);
    let mut shm = database.as_os_str().to_os_string(); shm.push("-shm"); let shm = PathBuf::from(shm);
    let has_wal = match std::fs::metadata(&wal) {
        Ok(meta) => meta.len() > 0,
        Err(e) if e.kind() == std::io::ErrorKind::NotFound => false,
        Err(e) => return Err(e.to_string()),
    };
    if has_wal && !shm.is_file() { return Err("store has a WAL without its shared-memory sidecar; run deploy on this stopped deployment to recover it before inspection".into()); }
    // Immutable SQLite omits WAL reads: use it only for the checkpointed main
    // image. Existing WAL+SHM can be read normally under the same exclusion.
    uri.push_str(if has_wal { "?mode=ro" } else { "?mode=ro&immutable=1" });
    let conn = Connection::open_with_flags(uri, OpenFlags::SQLITE_OPEN_READ_ONLY | OpenFlags::SQLITE_OPEN_URI).map_err(|e|e.to_string())?;
    let release: Option<String> = conn.query_row("SELECT value FROM _meta WHERE key='release'", [], |row|row.get(0)).optional().map_err(|e|e.to_string())?;
    let release = match release {
        Some(value) => value,
        None => std::fs::read_to_string(dir.join("current")).map_err(|_| missing())?,
    };
    release.trim().parse::<u64>().ok().filter(|n|*n>0).ok_or_else(missing)
}

    }
}

            }))).map_err(|e|e.to_string())?;
            crate::release_files::durable(&stage)?;
            std::fs::rename(&stage, &final_dir).map_err(|e| e.to_string())?;
            std::fs::File::open(&releases)
                .and_then(|f| f.sync_all())
                .map_err(|e| e.to_string())?;
            let result=activate(
                &dir,
                _deployment.as_ref(),
                n,
                previous,
                revision,
                staged_bound,
                false,
                options.flags.contains_key("restore-dormant"),
                options.flags.contains_key("allow-feed-adoption"),
                options,
            );
            match result {
                Ok(()) => { if let Err(error)=std::fs::remove_file(final_dir.join(".pending-adoption")) {eprintln!("release {n} committed; pending-marker cleanup failed: {error}; history proves adoption");} }
                Err(error) => {
                    // Keep the source reserved even after a lost reply; committed history is authoritative.
                    let matching_latest=latest(&dir).ok().filter(|id|crate::release_files::digest(&final_dir).ok()==crate::release_files::digest(&releases.join(id.to_string())).ok());
                    let error=if error.contains("dormant names") && matching_latest.is_some() {
                        format!("E_ADOPT: source matches latest; use `snapback4 rollback {} --to latest` to reactivate it; alternatively `snapback4 deploy {} --restore-dormant`. {error}",dir.display(),dir.display())
                    } else {error};
                    if options.json && error.contains("dormant names") {
                        let mut inventory=Vec::new();let mut cursor=None;
                        loop {
                            let page=status(&dir,_deployment.as_ref(),0,cursor,None)?;
                            let items=page["items"].as_array().ok_or("invalid dormant inventory response")?;
                            if items.is_empty() {break;}
                            inventory.extend(items.iter().cloned());
                            cursor=items.last().and_then(|i|i["key"].as_str()).map(str::to_owned);
                            if cursor.is_none() {break;}
                        }
                        println!("{}",serde_json::json!({"ok":false,"code":"E_ADOPT","message":error,"dormant":inventory}));
                    }
                    return Err(error);
                }
            }
            Ok(())
        })(),
    )
}

pub fn rollback(options: &Options) -> i32 {
    report(
        "rollback",
        (|| {
            let dir = existing_deployment(&directory(options)?)?;
            let _command = command_lock(&dir)?;
            let _deployment = target(&dir)?;
            if let Some(owner) = &_deployment { owner.report_reclaim(options.flags.contains_key("verbose")); }
            let held = status(&dir, _deployment.as_ref(), 0, None, None)?;
            let active = if let Some(owner) = _deployment.as_ref() {
                current(&dir, owner)?
            } else {
                held["release"].as_u64()
            }
            .ok_or("no current release")?;
            let revision = held["revision"].as_u64().unwrap_or(0);
            let selected = options
                .flags
                .get("to")
                .map(|n| {
                    if n=="latest" {return latest(&dir).map_err(|_| "no complete latest release");}
                    n.parse::<u64>()
                        .ok()
                        .filter(|n| *n > 0)
                        .ok_or("--to requires a positive immutable release id")
                })
                .transpose()?;
            let previous = selected
                .or_else(|| held["previous_active"].as_u64())
                .ok_or("nothing to roll back to")?;
            selectable(&dir, previous)?;
            let bound = bound_at(dir.join("releases").join(previous.to_string()))?;
            // @ref LLP 3000.000.000#14-wave-14-rollback-narration — resolve bare/latest selection visibly before any preview or live activation.
            let target_line=format!("resolved rollback target: release {previous} (current release {active})");
            if options.json {eprintln!("{target_line}");} else {println!("{target_line}");}
            check_change(&dir, Some(active), &bound, options, Some(previous))?;
            if options.flags.contains_key("check") {
                return super::release_preview::preview(&dir.join("store.sqlite"),dir.join("releases").join(previous.to_string()),previous,options);
            }
            activate(
                &dir,
                _deployment.as_ref(),
                previous,
                Some(active),
                revision,
                bound,
                true,
                true,
                options.flags.contains_key("allow-feed-adoption"),
                options,
            )?;

            Ok(())
        })(),
    )
}

pub fn prune(options: &Options) -> i32 {
    report(
        "prune",
        (|| {
            let dir = existing_deployment(&directory(options)?)?;
            let _command = command_lock(&dir)?;
            let owner = target(&dir)?;
            let held = status(&dir, owner.as_ref(), 0, None, None)?;
            let revision = held["revision"].as_u64().unwrap_or(0);
            let confirm = options.flags.contains_key("allow-destructive");
            let result = match owner.as_ref() {
                Some(owner) => owner
                    .open_store(std::time::Duration::from_secs(5))
                    .and_then(|mut s| s.prune_dormant(confirm, Some(revision)))
                    .map_err(|e| e.to_string())?,
                None => crate::release_control::request(
                    &dir,
                    crate::release_control::Command::Prune {
                        confirm,
                        expected_revision: revision,
                    },
                )?,
            };
            println!("{}", serde_json::to_string_pretty(&result).unwrap());
            // Live responses are bounded. Enumerate the committed audit set (or
            // current preview) to name every item, never silently omit a drop.
            if owner.is_none() {
                let mut after = result["items_after"].as_str().map(str::to_owned);
                while let Some(key) = after {
                    let page = status(&dir, None, 0, Some(key), result["event"].as_u64())?;
                    let items = page["items"].as_array().ok_or("invalid inventory reply")?;
                    if items.is_empty() {
                        break;
                    }
                    println!("{}", serde_json::to_string_pretty(items).unwrap());
                    after = page["items_after"].as_str().map(str::to_owned);
                }
            }
            if !confirm {
                println!(
                    "preview only; to permanently drop this dormant storage: snapback4 prune {} --allow-destructive",
                    dir.display()
                );
            }
            Ok(())
        })(),
    )
}

pub fn serve_release(options: &Options) -> Result<(), String> {
    let requested_port = crate::port_flag::parse("serve", options.flags.get("port").map(String::as_str))?;
    let bind = release_bind(options)?;
    let capacity = crate::host_limits::Capacity::from_options(&options.flags, |name| {
        std::env::var(name).ok()
    })?;
    let dir = existing_deployment(&directory(options)?)?;
    crate::host_limits::initialize();
    let owner = crate::store_owner::host(&dir.join("store.sqlite"))?;
    owner.report_reclaim(options.flags.contains_key("verbose"));
    let current = current(&dir, &owner)?.ok_or("no current release; deploy first")?;
    let release = Project {
        root: dir.join("releases").join(current.to_string())
            .canonicalize().map_err(|e| e.to_string())?,
    };
    crate::release_files::link_store(&release.root)?;
    let (port, listener) = crate::port_flag::listen("serve", requested_port, bind, None)?;
    let worker = options
        .flags
        .get("worker")
        .map(|path| {
            Path::new(path)
                .canonicalize()
                .map_err(|e| format!("--worker {path}: {e}"))
        })
        .transpose()?;
    crate::serve::serve(
        release,
        port,
        crate::serve::Mode::Release {
            expected: current,
            worker,
            owner: std::sync::Arc::new(owner),
        },
        &bind.to_string(),
        listener,
        capacity,
    )
}

pub(crate) fn release_bind(options: &Options) -> Result<std::net::IpAddr, String> {
    let bind = options
        .flags
        .get("bind")
        .map(String::as_str)
        .unwrap_or("127.0.0.1");
    let address: std::net::IpAddr = bind
        .parse()
        .map_err(|_| "--bind must be a literal IPv4 or IPv6 address".to_owned())?;
    if !address.is_loopback() {
        if !options.flags.contains_key("i-understand-cleartext") {
            return Err("non-loopback HTTP sends passwords and bearer tokens in cleartext; use a loopback listener behind a TLS proxy, or explicitly pass --i-understand-cleartext".into());
        }
        eprintln!(
            "WARNING: listening on {address} over CLEARTEXT HTTP. Passwords, bearer tokens, schema and data have no transport protection. Expose only through a TLS proxy on a trusted private hop; see snapback4/README.md."
        );
    }
    Ok(address)
}

//! `snapback4 reset`: the CLI mirror of a running dev's `POST /dev/reset`.
//! It only talks to a loopback dev host; it never opens a store itself.
use serde_json::{Value as Json, json};
use std::path::Path;
use std::time::Duration;

fn usage() -> String { format!("usage: {}", crate::cli_flags::synopsis("reset")) }

pub fn command(options: &crate::verbs::Options) -> i32 {
    match run(options) {
        Ok(reply) => { println!("{reply}"); 0 }
        Err(error) => { eprintln!("{error}"); 1 }
    }
}

/// Add fixture rows through the same dev boundary without clearing existing data.
pub fn seed(options: &crate::verbs::Options) -> i32 {
    let result = (|| -> Result<Json, String> {
        let [path] = options.positionals.as_slice() else { return Err(format!("usage: {}", crate::cli_flags::synopsis("seed"))); };
        let body = Json::Array(rows(Path::new(path))?).to_string();
        let (address, instance) = crate::verbs::registered_dev_address(options)?;
        let (status, reply) = crate::verbs::run_forward::exchange_dev(address, "/seed", instance.as_deref(), &body, Duration::from_secs(60))
            .map_err(|error| format!("cannot reach a dev at {address}: {error}"))?;
        if let Some(denied) = reply.get("denied") { return Err(format!("{}: {}", denied["code"].as_str().unwrap_or("E_INPUT"), denied["message"].as_str().unwrap_or("seed refused"))); }
        if status != 200 { return Err(format!("dev answered HTTP {status} to POST /seed: {reply}")); }
        Ok(reply)
    })();
    match result { Ok(reply) => { println!("{reply}"); 0 }, Err(error) => { eprintln!("{error}"); 1 } }
}

fn run(options: &crate::verbs::Options) -> Result<Json, String> {
    if options.flags.contains_key("release") { return Err("E_AUTH: reset empties a running dev's store; snapback4 serve never resets a release\n  guide: testing".into()); }
    if !options.positionals.is_empty() { return Err(usage()); }
    let (address, instance) = crate::verbs::dev_address(options)?;
    let mut body = serde_json::Map::new();
    if let Some(seed) = options.flags.get("seed") { body.insert("seed".into(), Json::Array(rows(Path::new(seed))?)); }
    match options.flags.get("clock").map(String::as_str) {
        None => {}
        Some("reset") => { body.insert("clock".into(), json!("reset")); }
        Some(other) => return Err(format!("E_CLOCK_INPUT: --clock takes `reset` (back to wall time), not {other:?}\n{}", usage())),
    }
    let (status, reply) = crate::verbs::run_forward::exchange_dev(address, "/dev/reset", instance.as_deref(), &Json::Object(body).to_string(), Duration::from_secs(60))
        .map_err(|error| format!("cannot reach a dev at {address}: {error}; start `snapback4 dev` (or pass the --port it printed)"))?;
    if let Some(denied) = reply.get("denied") {
        return Err(format!("{}: {}\n  guide: {}", denied["code"].as_str().unwrap_or("E_INPUT"), denied["message"].as_str().unwrap_or("reset refused; nothing was reset"),
            denied["guide"].as_str().unwrap_or("testing")));
    }
    if status != 200 { return Err(format!("dev answered HTTP {status} to POST /dev/reset: {reply}")); }
    Ok(reply)
}

/// `/seed` rows from a fixture directory (`<table>.jsonl`, one row per line,
/// as `test --fixture` reads it) or a JSON file holding a `/seed` array.
fn rows(path: &Path) -> Result<Vec<Json>, String> {
    if !path.is_dir() {
        let text = std::fs::read_to_string(path).map_err(|error| format!("E_INPUT: cannot read {}: {error}", path.display()))?;
        return match serde_json::from_str(&text) {
            Ok(Json::Array(rows)) => Ok(rows),
            _ => Err(format!("E_INPUT: {} is not a /seed array [{{\"table\": \"t\", \"row\": {{\"id\": ..}}}}, ..]; pass a fixture directory of <table>.jsonl files or a JSON file holding that array", path.display())),
        };
    }
    let mut names: Vec<String> = std::fs::read_dir(path).map_err(|error| format!("E_INPUT: cannot read {}: {error}", path.display()))?
        .filter_map(|entry| entry.ok().map(|entry| entry.file_name().to_string_lossy().into_owned())).collect();
    names.sort();
    let mut rows = Vec::new();
    for name in names {
        let Some(table) = name.strip_suffix(".jsonl") else { continue };
        if name == crate::verbs::CASES_FILE { continue; }
        let file = path.join(&name);
        let text = std::fs::read_to_string(&file).map_err(|error| format!("E_INPUT: cannot read {}: {error}", file.display()))?;
        for (number, line) in text.lines().enumerate().filter(|(_, line)| !line.trim().is_empty()) {
            let row: Json = serde_json::from_str(line).map_err(|error| format!("E_INPUT: {name}:{}: {error}", number + 1))?;
            rows.push(json!({ "table": table, "row": row }));
        }
    }
    Ok(rows)
}

//! Locate the store's successor refusals in the author's current sources.
use crate::project::Project;
use snapback4_core::{Schema, Store};
use snapback4_core::successor::Change;
use snapback4_lang::{Diagnostic, Site};
use std::collections::BTreeMap;
use std::path::PathBuf;

pub(super) fn diagnostics(project: &Project, schema: &Schema, explicit: Option<&String>, detailed:bool) -> Result<(Vec<Diagnostic>, Vec<Diagnostic>), String> {
    let path = explicit.map(PathBuf::from).unwrap_or_else(|| project.store_path());
    if explicit.is_none() && !path.exists() { return Ok((Vec::new(), Vec::new())); }
    let mut store = Store::open_read_only(&path).map_err(|e| e.to_string())?;
    let (estimates, timeline_error) = match store.check_timeline_activation(schema) {
        Ok(estimates) => (estimates, None),
        Err(error @ snapback4_core::RuntimeError::Structural { .. }) => (Vec::new(), Some(error)),
        Err(error) => return Err(error.to_string()),
    };
    let steps = store.successor(schema).map_err(|e| e.to_string())?;
    let pause = store.adoption_estimate(schema).map_err(|e| e.to_string())?;
    let windows = super::sql::window_populations(&path, schema).map_err(|(_, message)| message)?;
    if windows.is_empty() && pause.is_none() && estimates.iter().all(|e| e.sources == 0) && timeline_error.is_none() && !steps.iter().any(|s| s.change == Change::Narrowing) { return Ok((Vec::new(), Vec::new())); }
    let mut sites = BTreeMap::new();
    for (file, source) in project.sources().map_err(|e| e.to_string())? {
        let (document, _) = snapback4_lang::parse_document(&file, &source);
        for view in document.views { sites.insert(format!("timelines.{}", view.name), view.site); }
        for view in document.feed_views { sites.insert(format!("timelines.{}", view.name), view.site); }
        for timeline in document.timelines { sites.insert(format!("timelines.{}", timeline.name), timeline.site); }
        for maintain in document.maintains {
            sites.insert(format!("timelines.{}", maintain.name), maintain.site.clone());
            if let Some(position) = schema.maintains.iter().position(|m| m.target == maintain.name) {
                sites.insert(format!("maintains[{position}]"), maintain.site);
            }
        }
        for table in document.tables {
            let path = format!("tables.{}", table.name);
            sites.insert(path.clone(), table.site);
            for column in table.columns { sites.insert(format!("{path}.columns.{}", column.name), column.site); }
            for index in table.indexes { sites.insert(format!("{path}.indexes.{}", index.name), index.site); }
        }
    }

/// Only literal loopback addresses and localhost: no DNS, redirects, credentials,
/// paths or alternate IP encodings can turn this local inspector into a remote request.
fn loopback_url(url: &str) -> Result<std::net::SocketAddr, String> {
    use std::net::{IpAddr, Ipv4Addr, SocketAddr};
    let bad = || "--url requires a loopback HTTP origin, for example http://127.0.0.1:4400 or http://[::1]:4400; use the address dev printed at startup".to_owned();
    let authority = url.strip_prefix("http://").ok_or_else(bad)?;
    let authority = authority.strip_suffix('/').unwrap_or(authority);
    let address = if authority == "localhost" {
        SocketAddr::from((Ipv4Addr::LOCALHOST, 80))
    } else if let Some(port) = authority.strip_prefix("localhost:") {
        SocketAddr::from((Ipv4Addr::LOCALHOST, port.parse::<u16>().map_err(|_| bad())?))
    } else if let Ok(address) = authority.parse::<SocketAddr>() { address }
    else {
        let ip = authority.strip_prefix('[').and_then(|s| s.strip_suffix(']')).unwrap_or(authority);
        SocketAddr::new(ip.parse::<IpAddr>().map_err(|_| bad())?, 80)
    };
    if !address.ip().is_loopback() || address.port() == 0 { return Err(bad()); }
    Ok(address)
}

pub(super) fn request(address: std::net::SocketAddr, method: &str, path: &str) -> Option<serde_json::Value> {
    request_instance(address,method,path,None)
}
pub(super) fn request_instance(address:std::net::SocketAddr,method:&str,path:&str,instance:Option<&str>)->Option<serde_json::Value> {
    use std::{io::{Read, Write}, net::TcpStream, time::Duration};
    let instance=instance.map(|i|format!("x-snapback-dev-instance: {i}\r\n")).unwrap_or_default();
    let mut stream = TcpStream::connect_timeout(&address, Duration::from_millis(200)).ok()?;
    stream.set_read_timeout(Some(Duration::from_secs(1))).ok()?;
    stream.set_write_timeout(Some(Duration::from_secs(1))).ok()?;
    // The complete latency registry can exceed tiny_http's chunk threshold.
    // HTTP/1.0 requests a length-delimited reply for this close-only reader.
    write!(stream, "{method} {path} HTTP/1.0\r\nHost: {address}\r\n{instance}Connection: close\r\nContent-Length: 0\r\n\r\n").ok()?;
    let mut reply = String::new(); stream.take((16 << 20) + 4096).read_to_string(&mut reply).ok()?;
    let (head, body) = reply.split_once("\r\n\r\n")?;
    if !head.starts_with("HTTP/1.0 200 ") && !head.starts_with("HTTP/1.1 200 ") { return None; }
    serde_json::from_str(body).ok()
}
