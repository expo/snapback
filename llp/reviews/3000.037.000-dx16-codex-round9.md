//! fit the schema, are leftovers: `check` names them and `init --adapt`
//! removes or rewrites them. A file the project changed is never touched.
// @ref LLP 3000.011.003.000#7.2 — leftovers are judged by the starter's original bytes, never by name alone.
use super::starter::{self, Body, Platform};
use super::{OwnSchema, own_schema};
use std::path::Path;

/// One unmodified starter file that no longer fits the project's schema.
pub(crate) struct Leftover {
    pub path: &'static str,
    pub why: String,
    /// The text `init --adapt` writes; `None` removes the file.
    pub rewrite: Option<String>,
}

/// Starter operations its fixtures, cases and `Todos` screen call.
const STARTER_OPERATIONS: [&str; 4] = ["list", "add", "toggle", "remove"];

/// Whether a schema still has the starter's `todos` table and the operations
/// its fixtures, cases and `Todos` screen call.
pub(super) fn fits_todos(own: &OwnSchema) -> bool {
    own.tables.iter().any(|table| table == "todos")
        && STARTER_OPERATIONS.iter().all(|op| own.operations.iter().any(|name| name == op))
}

/// What the schema in `dir` leaves of the starter, and the advice that goes
/// with it (a sign-in the project must write by hand). No schema: nothing.
pub(crate) fn leftovers(dir: &Path) -> (Vec<Leftover>, Vec<String>) {
    let (found, advice, _) = plan(dir);
    (found, advice)
}

fn plan(dir: &Path) -> (Vec<Leftover>, Vec<String>, Vec<String>) {
    let Ok(Some(own)) = own_schema(dir) else { return (Vec::new(), Vec::new(), Vec::new()) };
    let mut found = Vec::new();
    let mut advice = Vec::new();
    let mut modified = Vec::new();
    let todos = fits_todos(&own);
    for (path, text) in starter::FIXTURES {
        let current = read(dir, path);
        if current.as_ref().is_some_and(|current| *current != super::without_design_references(text)) { modified.push(path.to_owned()); }
        if !todos && current.is_some_and(|current| current == super::without_design_references(text)) {
            let what = if path.ends_with("todos.jsonl") { "rows for the todo starter's `todos` table" } else { "cases for the todo starter's list/add/toggle/remove" };
            found.push(Leftover { path, why: format!("the unmodified {what}; your snapback/*.q has no todos table with those operations, so `snapback4 test` would refuse it"), rewrite: None });
        }
    }
    for platform in [Platform::Web, Platform::Expo] {
        if !screens(dir, platform, &own, todos, &mut found, &mut advice) {
            for path in [platform.paths().0, platform.paths().1] { if read(dir, path).is_some() { modified.push(path.to_owned()); } }
        }
    }
    if let Some(text) = read(dir, "src/native-smoke.ts") {
        let wanted = starter::native_smoke(own.identity.is_some_and(|(password, guests)| password && !guests));
        let recognized = if std::fs::symlink_metadata(dir.join("src/native-smoke.ts")).is_ok_and(|m| m.file_type().is_symlink()) { None } else { super::proof_advice::native_template(&text) };
        if recognized.is_none() { modified.push("src/native-smoke.ts".to_owned()); }
        if recognized.is_some() && !smoke_referenced(dir, &found) {
            found.push(Leftover { path: "src/native-smoke.ts", why: "the unmodified helper is now supplied by snapback4-expo/native-smoke; the planned screens no longer import this copy".into(), rewrite: None });
        } else if text != wanted && let Some((version, password)) = recognized && smoke_referenced(dir, &found) {
            let old = text != starter::native_smoke(password);
            let why = if old {
                let mode = if password != own.identity.is_some_and(|(password, guests)| password && !guests) { "; refresh also updates its sign-in mode" } else { "" };
                format!("an older starter version ({version}; adapt can refresh it){mode}")
            } else {
                "the unmodified native smoke helper uses a different starter sign-in mode (adapt can refresh it)".into()
            };
            found.push(Leftover { path: "src/native-smoke.ts", why, rewrite: Some(wanted) });
        }
    }
    if read(dir, "src/native-smoke.ts").is_some() && !smoke_referenced(dir, &found) && !found.iter().any(|f| f.path == "src/native-smoke.ts" && f.rewrite.is_none()) {
        advice.push("info: src/native-smoke.ts has no source reference; review it before a native smoke proof".into());
    }
    if !todos && let Some(web) = [false, true].into_iter().find(|web| super::proof_advice::frontend_template(dir, *web).is_some_and(|(_, template)| template.is_none())) {
        found.push(Leftover { path: "scripts/frontend-proof.mjs", why: "the unmodified todo proof needs your screen's write assertions; keep a runnable frontend scaffold that explicitly reports the missing scenario".into(), rewrite: Some(super::proof::frontend_scaffold(web)) });
    }
    if found.iter().any(|change| change.rewrite.as_ref().is_some_and(|text| text.contains("/shared.ts\""))) {
        for (path, source) in [("src/shared.ts", include_str!("starter/shared.ts")), ("src/package.json", "{\"type\":\"module\"}\n")] {
            if !dir.join(path).exists() { found.push(Leftover { path, why: "shared TypeScript imported by the starter and Node proof".into(), rewrite: Some(source.into()) }); }
        }
    }
    if let Some(text) = read(dir, ".gitignore") {
        let revised = super::starter_gitignore(&text);
        if revised != text { found.push(Leftover { path: ".gitignore", why: "the starter's generated runtime imports must survive a git copy".into(), rewrite: Some(revised) }); }
    }
    readme(dir, &own, todos, &mut found, &mut modified);
    app_json(dir, &mut found);
    expo_favicon(dir, &mut found);
    (found, advice, modified)
}

// Default proof upgrades share the screen's custody decision. Read planned
// helper bytes so a proof cannot acquire imports or controls its kept app lacks.
fn default_frontend_ready(dir: &Path, changes: &[Leftover], modified: &[String]) -> bool {
    let app_path = if dir.join("src/App.tsx").is_file() { "src/App.tsx" } else { "App.tsx" };
    if read(dir, app_path).is_none() || read(dir, "src/client.ts").is_none()
        || modified.iter().any(|path| path == app_path || path == "src/client.ts") { return false; }
    [("src/shared.ts", include_str!("starter/shared.ts")), ("src/package.json", "{\"type\":\"module\"}\n")].into_iter().all(|(path, expected)| {
        if let Some(change) = changes.iter().find(|change| change.path == path) {
            change.rewrite.as_deref() == Some(expected)
        } else { read(dir, path).as_deref() == Some(expected) }
    })
}

// A custom entry/component/proof can reference the helper too. Generated trees
// and dependency directories are not authored source; do not follow symlinks.
fn smoke_referenced(root: &Path, changes: &[Leftover]) -> bool {
    fn walk(root: &Path, dir: &Path, helper: &Path, changes: &[Leftover]) -> bool {
        let Ok(entries)=std::fs::read_dir(dir) else { return true; };
        for entry in entries {
            let Ok(entry)=entry else {return true;};
            let Ok(kind)=entry.file_type() else {return true;};
            let path=entry.path();
            if kind.is_symlink() {
                if matches!(entry.file_name().to_str(),Some("node_modules"|".git"|".snapback4"|".expo"|"target"|"dist"|"build"|"generated")) {continue;}
                return true; // Unknown source may refer back to the local helper.
            }
            if kind.is_dir() {
                if !matches!(entry.file_name().to_str(),Some("node_modules"|".git"|".snapback4"|".expo"|"target"|"dist"|"build"|"generated")) && walk(root,&path,helper,changes) {return true;}
            } else if path!=helper && !path.to_string_lossy().ends_with(".d.ts")
                && path.extension().and_then(|e|e.to_str()).is_some_and(|e|matches!(e,"ts"|"tsx"|"js"|"jsx"|"mjs"|"cjs")) {
                let relative = path.strip_prefix(root).unwrap().to_string_lossy();
                let text = if let Some(change) = changes.iter().find(|f| f.path == relative) {
                    let Some(text) = &change.rewrite else { continue; }; Some(text.clone())
                } else { std::fs::read_to_string(path).ok() };
                if text.is_none_or(|text| {
                    let text = text.replace("snapback4-expo/native-smoke", "");
                    if text.contains("/native-smoke") { return true; }
                    let compact: String = text.chars().filter(|c| !c.is_whitespace()).collect();
                    if ["from\"native-smoke\"", "from'native-smoke'", "import\"native-smoke\"", "import'native-smoke'"].iter().any(|s| compact.contains(*s)) { return true; }
                    // A CLI mode named native-smoke is not a module reference.
                    // Unknown dynamic imports/requires keep the local helper.
                    ["import(", "require("].iter().any(|call| compact.match_indices(*call).any(|(at, _)| {
                        let argument = &compact[at + call.len()..];
                        !argument.starts_with(['\'', '"']) || argument.starts_with("'native-smoke'") || argument.starts_with("\"native-smoke\"")
                    }))
                }) {return true;}
            }
        }
        false
    }
    walk(root,root,&root.join("src/native-smoke.ts"),changes)
}

/// The starter README, while it is still one init wrote (for either screen,
/// at the ports it names), when it describes another screen than the schema
/// now gets: the Todos README beside a project's schema, or the reverse.
// @ref LLP 3000.011.003.000#7.4 — wave 6 (042): adapt rewrote the screen but left a README about todos and guest sign-in.
fn readme(dir: &Path, own: &OwnSchema, todos: bool, found: &mut Vec<Leftover>, modified: &mut Vec<String>) {
    let platform = if dir.join("src/App.tsx").is_file() { Platform::Web } else if dir.join("App.tsx").is_file() { Platform::Expo } else { return };
    let title = super::title(dir);
    let personas = own.identity.is_none() && !own.personas.is_empty();
    let wanted_body = if todos { Body::Todos } else { Body::Home(None) };
    for path in ["README.md", "README.snapback.md"] {
        let Some(text) = read(dir, path) else { continue };
        let port = number_after(&text, "127.0.0.1:");
        let web_port = number_after(&text, "localhost:");
        let ports = [(port, web_port), (None, web_port), (port, None), (None, None)];
        // The offline paragraph names the schema's synced tables when init
        // wrote it; any of them (or 0.4.22's unconditional claim) still matches.
        let masked = starter::without_offline(&text);
        let explicit_online = platform == Platform::Web && read(dir, "src/client.ts").is_some_and(|client| client.contains("writes: \"online\""));
        let matched = ports.into_iter().find(|&(port, web_port)| {
            [(Body::Todos, false), (Body::Home(None), false), (Body::Home(None), true)].iter()
                .any(|(body, personas)| [false, true].iter().any(|password| [false, true].iter().any(|online| {
                    let readme = starter::identity_readme(platform, &title, body, *personas, &starter::Offline::Unknown, *password);
                    let readme = if *online { starter::online_readme(&readme) } else { readme };
                    masked == starter::without_offline(&super::localize(path, &super::without_design_references(&readme), port, web_port))
                })))
        });
        let Some((port, web_port)) = matched else { modified.push(path.to_owned()); continue };
        let wanted = super::localize(path, &super::without_design_references(&starter::identity_readme(platform, &title, &wanted_body, personas, &own.offline, own.identity.is_some_and(|(password, guests)| password && !guests))), port, web_port);
        let wanted = if explicit_online { starter::online_readme(&wanted) } else { wanted };
        let wanted = if todos { super::proof::readme(wanted) } else { wanted };
        let wanted = if dir.join("scripts/install.mjs").is_file() {wanted.replace("npm install", "node scripts/install.mjs")} else {wanted};
        if wanted == text { continue; }
        let about = if todos { "the todo starter's operations, which your snapback/*.q declares again" } else { "your snapback/*.q and the Home screen, named after this directory" };
        let only_proof = super::proof::without_readme(&wanted) == text;
        let only_offline = !text.starts_with("# Todos") && starter::without_offline(&wanted) == masked;
        // A schema that does not compile right now says nothing about what it syncs.
        if only_offline && !only_proof && own.offline == starter::Offline::Unknown { continue; }
        let why = if only_proof {
            "the unmodified starter README lacks frontend/day-two proof instructions; adapt adds them".to_owned()
        } else if only_offline {
            "the unmodified starter README says what works offline for another schema; adapt rewrites that paragraph from the tables your snapback/*.q syncs".to_owned()
        } else {
            format!("the unmodified starter README describes {}; adapt rewrites it to describe {about}", if text.starts_with("# Todos") { "the todo starter (its todos table and guest sign-in)" } else { "the Home screen for another schema" })
        };
        found.push(Leftover { path, why, rewrite: Some(wanted) });
    }
}

/// An Expo `app.json` that still carries the todo starter's slug, scheme and
/// bundle ids (written by 0.4.21 and earlier): adapt names them after the
/// directory and changes nothing else in the file.
// @ref LLP 3000.011.003.000#7.4 — wave 6 (067): adapted Expo apps kept dev.snapback4.todos.
fn app_json(dir: &Path, found: &mut Vec<Leftover>) {
    let Some(text) = read(dir, "app.json") else { return };
    let Ok(config) = serde_json::from_str::<serde_json::Value>(&text) else { return };
    let expo = &config["expo"];
    let [(_, slug), (_, scheme), (_, bundle)] = starter::TODO_IDS;
    if expo["slug"] != slug || expo["scheme"] != scheme || expo["ios"]["bundleIdentifier"] != bundle || expo["android"]["package"] != bundle { return; }
    let (new_slug, new_scheme, new_bundle) = starter::app_ids(&super::title(dir));
    let quoted = |value: &str| format!("\"{value}\"");
    let rewrite = text.replace(&quoted(bundle), &quoted(&new_bundle)).replace(&quoted(slug), &quoted(&new_slug)).replace(&quoted(scheme), &quoted(&new_scheme));
    found.push(Leftover { path: "app.json", why: format!("the todo starter's ids (slug {slug}, scheme {scheme}, bundle id {bundle}); adapt names them after this directory ({new_slug}, {new_scheme}, {new_bundle})"), rewrite: Some(rewrite) });
}

// Repair only the exact SVG asset and setting emitted by 0.4.26. Custom
// images/settings remain owned by the app, including other SVG paths.
fn expo_favicon(dir: &Path, found: &mut Vec<Leftover>) {
    if read(dir, "assets/favicon.svg").as_deref() != Some(include_str!("starter/favicon.svg")) { return; }
    let Some(text) = read(dir, "app.json") else { return };
    let Ok(config) = serde_json::from_str::<serde_json::Value>(&text) else { return };
    if config.pointer("/expo/web/favicon").and_then(|value| value.as_str()) != Some("./assets/favicon.svg") { return; }
    let why = "the unmodified Expo favicon is SVG, which Expo's web export requires to be raster".to_owned();
    if let Some(existing) = found.iter_mut().find(|change| change.path == "app.json") {
        let mut config: serde_json::Value = serde_json::from_str(existing.rewrite.as_ref().unwrap()).unwrap();
        config["expo"]["web"].as_object_mut().unwrap().remove("favicon");
        existing.rewrite = Some(super::pretty(&config));
        existing.why.push_str("; Expo's web export also requires a raster favicon");
    } else {
        let mut config = config;
        config["expo"]["web"].as_object_mut().unwrap().remove("favicon");
        found.push(Leftover { path: "app.json", why: why.clone(), rewrite: Some(super::pretty(&config)) });
    }
    found.push(Leftover { path: "assets/favicon.svg", why, rewrite: None });
}

fn number_after(text: &str, prefix: &str) -> Option<u16> {
    let at = text.find(prefix)? + prefix.len();
    let digits: String = text[at..].chars().take_while(char::is_ascii_digit).collect();
    digits.parse().ok()
}

/// The sign-in a schema admits, as the starter screen it needs: the guest
/// screen (`None`), the persona picker (`Some(personas)`), or no starter
/// screen fits (`Err(advice)`).
#[derive(Clone, PartialEq, Eq)]
enum SignIn { Guest, Password, Personas(Vec<String>) }
impl SignIn {
    fn label(&self) -> String {
        match self { Self::Guest => "the guest sign-in".into(), Self::Password => "the password sign-in".into(), Self::Personas(names) => format!("a persona picker ({})", names.join(", ")) }
    }
    fn personas(&self) -> Option<&[String]> { match self { Self::Personas(names) => Some(names), _ => None } }
}
fn wanted(own: &OwnSchema) -> Result<SignIn, &'static str> {
    match own.identity {
        Some((_, true)) => Ok(SignIn::Guest),
        Some((true, false)) => Ok(SignIn::Password),
        None if !own.personas.is_empty() => Ok(SignIn::Personas(own.personas.clone())),
        _ => Err("declares neither `use identity` nor `use personas`, so nobody can act: add `use identity` for the guest sign-in or `use personas alice, bob` for development, then rerun `snapback4 init --adapt`"),
    }
}

fn screens(dir: &Path, platform: Platform, own: &OwnSchema, todos: bool, found: &mut Vec<Leftover>, advice: &mut Vec<String>) -> bool {
    let (app_path, client_path) = platform.paths();
    let (Some(app), Some(client)) = (read(dir, app_path), read(dir, client_path)) else { return true };
    // Which starter screen is on disk, if both files are still as init wrote
    // them: its sign-in (guest, or a persona picker) and its body.
    let journal_shape = starter::Offline::Tables { synced: vec![], total: 0 };
    let explicit_online = platform == Platform::Web && client.contains("writes: \"online\"");
    let client_shape = normalized(&client);
    let matches_client = |reference: &str, personas: bool| [reference, starter::previous_client(platform, personas)].iter().any(|reference|
        client_shape == normalized(reference) || client_shape == normalized(&starter::web_client((*reference).to_owned(), &journal_shape)) || client_shape == normalized(&starter::online_web_client((*reference).to_owned())) || client_shape == normalized(&starter::expo_client((*reference).to_owned(), &journal_shape)));
    let matches_app = |reference: &str| app == reference || app == reference.replace("export default function App", "export function App");
    let current = [Body::Todos, Body::Home(first_read(&app))].into_iter().find_map(|body| [0, 1, 2, 3, 4, 5].into_iter().find_map(|version| {
        let screens = |personas: Option<&[String]>, password| match version { 5 => starter::dx13_screens(platform, personas, &body, password), 1 => starter::previous_screens(platform, personas, &body, password), 2 => starter::wave14_screens(platform, personas, &body, password), 3 => starter::wave20_screens(platform, personas, &body, password), 4 => starter::wave22_screens(platform, personas, &body, password), _ => starter::identity_screens(platform, personas, &body, password) };
        let (guest_app, guest_client) = screens(None, false);
        let (persona_app, persona_client) = screens(Some(&[]), false);
        let (password_app, password_client) = screens(None, true);
        if matches_app(&guest_app) && matches_client(&guest_client, false) { Some((SignIn::Guest, body.clone())) }
        else if matches_app(&password_app) && (matches_client(&password_client, false) || matches_client(starter::pre_dx13_password_client(platform), false)) { Some((SignIn::Password, body.clone())) }
        else if matches_app(&persona_app) && matches_client(&persona_client, true) { Some((SignIn::Personas(personas_of(&client)), body.clone())) }
        else { None }
    }));
    let Some((sign_in, body)) = current else { return false };
    // A Home keeps its read while the schema still declares that query
    // argument-free; a Home without one stays as written (it fits any schema).
    let wanted_body = match &body {
        _ if todos => Body::Todos,
        Body::Home(None) => Body::Home(None),
        Body::Home(Some(op)) if own.reads.contains(op) => body.clone(),
        _ => Body::Home(own.reads.first().cloned()),
    };
    // A sign-in no starter screen fits keeps the one on disk, with advice.
    let wanted_sign_in = match wanted(own) {
        Ok(wanted) => wanted,
        Err(why) => {
            advice.push(format!("{app_path} is the starter's {}, but your snapback/*.q {why}", sign_in.label()));
            sign_in.clone()
        }
    };
    let (app_text, mut client_text) = starter::identity_screens(platform, wanted_sign_in.personas(), &wanted_body, wanted_sign_in == SignIn::Password);
    if app_text.contains("/shared.ts\"") && dir.join("src/shared.ts").symlink_metadata().is_ok()
        && read(dir, "src/shared.ts").as_deref() != Some(include_str!("starter/shared.ts")) {
        advice.push(format!("{app_path}: keep your screen while src/shared.ts is application-owned; the refreshed starter needs its prepareTitle/matchesFilter exports"));
        return false;
    }
    client_text = if explicit_online { starter::online_web_client(client_text) } else if platform == Platform::Expo { starter::expo_client(client_text, &own.offline) } else { starter::web_client(client_text, &own.offline) };
    let client_changed = client_shape != normalized(&client_text);
    if platform == Platform::Expo && app_text.contains("export default function App") {
        let entry = include_str!("../../../../../apps/chat-expo/index.js");
        if read(dir,"index.js").is_some_and(|text| text == entry.replace("import App from", "import { App } from")) {
            found.push(Leftover { path: "index.js", why: "the owned Expo entry imports the old named App; the screen now exports App by default".into(), rewrite: Some(entry.into()) });
        }
    }
    if wanted_sign_in == sign_in && app == app_text && wanted_body == body && !client_changed { return true; }
    let mut reasons = Vec::new();
    let mut client_leftover = None;
    if wanted_sign_in != sign_in {
        let (from, to) = (sign_in.label(), wanted_sign_in.label());
        let because = match &wanted_sign_in { SignIn::Personas(_) => "your snapback/*.q declares `use personas` and no `use identity`", SignIn::Password => "your snapback/*.q declares `use identity password`", SignIn::Guest => "your snapback/*.q admits guests" };
        reasons.push(format!("the unmodified starter screen shows {from}, but {because}; adapt writes {to}"));
        // Keep the backend port init wrote.
        if let Some(port) = port_of(&client) { client_text = client_text.replace("127.0.0.1:4400", &format!("127.0.0.1:{port}")); }
        client_leftover = Some(Leftover { path: client_path, why: format!("the unmodified starter client for {from}; adapt writes {to}"), rewrite: Some(client_text.clone()) });
    }
    if wanted_sign_in == sign_in && client_changed {
        if let Some(port) = port_of(&client) { client_text = client_text.replace("127.0.0.1:4400", &format!("127.0.0.1:{port}")); }
        let why = "the unmodified web client needs the device or principal journal selected by your schema's synced tables".to_owned();
        reasons.push(why.clone());
        client_leftover = Some(Leftover { path: client_path, why, rewrite: Some(client_text.clone()) });
    }
    if wanted_body != body {
        reasons.push(match (&wanted_body, &body) {
            (Body::Home(_), Body::Home(Some(op))) => format!("its Home screen reads api.{op}, which your snapback/*.q no longer declares as a query without required arguments; adapt writes the Home screen again{}", match &wanted_body { Body::Home(Some(next)) => format!(", reading api.{next}"), _ => String::new() }),
            (Body::Home(_), _) => "its Todos screen calls the starter's list/add/toggle/remove, which your snapback/*.q does not declare, so it would not typecheck; adapt writes a Home screen that compiles against any schema (the signed-in viewer, your operations from snapback/generated/api.ts, and your first query without required arguments rendered with matchRead) for you to replace".to_owned(),
            (Body::Todos, _) => "your snapback/*.q declares the todo starter's operations again; adapt writes the Todos screen back".to_owned(),
        });
    }
    let why = reasons.join("; ");
    let why = if why.starts_with("the unmodified") { why } else { format!("the unmodified starter screen: {why}") };
    found.push(Leftover { path: app_path, why, rewrite: Some(app_text) });
    found.extend(client_leftover);
    true
}

/// The `api` path a starter Home's first read calls, if it has one.
fn first_read(app: &str) -> Option<String> {
    let after = app.split_once("function FirstRead()")?.1;
    let op = after.split_once("useQuery(api.")?.1.split_once(',')?.0;
    Some(op.to_owned())
}

fn read(dir: &Path, path: &str) -> Option<String> {
    let path = dir.join(path);
    path.symlink_metadata().ok().filter(|metadata| metadata.is_file())?;
    std::fs::read_to_string(path).ok()
}

/// A starter client without what init or the schema varies: the backend port
/// and the persona list.
fn normalized(client: &str) -> String {
    let lines = client.split_inclusive('\n').map(|line| {
        if line.starts_with(PERSONAS) { format!("{PERSONAS}[];\n") } else { line.to_owned() }
    }).collect::<String>();
    match port_of(&lines) { Some(port) => lines.replace(&format!("127.0.0.1:{port}"), "127.0.0.1:4400"), None => lines }
}

const PERSONAS: &str = "export const personas: readonly string[] = ";

fn personas_of(client: &str) -> Vec<String> {
    client.lines().find_map(|line| line.strip_prefix(PERSONAS))
        .and_then(|list| serde_json::from_str(list.trim_end().trim_end_matches(';')).ok()).unwrap_or_default()
}

fn port_of(client: &str) -> Option<&str> {
    let at = client.find("127.0.0.1:")? + "127.0.0.1:".len();
    let digits = client[at..].find(|c: char| !c.is_ascii_digit()).unwrap_or(client.len() - at);
    (digits > 0).then(|| &client[at..at + digits])
}

/// One `check` line, or none: the leftovers and the command that fixes them.
pub(crate) fn report(dir: &Path) -> Option<String> {
    let (found, mut advice) = leftovers(dir);
    advice.extend(super::proof_advice::report(dir));
    if found.is_empty() && advice.is_empty() { return None; }
    if found.is_empty() { return Some(advice.join("\n")); }
    let mut lines = Vec::new();
    for leftover in &found { lines.push(format!("  {}: {}", leftover.path, leftover.why)); }
    for line in &advice { lines.push(format!("  {line}")); }
    if !found.is_empty() {
        lines.push("  fix: snapback4 init --adapt (removes or rewrites only these files, which still match the starter; your own files are never touched)".into());
    }
    Some(format!("info: starter: {} from `snapback4 init` to review\n{}",
        if found.is_empty() { "a screen".to_owned() } else { format!("{} file{}", found.len(), if found.len() == 1 { "" } else { "s" }) },
        lines.join("\n")))
}

/// The same, for `check --json`: `null` when nothing is left over.

pub(super) fn adapt(dir: &Path, dry_run: bool, apply: bool, day2: bool) -> i32 {

        Err(error) => { eprintln!("init: {error}"); return 2; }

        match sources.and_then(|sources|super::frozen::files(dir,&sources)) {
            Ok(files) => proof.extend(files.into_iter().filter(|(name,_)| !proof.iter().any(|(existing,_)|existing==name)).collect::<Vec<_>>()),
            Err(error) => { eprintln!("init --adapt frozen client: {error}"); return 2; }
        }
    }

    }
    for line in super::proof_advice::legacy_differences(dir, &proof) { println!("init --adapt: {line}"); }
    let frozen_count=proof.iter().filter(|(path,_)|path.starts_with("v1/")).count();

            let new: serde_json::Value = serde_json::from_str(body).unwrap();

                for key in keys { if old[group][key] != new[group][key] { println!("    {group}.{key}: {} → {}", old[group][key], new[group][key]); } }
            }
            if old.pointer("/snapback4/starter/day2") != new.pointer("/snapback4/starter/day2") {
                println!("    snapback4.starter.day2: {} → {}", old["snapback4"]["starter"]["day2"], new["snapback4"]["starter"]["day2"]);
            }
            println!("    serialize package.json (including script key ordering)");
        }
    }
    if !modified.is_empty() { println!("init --adapt: application-owned or modified files stay unchanged: {}", modified.join(", ")); }

        println!("init --adapt: preview only; no files changed. Apply with:");
        println!("snapback4 init{} --adapt{} --apply", if dir == Path::new(".") { String::new() } else { format!(" '{}'", dir.to_string_lossy().replace('\'', "'\"'\"'")) }, if day2 { " --day2" } else { "" });
        return 0;

    let mut rewritten = Vec::new();

        let result = match &leftover.rewrite {
            Some(text) => super::replace_file(&path, text).map(|()| rewritten.push(leftover.path)),
            None => std::fs::remove_file(&path).map(|()| removed.push(leftover.path)),
        };
        if let Err(error) = result { eprintln!("init --adapt: {}: {error}", path.display()); return 2; }
    }
    // @ref LLP 3000.011.003.000#7.4 — adapt removes the files it names, never a directory: the next step writes fixtures/<table>.jsonl.
    if found.is_empty() && proof.is_empty() { println!("init --adapt: nothing to adapt — no unmodified starter file conflicts with your snapback/*.q"); }

pub(super) fn files(files: &mut Vec<(&'static str, String)>, options: &Options, backend: u16, frontend: u16, web: bool, shape: &super::starter::Shape, capture_client: bool) -> std::io::Result<()> {
    let has_frontend = true;
    let has_write_scenario = shape.body == super::starter::Body::Todos || options.flags.contains_key("template");
    for (_, text) in files.iter_mut().filter(|(name, _)| *name == "README.md") {
        if has_frontend { *text = readme(std::mem::take(text)); }
    }
    for (name, text) in files.iter_mut() {
        if *name == "vite.config.mts" { *text = runtime_vite(text, backend, frontend); }
    }
    let proof = if has_write_scenario { frontend_proof(include_str!("frontend-proof.mjs"), options.flags.get("template").map(String::as_str), web) } else { frontend_scaffold(web) };
    let dir = options.positionals.first().map(std::path::Path::new).unwrap_or_else(|| std::path::Path::new("."));
    let sources = if let Ok(Some(own)) = super::own_schema(dir) {
        own.files.iter().filter_map(|name| std::fs::read_to_string(dir.join("snapback").join(name)).ok().map(|text| (name.clone(), text))).collect::<Vec<_>>()
    } else { files.iter().find(|(name, _)| *name == "snapback/schema.q").map(|(_, text)| vec![("schema.q".to_owned(), text.clone())]).unwrap_or_default() };
    let day2 = options.flags.contains_key("day2") || !dir.join("package.json").is_file();
    // Starter proof sources are editable drafts. Freeze historical client code
    // only at the explicit day-two handover, after day-one customization.
    if capture_client {
        files.extend(super::frozen::files(dir,&sources)?);
    }
    for version in ["v1", "v2"].into_iter().filter(|_| day2) {
        if dir.join("versions").join(version).exists() { continue; }
        // Init is a one-shot CLI operation. Destinations live for this invocation's
        // plan, whose existing interface uses static generated names.
        for (name, text) in &sources {
            let path: &'static str = Box::leak(format!("versions/{version}/{name}").into_boxed_str());
            files.push((path, text.clone()));
        }
        let marker: &'static str = Box::leak(format!("versions/{version}/fixtures/.gitkeep").into_boxed_str());
        files.push((marker, String::new()));
        let mut fixtures = files.iter().filter_map(|(name, text)| name.strip_prefix("fixtures/").map(|name| (name.to_owned(), text.clone()))).collect::<Vec<_>>();
        if fixtures.is_empty() {
            if let Ok(entries) = std::fs::read_dir(dir.join("fixtures")) {
                for entry in entries.flatten().filter(|entry| entry.path().extension().is_some_and(|ext| ext == "jsonl") || entry.file_name() == "test.sweep.json") {
                    if let Ok(text) = std::fs::read_to_string(entry.path()) { fixtures.push((entry.file_name().to_string_lossy().into_owned(), text)); }
                }
            }
        }
        for (name, text) in fixtures {
            let path: &'static str = Box::leak(format!("versions/{version}/fixtures/{name}").into_boxed_str());
            files.push((path, text));
        }
    }
    if has_frontend {
        if shape.password {
            let proof=if options.flags.contains_key("password") && super::own_schema(dir).ok().flatten().is_none() {
                frontend_proof(include_str!("frontend-proof.mjs"),None,web).replace("BUILD_SCRIPT",if web {"build"}else{"build:web"}).replace("Task","Note").replace("todo-results","note-results").replace("RESULTS_SCOPE",if web {"page.getByRole(\"list\")"}else{"page.getByTestId(\"todo-results\")"})
            } else {proof};
            files.extend(super::password_proof::files(&sources,proof));
        }
        else {files.push(("scripts/frontend-proof.mjs",proof));}
    }
    if day2 { files.push(("scripts/day2-proof.mjs", include_str!("day2-proof.mjs").to_owned())); }
    files.push(("scripts/launch.mjs", include_str!("launch.mjs").replace("BACKEND", &backend.to_string()).replace("FRONTEND", &frontend.to_string())));
    for (name, text) in files.iter_mut().filter(|(name, _)| *name == "package.json") {
        let mut package: Value = serde_json::from_str(text).unwrap();
        if has_frontend { package["scripts"]["frontend:proof"] = json!("node scripts/frontend-proof.mjs"); }
        if shape.password {package["scripts"]["probe:serve"]=json!("node scripts/probe-serve.mjs");package["scripts"]["probe:check"]=json!("node scripts/probe-check.mjs");}
        if day2 { package["scripts"]["day2:proof"] = json!("node scripts/day2-proof.mjs versions/v1 versions/v2"); }
        if (package["snapback4"].is_null() || package["snapback4"].is_object()) && (package["snapback4"]["starter"].is_null() || package["snapback4"]["starter"].is_object()) {
            package["snapback4"]["starter"]["day2"] = json!(day2);
        }
        if has_frontend {
            package["scripts"]["proof:web"] = json!("node scripts/frontend-proof.mjs");
            package["devDependencies"]["playwright-core"] = json!("1.55.1");
        }
        package["scripts"]["backend"] = json!("node scripts/launch.mjs backend");
        if web {
            package["scripts"]["dev"] = json!("node scripts/launch.mjs vite");
            package["scripts"]["preview"] = json!("node scripts/launch.mjs vite-preview");
        } else {
            package["scripts"]["start"] = json!("node scripts/launch.mjs expo");
            package["scripts"]["web"] = json!("node scripts/launch.mjs expo-web");
            package["scripts"]["native:smoke"] = json!("node scripts/launch.mjs native-smoke");
        }
        *text = pretty(&package);
        let _ = name;
    }
    Ok(())
}

pub(super) fn frontend_proof(source: &str, template: Option<&str>, web: bool) -> String {
    // Social keeps account controls on its Profile tab; prove the actual route.
    let mut proof = source.replace("// TEMPLATE_READY", if template == Some("social") {
        "await page.getByText(\"Profile\", { exact: true }).click();"
    } else { "" });
    proof = proof.replace("// SHARED_IMPORT", if template.is_none() { "import { matchesFilter } from \"../src/shared.ts\";\nassert.equal(matchesFilter(false, \"open\"), true);" } else { "" });
    proof = proof.replace("// LABELLED_CONTROLS", if template.is_none() && web { "const filter = page.getByRole(\"combobox\", { name: \"Task status\", exact: true });\n    await filter.selectOption(\"open\");\n    await filter.selectOption(\"all\");" } else if template.is_none() { "await page.getByLabel(\"Write status\", { exact: true }).waitFor();" } else { "" });
    if template == Some("social") {
        proof = proof.replace("getByLabel(\"Task\"", "getByLabel(\"Display name\"")
            .replace("name: \"Add\"", "name: \"Save profile\"")
            .replace("beginForm(\"add\", { title }", "beginForm(\"saveProfile\", { name: title }");
    } else if template == Some("chat") {
        proof = proof.replace("    const task =", r#"    const viewer = await page.evaluate(() => window.__snapback4Witness.viewer());
    const seeded = await fetch(`${dev.url}/seed`, { method: "POST", headers: { "content-type": "application/json" }, body: JSON.stringify([
      { table: "conversations", row: { id: "proof-room", title: "Proof room", creatorId: viewer } },
      { table: "memberships", row: { id: "proof-member", conversationId: "proof-room", principal: viewer, role: "admin", joinedAt: Date.now() } }
    ]) });
    assert.ok(seeded.ok, await seeded.text());
    await page.getByText("Proof room", { exact: true }).click();
    const task ="#)
            .replace("page.getByLabel(\"Task\", { exact: true })", "page.getByPlaceholder(\"Message\", { exact: true })")
            .replace("name: \"Add\"", "name: \"Send\"")
            .replace("beginForm(\"add\", { title }", "beginForm(\"send\", { conversationId: \"proof-room\", body: title }");
        // Chat receipts echo drafts. Restrict evidence to its message display.
        if web { proof = proof.replace("renderedText(page, new RegExp(title))", "renderedText(page, new RegExp(title), { within: page.getByRole(\"region\", { name: \"Thread\" }) })"); }
    }
    let scope = match (template, web) {
        (Some("social"), _) => "page.getByTestId(\"profile-results\")",
        (Some("chat"), true) => "page.getByRole(\"region\", { name: \"Thread\" })",
        (Some("chat"), false) => "page.getByTestId(\"thread-results\")",
        (_, true) => "page.getByRole(\"list\")",
        (_, false) => "page.getByTestId(\"todo-results\")",
    };
    proof = proof.replace("RESULTS_SCOPE", scope);
    proof = proof.replace("BUILD_SCRIPT", if web { "build" } else { "build:web" });
    proof
}

Edit scripts/frontend-proof.mjs and replace this refusal with your write assertions.');\n  }});\n}} finally {{ await cleanup(() => web?.stop(), () => dev.stop()); }}\n")

/// Adapt adds missing proof infrastructure, preserving files and script names
/// already owned by the application. It never runs a package manager.
pub(super) fn plan(dir: &std::path::Path, own: &super::OwnSchema, enable_day2: bool, refresh_default_frontend: bool) -> std::io::Result<Vec<(&'static str, String)>> {
    let web = dir.join("src/App.tsx").is_file();
    let router = dir.join("app/_layout.tsx").is_file();
    if !web && !dir.join("App.tsx").is_file() && !router { return Ok(Vec::new()); }
    let shape = super::starter::Shape { title: super::title(dir), password: own.identity.is_some_and(|(p, g)| p && !g),
        personas: own.identity.is_none().then(|| own.personas.clone()).filter(|p| !p.is_empty()),
        body: if super::adapt::fits_todos(own) { super::starter::Body::Todos } else { super::starter::Body::Home(None) }, offline: own.offline.clone() };
    let path = dir.join("package.json");
    let text = std::fs::read_to_string(&path)?;
    let mut package: Value = serde_json::from_str(&text).map_err(std::io::Error::other)?;
    let router = router && !web && package["main"] == "expo-router/entry";
    if !web && !dir.join("App.tsx").is_file() && !router { return Ok(Vec::new()); }
    let original = package.clone();
    let day2_text = std::fs::read_to_string(dir.join("scripts/day2-proof.mjs")).ok();
    let referenced_day2 = package["scripts"].as_object().is_some_and(|scripts| scripts.values().any(|v| v.as_str().is_some_and(|s| s.contains("scripts/day2-proof.mjs"))));
    let missing_managed_day2 = day2_text.is_none() && matches!(package["scripts"]["day2:proof"].as_str(),
        Some("node scripts/day2-proof.mjs" | "node scripts/day2-proof.mjs versions/v1.q versions/v2.q" | "node scripts/day2-proof.mjs versions/v1 versions/v2"));
    let managed_day2 = day2_text.as_deref().is_some_and(super::proof_advice::known_day2) || missing_managed_day2;
    let day2 = enable_day2 || referenced_day2 || package.pointer("/snapback4/starter/day2") != Some(&json!(false)) && managed_day2;
    // Deleting both the command and file records a durable opt-out on apply.
    // Preview/check never write the inferred choice. Explicit --day2 opts in again.
    if (package["snapback4"].is_null() || package["snapback4"].is_object()) && (package["snapback4"]["starter"].is_null() || package["snapback4"]["starter"].is_object()) {
        package["snapback4"]["starter"]["day2"] = json!(day2);
    }
    let backend = package["scripts"]["backend"].as_str().and_then(|s| s.strip_prefix("snapback4 dev --port ")).and_then(|s| s.parse::<u16>().ok())
.or_else(|| launcher_port(dir, "backend")).unwrap_or(4400);
    let frontend = launcher_port(dir, "frontend").or_else(|| package["scripts"][if web { "dev" } else { "start" }].as_str()
        .and_then(|s| s.split_once("--port ")).and_then(|(_, rest)| rest.split_whitespace().next()).and_then(|n| n.parse::<u16>().ok())
        .or_else(|| std::fs::read_to_string(dir.join("vite.config.mts")).ok().and_then(|s| s.split_once("server: { port: ").and_then(|(_, rest)| rest.split(|c: char| !c.is_ascii_digit()).next()).and_then(|n| n.parse::<u16>().ok())))
        ).unwrap_or(if backend != 4400 { backend.saturating_add(1) } else if web { 5173 } else { 8081 });
    let mut generated = vec![("package.json", pretty(&package))];
    let mut flags = std::collections::BTreeMap::new();
    if router { flags.insert("template".into(), "social".into()); }
    else if let Some(Some(template)) = previous_frontend(dir, web) { flags.insert("template".into(), template.into()); }
    if day2 { flags.insert("day2".into(), String::new()); }
    let default_todos = shape.body == super::starter::Body::Todos && !flags.contains_key("template");
    files(&mut generated, &Options { positionals: vec![dir.to_string_lossy().into_owned()], flags, json: false }, backend, frontend, web, &shape, false)?;
    // A saved version is immutable application evidence. Never fill its missing
    // fixture files from today's working release during an adaptation.
    generated.retain(|(name, _)| !name.starts_with("versions/") || {
        let version = name.split('/').nth(1).unwrap();
        !dir.join("versions").join(version).exists()
    });
    // New directories capture the current source and contemporary fixtures. Flat
    // legacy snapshots remain untouched evidence; they are never propagated.
    // Custom proofs own their saved source layout; do not manufacture snapshots.
    if !managed_day2 && !enable_day2 { generated.retain(|(name,_)| !name.starts_with("versions/")); }
    let wanted: Value = serde_json::from_str(&generated[0].1).unwrap();
    for group in ["scripts", "devDependencies"] {
        if !package[group].is_object() { package[group] = json!({}); }
        for (key, value) in wanted[group].as_object().unwrap() {
            let old = package[group].get(key).and_then(Value::as_str);
            let managed = group == "scripts" && old.is_some_and(|old| match key.as_str() {
                "day2:proof" => managed_day2 && matches!(old, "node scripts/day2-proof.mjs" | "node scripts/day2-proof.mjs versions/v1.q versions/v2.q"),
                "backend" => old.strip_prefix("snapback4 dev --port ").is_some_and(|n| n.parse::<u16>().is_ok()),
                "dev" | "preview" => old.strip_prefix(if key == "dev" { "vite --port " } else { "vite preview --port " }).and_then(|s| s.strip_suffix(" --strictPort")).is_some_and(|n| n.parse::<u16>().is_ok()),
                "start" | "web" => old.strip_prefix(if key == "start" { "expo start --port " } else { "expo start --web --port " }).is_some_and(|n| n.parse::<u16>().is_ok()),
                _ => false,
            });
            if managed || ((key == "day2:proof" && enable_day2 || key == "frontend:proof" && super::proof_advice::known_frontend(dir) || key == "playwright-core" && (super::proof_advice::known_frontend(dir) || super::proof_advice::referenced(dir, "scripts/frontend-proof.mjs"))) && old.is_none()) { package[group][key] = value.clone(); }
        }
    }
    let mut config_changes = Vec::new();
    if web {
        if let Ok(current) = std::fs::read_to_string(dir.join("vite.config.mts")) {
            // Replace only the exact initializer forms, allowing their port literals.
            let numbers = current.split("port: ").skip(1).filter_map(|s| s.split(|c: char| !c.is_ascii_digit()).next()?.parse::<u16>().ok()).collect::<Vec<_>>();
            let mut normalized = current.clone();
            for number in &numbers { normalized = normalized.replace(&format!("port: {number}"), "port: 5173"); }
            let reference = include_str!("starter/web/vite.config.mts").replace("port: 4173", "port: 5173");
            if normalized == reference { config_changes.push(("vite.config.mts", runtime_vite(&current, backend, numbers.first().copied().unwrap_or(5173)))); }
        }
    }
    let inherited_frontend = previous_frontend(dir, web).filter(|template| template.is_some()
        || super::adapt::fits_todos(own) && refresh_default_frontend);
    if let Some(template) = inherited_frontend {
        if let Some((_, text)) = generated.iter_mut().find(|(name, _)| *name == "scripts/frontend-proof.mjs") {
            *text = frontend_proof(include_str!("frontend-proof.mjs"), template, web);
        }
    }
    let referenced_after = |path: &str| package["scripts"].as_object().is_some_and(|scripts| scripts.values().any(|v|v.as_str().is_some_and(|s|s.contains(path))));
    let mut additions: Vec<_> = generated.into_iter().skip(1)
        .filter(|(name, text)| {
            if *name == "scripts/frontend-proof.mjs" {
                return (!default_todos || refresh_default_frontend)
                    && (inherited_frontend.is_some() || !dir.join(name).exists() && referenced_after(name));
            }
            if *name == "scripts/day2-proof.mjs" { return day2 && (managed_day2 || !dir.join(name).exists() && (enable_day2 || referenced_day2))
                && std::fs::read_to_string(dir.join(name)).ok().as_deref()!=Some(text.as_str()); }
            if !dir.join(name).exists() {
                return name.starts_with("versions/") || name.starts_with("v1/") || referenced_after(name)
                    || shape.password && *name=="scripts/staging.mjs";
            }
            false
        }).collect();
    additions.extend(config_changes);
    if package != original { additions.push(("package.json", pretty(&package))); }
    Ok(additions)
}

fn launcher_port(dir: &std::path::Path, name: &str) -> Option<u16> {
    let text = std::fs::read_to_string(dir.join("scripts/launch.mjs")).ok()?;
    // Recognize the generated fallback expression, not client application code.
    let marker = if name == "backend" { "low === null ? " } else { ": low === null ? " };
    text.split_once(marker)?.1.split_whitespace().next()?.parse().ok()
}


pub(super) fn apply(dir: &std::path::Path, additions: &[(&str, String)]) -> std::io::Result<()> {
    let frozen:Vec<_>=additions.iter().filter(|(name,_)|name.starts_with("v1/")).collect();
    if !frozen.is_empty() {
        if dir.join("v1").symlink_metadata().is_ok() {return Err(std::io::Error::other("v1 already exists; no historical source replaced"));}
        let nonce=std::time::SystemTime::now().duration_since(std::time::UNIX_EPOCH).unwrap_or_default().as_nanos();
        let stage=dir.join(format!(".snapback4-frozen-{}-{nonce}",std::process::id()));
        std::fs::create_dir(&stage)?;
        let result=(|| -> std::io::Result<()> {
            for (name,body) in &frozen {
                let relative=std::path::Path::new(name.strip_prefix("v1/").unwrap());
                if relative.components().any(|c|!matches!(c,std::path::Component::Normal(_))) {return Err(std::io::Error::other("frozen destination escapes v1"));}
                let path=stage.join(relative);
                std::fs::create_dir_all(path.parent().unwrap())?;
                std::fs::write(path,body)?;
            }
            if dir.join("v1").symlink_metadata().is_ok() {return Err(std::io::Error::other("v1 appeared during capture; no historical source replaced"));}
            std::fs::rename(&stage,dir.join("v1"))
        })();
        if result.is_err() {let _=std::fs::remove_dir_all(&stage);}
        result?;
    }
    for (name, body) in additions {
        if name.starts_with("v1/") {continue;}
        let path = dir.join(name);
        std::fs::create_dir_all(path.parent().unwrap())?;
        if *name == "package.json" || *name == "vite.config.mts" || *name == "scripts/frontend-proof.mjs" || *name == "scripts/day2-proof.mjs" { super::replace_file(&path, body)?; }
        else {
            use std::io::Write;
            let mut file = std::fs::OpenOptions::new().write(true).create_new(true).open(&path)?;
            file.write_all(body.as_bytes())?;
        }
    }
    Ok(())
}

/// Environment is evaluated whenever Vite loads the config, never at init time.
pub(super) fn runtime_vite(text: &str, backend: u16, frontend: u16) -> String {
    if text.contains("const proofRange") { return text.to_owned(); }
    let header = format!(r#"const proofRange = [process.env.PORT_LO, process.env.PORT_HI];
if (proofRange.some(v => v !== undefined) && (!proofRange.every(v => v && /^\d+$/.test(v)) || Number(proofRange[0]) < 1 || Number(proofRange[1]) > 65535 || Number(proofRange[0]) >= Number(proofRange[1]))) throw Error("PORT_LO/PORT_HI must provide at least two valid ports");
const backendPort = proofRange[0] === undefined ? {backend} : Number(proofRange[0]);
const frontendPort = proofRange[0] === undefined ? {frontend} : backendPort + 1;
process.env.VITE_SNAPBACK_URL ??= `http://127.0.0.1:${{backendPort}}`;
"#);
    let mut result = text.to_owned();
    // All generated configurations have literal ports; preserve unrelated config.
    for port in [frontend, 5173, 4173] { result = result.replace(&format!("port: {port}"), "port: frontendPort"); }
    format!("{header}{result}")
}

# llp-orient

<!-- @ref https://github.com/ccheever/llp/blob/v0.5.2/llp/0008-distributed-agent-skills.rfc.md#the-skills--core-five-plus-flags — skill contract -->
<!-- @ref https://github.com/ccheever/llp/blob/v0.5.2/llp/0000-linked-literate-programming.explainer.md#2-reference-syntax — the grammar this reads -->

Load the design rationale governing the area you're about to touch, before touching it. This is the consumption side of LLP — orienting first is what turns `@ref` from a possibility into a habit.

Invoke as `/llp-orient <path-or-task>` (file, directory, subsystem, or task description). With no argument, use the current change scope (`git diff --name-only`).

## Trigger

Start of a coding task that touches an area with documented design. The orient-first *rule* lives in the repo's always-loaded `AGENTS.md`; this skill is the detailed form that rule routes to for non-trivial work — don't rely on ambient description-match alone.

## Invariants

- MUST be read-only: never edit code or LLP documents.
- MUST scope to the task: pull the constraints that bind this change, not a corpus summary.
- MUST cite, not restate: reference policy as `LLP NNNN#anchor` rather than paraphrasing sections.
- MUST NOT invent constraints. If something isn't documented, say so.
- MUST NOT guess the rationale behind a broken `@ref` — report it and hand off to `llp-maintain`.

## Workflow

> **Recipe (advisory)** — corpus root = nearest `llp/` walking up from the path in scope (LLP 0001 §Unit of adoption); "no corpus" is an empty walk, not cwd-relative absence. Read `llp/foundation/` first (the kernel; the root LLP is there once `Active` — on a fresh adoption it is still `Draft` and sits in `current/`), then skim `llp/current/` (the declared working set); both hold relative links — symlinks, or small files containing the relative path — so `ls` answers and either form resolves. Then the root LLP's systems vocabulary. `Grep '@ref'` over the files in scope and `Read` each target — these are the highest-signal context. `Grep` the `**Systems:**` headers across `llp/` for LLPs governing the subsystems touched; read the governing ones (Spec, Decision, Principles, relevant RFCs). Follow `Related:` and nearby `@ref` targets one hop out; stop there unless something is clearly load-bearing. Stale-pin nudge (skip offline): read the installed tag from the skills receipt (or a `source:` pin), compare `git ls-remote --tags <origin>` — if upstream is newer, say so in one line ("skills at v0.3.0, upstream v0.5.2 — run /llp-adopt update") and move on; never update from here.

## Artifact

A context pack with these sections, faster to read than the documents it points at:

- **Scope** — files / subsystems queried
- **Governing LLPs** — numbered list, one-line gloss each; mark which are in `foundation/` and which are in `current/` (in-flight — its guidance may still move)
- **Active constraints** — bullets, each citing `LLP NNNN#anchor`
- **Relevant rationale** — the *why*, same citation form
- **Open questions / risks** for this area
- **Likely doc updates** if the change lands
- **Hand-off** — one of `none | llp-adopt | llp-create | llp-review | llp-maintain`

## Hand-offs

- No corpus (no `llp/` walking up from the path in scope) → `llp-adopt`.
- The task needs a decision no LLP covers → `llp-create` (or `llp-review` if proposal-shaped).
- An `@ref` target or anchor doesn't resolve → `llp-maintain`.

## Summary

Snapback 4 is one of two parallel candidates for the successor to Snapback 2,
and the sibling of Snapback 3, not its successor. It builds
[LLP 2000.000](./2000.000-keep-the-core-flip-two-substrates.rfc.md) as

resources metered at runtime instead of priced statically, and a client that
replicates the viewer's declared partition instead of caching receipted
results — proved first on the hub fixture, with a deterministic simulator as
the primary harness. Snapback 3 ([LLP 2000](./2000-snapback3.rfc.md), led by
Codex / GPT-6 Astra) keeps the same two questions open and discovers their
answers in its first slice. Snapback 2 ([LLP 1000](./1000-snapback2.explainer.md))
continues as the incumbent and the control. All three are judged by one
pre-registered comparison on one clock.

This document records the author's decisions of 2026-09-12 that created the
track, restates the bet on one page so a reader can orient without the brief,
fixes the bakeoff protocol, draws the fence, and plans the first four weeks.
The design argument and its evidence stay in LLP 2000.000. Only running code
will bind (`rules/RULES.md`).
