//! Candidate metadata is executable: never erase it merely to admit evolution.
// @ref LLP 3000.009.001.000#9 — column evolution changes derived prices, not arbitrary execution declarations.
use crate::{Program, RuntimeError, Schema, ir::ProgramKind, schema::{Execution, ResidencySite, SiteClass}};
use std::collections::BTreeMap;

pub(super) fn ordinary(schema: &Schema) -> Schema {
    let mut ordinary = schema.clone();
    ordinary.tables.retain(|name, _| schema.private_owner(name).is_none());
    ordinary.maintains.retain(|m| m.private.is_none());
    for table in ordinary.tables.values_mut() {
        table.columns.retain(|field, _| !table.maintained.contains_key(field));
        table.indexes.retain(|_, index| !index.columns().iter().any(|field| table.maintained.contains_key(field)));
        table.maintained.clear();
    }
    // Execution is compared separately against unchanged programs below. Prices
    // and the receipt graph must be recalculated on this reduced declaration.
    ordinary.operations.clear();
    ordinary.source_reservations.clear();
    ordinary.receipt_dependencies.clear();
    ordinary.prepare_contribution_indexes();
    ordinary.prepare_receipt_dependencies();
    ordinary
}

pub(super) fn prepare(schema: &mut Schema) -> Result<(), RuntimeError> {
    let mut derived = schema.clone();
    // The fixed-point helper also serves projected device schemas, where it
    // deliberately preserves seeds. A full server schema derives from nothing.
    derived.receipt_dependencies.clear();
    derived.prepare_receipt_dependencies();
    if !schema.receipt_dependencies.is_empty() && schema.receipt_dependencies != derived.receipt_dependencies {
        return Err(RuntimeError::Schema("receipt dependencies differ from the schema derivation".into()));
    }
    if !schema.source_reservations.is_empty() && schema.source_reservations != derived.source_reservations {
        return Err(RuntimeError::Schema("source reservations differ from the schema derivation".into()));
    }
    schema.receipt_dependencies = derived.receipt_dependencies;
    schema.source_reservations = derived.source_reservations;
    Ok(())
}

// `held` is an author declaration absent from IR, not a derivable runtime hint.
// Preserve all execution declarations for unchanged programs. Only diagnostic
// locations may move when a maintained declaration adds/removes source lines.
pub(super) fn check_operations(previous: &Schema, next: &Schema,
    before: &BTreeMap<String, Program>, after: &BTreeMap<String, Program>,
) -> Result<(), RuntimeError> {
    // Reopening a stored backend may only refresh derived
    // layout/reservations. Preserve its historical execution declarations;
    // it is not a request to replace them with this compiler's newer facts.
    // One-way equality includes every operation fact, so a caller cannot use
    // this path to change held, prediction, capability flags or declarations.
    let mut prepared_previous = previous.clone();
    prepared_previous.prepare_contribution_indexes();
    prepared_previous.prepare_delivery_lookup();
    if before == after && prepared_previous == *next { return Ok(()); }
    // General SQLite evolution may deliberately change residency through rules,
    // sync, etc. This check governs only the otherwise-unchanged declarations
    // admitted by the Postgres column slice, including metadata-only forgeries.
    if ordinary(previous) != ordinary(next) { return Ok(()); }
    for name in previous.operations.keys().chain(next.operations.keys()).filter(|name| !before.contains_key(*name) && !after.contains_key(*name)) {
        if previous.operations.get(name) != next.operations.get(name) {
            return Err(RuntimeError::Schema(format!("{name}: execution facts changed without a program change")));
        }
    }
    fn unlocated(mut value: serde_json::Value) -> serde_json::Value {
        fn walk(value: &mut serde_json::Value) {
            match value {
                serde_json::Value::Object(fields) => {
                    if fields.contains_key("Literal") { return; }
                    fields.remove("site");
                    for value in fields.values_mut() { walk(value); }
                    if let Some(serde_json::Value::Array(sites)) = fields.get_mut("sites") { sites.sort_by_key(serde_json::Value::to_string); }
                }
                serde_json::Value::Array(values) => for value in values { walk(value); },
                _ => {}
            }
        }
        walk(&mut value); value
    }
    for (name, program) in after {
        let Some(old) = before.get(name) else { continue; };
        if unlocated(serde_json::to_value(old).expect("program serialization"))
            != unlocated(serde_json::to_value(program).expect("program serialization")) { continue; }
        let expected = previous.operations.get(name).map(|fact| rederive(next, program, fact.clone()));
        let old = unlocated(serde_json::to_value(expected).expect("execution serialization"));
        let new = unlocated(serde_json::to_value(next.operations.get(name)).expect("execution serialization"));
        if old != new {
            return Err(RuntimeError::Schema(format!("{name}: execution facts changed without a program change")));
        }
    }
    Ok(())
}

// Ordinary declarations and the program are unchanged. Retain their admitted
// execution declarations, and re-derive precisely the facts column maintenance
// can affect. In particular, an added receipt-dependent sum makes its source
// mutation never-predicting even though that mutation's IR did not change.
fn rederive(schema: &Schema, program: &Program, mut fact: Execution) -> Execution {
    let was_server = fact.sites.iter().any(|site| site.class == SiteClass::Server);
    fact.sites.retain(|site| site.kind != "source reservation" && site.kind != "deferred receipt maintenance"
        && !(site.kind == "receipt dependency" && site.reason.starts_with("write to ") && site.reason.contains(" evaluates receipt-derived maintenance ")));
    for (table, effect) in crate::ir::write_tables(program) {
        if let Some(reason) = schema.write_prediction_refusal(&table) {
            fact.sites.push(ResidencySite { site: String::new(), table: table.clone(), index: String::new(),
                kind: "source reservation".into(), class: SiteClass::Server, reason: reason.into(),
                rewrite: "queue this mutation for the server's complete maintenance".into() });
        }
        for maintain in schema.maintains.iter().filter(|maintain| maintain.private.is_none()
            && schema.receipt_dependencies.contains_key(&maintain.target)
            && (maintain.source == table || maintain.row_key.as_ref().is_some_and(|key| key.table == table))) {
            fact.sites.push(ResidencySite { site: String::new(), table: table.clone(), index: String::new(),
                kind: "deferred receipt maintenance".into(), class: SiteClass::Held,
                reason: format!("receipt-derived maintenance {} stays settled during prediction; affected groups report incomplete until settlement", maintain.target), rewrite: String::new() });
        }
        if program.kind == ProgramKind::Mutation && !fact.sites.iter().any(|site| site.kind == "receipt dependency") {
            if let Some(reason) = schema.receipt_maintenance_refusal(&table, effect) {
                fact.sites.insert(0, ResidencySite { site: String::new(), table, index: String::new(),
                    kind: "receipt dependency".into(), class: SiteClass::Server, reason,
                    rewrite: "queue this mutation for server settlement; predict using only non-clause fields and stable ids".into() });
            }
        }
    }
    let server = fact.sites.iter().any(|site| site.class == SiteClass::Server);
    if program.kind == ProgramKind::Mutation && fact.predicts.is_some() && server != was_server {
        fact.predicts = Some(if server { "never" } else { "eligible" }.into());
    }
    fact
}

impl Handle {
    pub(super) fn connect(config: postgres::Config, name: &str, fresh: bool) -> Result<Self, RuntimeError> {
        Self::connect_inner(config, name, fresh, None)
    }
    pub(super) fn connect_owned(config: postgres::Config, name: &str, fence: Option<super::ownership::Fence>) -> Result<Self, RuntimeError> {
        Self::connect_inner(config, name, false, fence)
    }
    fn connect_inner(config: postgres::Config, name: &str, fresh: bool, fence: Option<super::ownership::Fence>) -> Result<Self, RuntimeError> {
        let quoted = store::namespace(name)?;
        let (mut client, warning) = super::ownership::connect(config.clone())?;
        if fresh {
            if client.query_one("SELECT to_regnamespace($1) IS NOT NULL", &[&name])?.get::<_, bool>(0) {
                return Err(RuntimeError::Store("Postgres namespace already exists".into()));
            }
        }
        #[cfg(any(test, feature = "test-support"))]
        crate::round_trips::postgres(1);
        client.batch_execute(&format!("SET search_path = {quoted}, pg_catalog; SET statement_timeout = '5s'; SET lock_timeout = '5s'"))?;
        let mut store = store::Store::new(client);
        store.unified = true;
        store.commit_warning = warning;
        store.fence = fence;
        if !fresh {
            if store.fence.is_none() {
                #[cfg(any(test, feature = "test-support"))]
                crate::round_trips::postgres(2);
                let id = store.client.query_opt("SELECT value FROM _meta WHERE key='store_id'", &[])?.map(|r| r.get::<_, String>(0));
                if let Some(store_id) = id { store.fence = Some(super::ownership::Fence { store_id, holder: "0000".into(), token: 0 }); }
            }
            if let Some(fence) = &store.fence {
                #[cfg(any(test, feature = "test-support"))]
                crate::round_trips::postgres(2);
                store.client.execute("SELECT set_config('application_name',$1,false)", &[&fence.tag()?])?; }
            let tx = store.opening_transaction()?;
            let row = tx.query("SELECT version,schema,programs,generation FROM _backend WHERE singleton", &[])?.remove(0);
            if row.get::<_, &str>(0) != store::VERSION { return Err(RuntimeError::StoreUnsupported { family: "opening" }); }
            let schema: Schema = store::decode(row.get(1))?;
            let programs = store::decode(row.get(2))?;
            super::profile::admit(&schema, &programs)?;
            let generation = u64::try_from(row.get::<_, i64>(3)).map_err(|_| RuntimeError::Generation("negative generation".into()))?;
            tx.rollback()?;
            store.generation = generation;

        }
        Ok(Self { store: RefCell::new(store), report: AdoptionReport::default(), connection: (config, name.into()), fresh, bootstrap_id: None })
    }

    pub(crate) fn activate<R>(&mut self, schema: &Schema, programs: Option<&BTreeMap<String, Program>>, reset: bool, now: i64,
        check: impl Fn(&AnyPreparation<'_, '_>) -> Result<(), RuntimeError>,
        mut build: impl FnMut(&AnyTx<'_>, bool, &mut AdoptionReport) -> Result<(), RuntimeError>,
        finish: impl Fn(&AnyTx<'_>) -> Result<R, RuntimeError>,
    ) -> Result<Committed<(u64, R)>, RuntimeError> {
        let programs = programs.ok_or(RuntimeError::StoreUnsupported { family: "release-activation" })?;
        super::profile::admit(schema, programs)?;
        // The adapter can also be called below Engine's mutable preparation.
        // Its immutable input must already contain exactly the facts published.
        let mut derived = schema.clone();
        crate::engine::activation_facts::prepare(&mut derived)?;
        if derived.source_reservations != schema.source_reservations || derived.receipt_dependencies != schema.receipt_dependencies {
            return Err(RuntimeError::Schema("activation requires schema-derived reservations and receipt dependencies".into()));
        }
        if reset { return Err(RuntimeError::StoreUnsupported { family: "dev-reset" }); }
        let fresh = self.fresh;
        let bootstrap_fence = fresh.then(|| super::ownership::Fence {
            store_id: self.bootstrap_id.clone().unwrap_or_else(|| format!("{:016x}{:016x}", crate::ids::seed(), crate::ids::seed())), holder: "0000".into(), token: 0,
        });
        let quoted = store::namespace(&self.connection.1)?;
        let mut report = AdoptionReport::default();
        let result = Committer::run_committed_with(self.store.get_mut(), |store| Committer::run_solo(Kind::Administrative, |solo| {
            let expected_generation = store.generation;
            let tx = solo.open_solo_with(|| store.opening_transaction())?;
            if fresh {
                tx.batch(&format!("CREATE SCHEMA {quoted}; SET search_path = {quoted}, pg_catalog;"))?;
                if let Some(fence) = &bootstrap_fence {
                    super::ownership::install(&tx, &fence.store_id)?;
                    tx.query("SELECT token FROM _owner WHERE singleton FOR KEY SHARE", &[])?;
                }
                tx.batch(store::DDL)?;
                tx.execute_one("INSERT INTO _backend(singleton,version,schema,programs) VALUES (true,$1,$2,$3)", &[&store::VERSION, &crate::value::canonical_json(schema), &crate::value::canonical_json(programs)])?;
                tx.batch("ALTER TABLE _backend ADD COLUMN generation BIGINT NOT NULL DEFAULT 1; ALTER TABLE _backend ADD COLUMN through_shortened TEXT NOT NULL DEFAULT '{}'; ALTER TABLE _backend ADD COLUMN maintained_identity_version TEXT;")?;
            }
            let head = tx.query("SELECT seq FROM _head WHERE singleton FOR UPDATE", &[])?[0].get::<_, i64>(0) as u64;
            tx.head.set(Some(head)); tx.serialized_head.set(true);
            let held = tx.query("SELECT generation,schema,programs FROM _backend WHERE singleton", &[])?.remove(0);
            let before = Backend(held.get(0), held.get(1), held.get(2));
            if !fresh && before.0 != expected_generation as i64 { return Err(RuntimeError::Generation("another engine activated this store".into())); }
            // Admission precedes every preparation write, under the head fence.
            let previous: Schema = store::decode(&before.1)?;
            let previous_programs = store::decode(&before.2)?;
            crate::engine::activation_facts::check_operations(&previous, schema, &previous_programs, programs)?;
            let schema_changed = !fresh && previous.hash() != schema.hash();
            if schema_changed { super::column_evolution::admit(&previous, schema)?; }
            initialize(&tx, schema)?;
            if let Some(fence) = &bootstrap_fence {
                tx.set_meta("store_id", &fence.store_id)?;
                tx.execute("SELECT set_config('application_name',$1,false)", &[&fence.tag()?])?;
            }
            let store_id = tx.meta("store_id")?.ok_or(RuntimeError::WorkerStoreIdentity)?;
            super::ownership::install(&tx, &store_id)?;
            let changed = fresh || schema_changed || before.2 != crate::value::canonical_json(programs);
            let generation = if !fresh && changed { before.0.checked_add(1).ok_or_else(|| RuntimeError::Generation("generation overflow".into()))? } else { before.0 };
            let tx = AnyTx::from_postgres(tx);
            tx.with_postgres(|inner| check(&AnyPreparation::Postgres(Preparation(inner))))?;
            if schema_changed { tx.with_postgres(|inner| super::column_evolution::project(inner, &previous, schema))?; }
            build(&tx, changed, &mut report)?;
            if schema_changed { tx.with_postgres(|inner| super::column_evolution::validate_groups(inner, &previous, schema))?; }
            tx.with_postgres(|inner| {
                inner.publish_backend(&Backend(generation, crate::value::canonical_json(schema), crate::value::canonical_json(programs)), now)?;
                inner.set_meta("schema_hash", &schema.hash())?;
                inner.set_meta("activation_hash", &crate::activation::activation_hash(schema, programs, inner.release_native_digest()?.as_deref()))?;
                inner.set_meta("time_audience_publications_v1", "1")
            })?;
            let value = finish(&tx)?;
            solo.commit(tx)?;
            Ok(Pending::new((generation as u64, value), generation as u64))
        }), |store, generation, value| { store.generation = generation; Ok(value) });
        if result.is_ok() {
            self.fresh = false; self.report = report;
            if let Some(fence) = bootstrap_fence { self.store.get_mut().fence = Some(fence); }
        }
        result
    }
}

pub(super) fn initialize(tx: &Tx<'_>, schema: &Schema) -> Result<(), RuntimeError> {
    super::activation_schema::verify_core(tx)?;
    // Apply the evaluator's portable key bound before accepting an existing
    // generation, as well as on each shared row/index write.
    if let Some(row) = tx.query(
        "SELECT t, ix, COUNT(*) FROM i WHERE octet_length(k) > $1 GROUP BY t, ix ORDER BY t, ix LIMIT 1",
        &[&(crate::rows::MAX_INDEX_KEY_BYTES as i32)],
    )?.first() {
        return Err(crate::rows::index_key_refusal(row.get(0), row.get(1), row.get::<_, i64>(2) as u64));
    }
    super::identity_schema::verify_existing(tx)?;
    super::metadata::initialize(tx)?;
    super::receipts::initialize(tx)?;
    super::change_evidence::initialize(tx)?;
    tx.batch(store::READ_SESSIONS_DDL)?;
    tx.batch(super::identity::DDL)?;
    super::requests::initialize(tx)?;
    super::outbox_schema::install(tx)?;
    super::caps::install(tx, schema)?;
    super::jobs::initialize(tx)?;
    super::assets_schema::initialize(tx)?;
    super::activation_schema::initialize(tx)?;
    super::release_schema::initialize(tx)
}

impl Tx<'_> {
    pub(crate) fn finish_open(self) -> Result<crate::store_port::OpeningMetadata, RuntimeError> {
        // A detached reader never upgrades. A writer retains its head fence and
        // all validated construction facts until the opening transaction commits.
        if self.serialized_head.get() {
            let schema: Schema = store::decode(&crate::activation::ActivationReader::opening_backend(&self)?.1)?;
            initialize(&self, &schema)?;
            let id = self.meta("store_id")?.ok_or(RuntimeError::WorkerStoreIdentity)?;
            super::ownership::install(&self, &id)?;
            let metadata = crate::store_port::OpeningMetadata::read(&self)?;
            self.commit()?;
            Ok(metadata)
        } else {
            let metadata = crate::store_port::OpeningMetadata::read(&self)?;
            self.rollback()?;
            Ok(metadata)
        }
    }
}

    pub fn synced(&self) -> impl Iterator<Item = (&str, &Table, &Sync)> {

            .filter_map(|(name, table)| table.sync.as_ref().map(|sync| (name.as_str(), table, sync)))

    pub fn maintain_for(&self, table: &str) -> Option<&Maintain> {

    }

    /// A source cap proves a bound on distinct derived groups; no target cap is enforced.
    pub fn group_bound(&self, name: &str, pinned: &[String]) -> Option<u32> {
        let table = self.table(name)?;
        let ordinary = table.constraints.iter().filter_map(|c| match c {
            Constraint::Cap { by, max } if by.iter().all(|field| pinned.contains(field)) && !table.cap_exempts_null(by) => Some(*max), _ => None,
        }).min();
        let derived = self.maintain_for(name).and_then(|m| {
            let source = self.table(&m.source)?;
            source.constraints.iter().filter_map(|c| match c {
                Constraint::Cap { by, max } if by.iter().all(|field| pinned.contains(field) && m.bare_key(field)) && !source.cap_exempts_null(by) => Some(*max), _ => None,
            }).min()
        });
        ordinary.into_iter().chain(derived).min()
    }

    /// Whether an `exists` probe's target row is structurally readable by the
    /// probing viewer (public target, or the read rule is proven true for the
    /// exact indexed row being probed). Copied from Snapback 2: the proof is
    /// structural on purpose, so a hidden probe never leaks by its result.
    pub fn rule_input_is_readable(&self, table_name: &str, index_name: &str, key: &[RuleTerm]) -> bool {
        let Some(table) = self.table(table_name) else { return false };
        if table.rules.read.is_uniformly_readable() {
            return true;
        }
        let Some(index) = table.index(index_name) else { return false };
        if !index.is_plain() {
            return false;
        }
        let columns = index.columns();
        prove_probe_row_readable(&table.rules.read, table_name, index_name, &columns, key)
    }
}

fn prove_probe_row_readable(rule: &RuleExpr, table: &str, index: &str, index_columns: &[String], probe_key: &[RuleTerm]) -> bool {
    let substitute = |term: &RuleTerm| match term {
        RuleTerm::Field(field) => index_columns.iter().position(|column| column == field).and_then(|position| probe_key.get(position)).cloned(),
        RuleTerm::OldField(_) | RuleTerm::NextField(_) => None,
        other => Some(other.clone()),
    };
    match rule {
        RuleExpr::Public { .. } | RuleExpr::Allow => true,
        RuleExpr::Eq(left, right) => substitute(left).zip(substitute(right)).is_some_and(|(left, right)| left == right),
        RuleExpr::And(rules) => !rules.is_empty() && rules.iter().all(|rule| prove_probe_row_readable(rule, table, index, index_columns, probe_key)),
        RuleExpr::Or(rules) => rules.iter().any(|rule| prove_probe_row_readable(rule, table, index, index_columns, probe_key)),
        RuleExpr::Exists { table: target, index: target_index, key, equalities, .. } if target == table && target_index == index && equalities.is_empty() => {
            key.iter().map(substitute).collect::<Option<Vec<_>>>().is_some_and(|key| key == probe_key)
        }
        RuleExpr::Deny | RuleExpr::Compare(..) | RuleExpr::Ne(_, _) | RuleExpr::Not(_) | RuleExpr::Exists { .. } | RuleExpr::Created { .. } => false,
    }
}

impl Table {
    pub fn index(&self, name: &str) -> Option<&Index> {
        self.indexes.get(name)
    }

    pub fn cap(&self, by: &[String]) -> Option<u32> {
        self.constraints.iter().find_map(|constraint| match constraint {
            Constraint::Cap { by: group, max } if group == by && !self.cap_exempts_null(group) => Some(*max),
            _ => None,
        })
    }

    /// A cap grouping by a `set null` reference does not count its null rows
    /// (LLP 3000.039 §2.2), so it proves no population bound.
    pub fn cap_exempts_null(&self, by: &[String]) -> bool {
        by.iter().any(|column| self.on_delete.get(column) == Some(&references::OnDelete::SetNull))
    }

    pub fn immutable_fields(&self) -> impl Iterator<Item = &str> {
        self.constraints.iter().flat_map(|constraint| match constraint {
            Constraint::Immutable { fields } => fields.iter().map(String::as_str).collect::<Vec<_>>(),
            _ => Vec::new(),
        })
    }

    /// Whether `row` is past its `expire` column at `now`: absent to every
    /// reader, pending the sweep. A null time never expires.
    pub fn expired(&self, row: &Value, now: i64) -> bool {
        self.deadline(row).is_some_and(|at| at <= now)
    }

    /// The index the expiry sweep walks: the first leading with the
    /// `expire` column, as the declaration requires.
    pub fn expiry_index(&self) -> Option<(&str, &Index)> {
        let column = self.expiry_column()?;
        self.indexes.iter().find(|(_, index)| {
                index.is_plain() && index.columns().first().map(String::as_str) == Some(column)
            }).map(|(name, index)| (name.as_str(), index))
    }

    /// The column carrying the principal a `sync to T[...]` audience reads.
    pub fn principal_columns(&self) -> Vec<&str> {
        self.columns.iter().filter(|(_, kind)| matches!(kind, ColumnType::Principal)).map(|(name, _)| name.as_str()).collect()
    }
}

/// `sync public`, `sync to .column`, or `sync to target[.prefix, ...]`, each
/// with an optional horizon `last N by index` (LLP 2000.000 §7.1, LLP 3000 §7).
#[derive(Clone, Debug, PartialEq, Eq, Serialize, Deserialize)]
pub struct Sync {
    /// The group arm when the table has one, else the first column arm.
    pub audience: Audience,
    #[serde(default, skip_serializing_if = "Option::is_none")]
    pub horizon: Option<Horizon>,
    /// A union's further arms, principal columns only: a device holds a
    /// row while any arm admits its viewer (LLP 3000.036.000.000 §2.3).
    #[serde(default, skip_serializing_if = "Vec::is_empty")]
    pub also: Vec<SyncArm>,
}

/// One further arm of a union audience. Its legacy window field is refused at admission.
#[derive(Clone, Debug, PartialEq, Eq, Serialize, Deserialize)]
pub struct SyncArm {
    pub audience: Audience,
    #[serde(default, skip_serializing_if = "Option::is_none")]
    pub horizon: Option<Horizon>,
}

#[derive(Clone, Debug, PartialEq, Eq, Serialize, Deserialize)]
pub enum Audience {
    /// Every principal holds the same partition.
    Public,
    /// The principal named by this row's column.
    Column(String),
    /// Every principal with a row in `table` whose `prefix` columns equal this
    /// row's `prefix` columns, read from the target's `principal` column.
    Target { table: String, prefix: Vec<String>, principal: String },
}

/// `last N by index`: a device holds at most N rows per group, where the group
/// is the audience prefix, in the index's descending order.
#[derive(Clone, Debug, PartialEq, Eq, Serialize, Deserialize)]
pub struct Horizon {
    pub last: u32,
    pub by: String,
}

impl Audience {
    /// The content columns that name this audience's group: none for the
    /// public, the column for a column audience, the prefix for a target.
    pub fn columns(&self) -> Vec<String> {
        match self {
            Audience::Public => Vec::new(),
            Audience::Column(column) => vec![column.clone()],
            Audience::Target { prefix, .. } => prefix.clone(),
        }
    }

    /// `.owner` or `members[.team] by .principal`, as one arm of a `sync to`
    /// line: the recipient is named, so the line binds when the target has
    /// several principal columns (round 3, GPT finding 4).
    pub fn arm(&self) -> String {
        match self {
            Audience::Public => "public".to_owned(),
            Audience::Column(column) => format!(".{column}"),
            Audience::Target { table, prefix, principal } => format!("{table}[{}] by .{principal}", prefix.iter().map(|c| format!(".{c}")).collect::<Vec<_>>().join(", ")),
        }
    }
}

impl Sync {
    pub fn single(audience: Audience, horizon: Option<Horizon>) -> Self {
        Self { audience, horizon, also: Vec::new() }
    }

    /// Whether this table syncs to a union of audiences.
    pub fn is_union(&self) -> bool {
        !self.also.is_empty()
    }

    /// Every arm: the first arm, then the union's horizon-free others.
    // @ref LLP 3000.036.000.000#2.3 — the group arm, when present, is always the first.
    pub fn arms(&self) -> impl Iterator<Item = (&Audience, Option<&Horizon>)> {
        std::iter::once((&self.audience, self.horizon.as_ref())).chain(self.also.iter().map(|arm| (&arm.audience, None)))
    }

    /// The principal columns of a union's column arms, in arm order.
    pub fn column_arms(&self) -> impl Iterator<Item = &str> {
        self.arms().filter_map(|(audience, _)| match audience { Audience::Column(column) => Some(column.as_str()), _ => None })
    }

    /// The arm whose window index is `index`, if any.
    pub fn single_window(&self, index: &str) -> Option<(&Audience, &Horizon)> {
        if self.is_union() { return None; }
        self.horizon.as_ref().filter(|h| h.by == index).map(|h| (&self.audience, h))
    }

    pub fn line(&self) -> String {
        let audience = if self.is_union() {
            // Column arms first, the group arm last; an empty-group roster
            // has no `to` spelling, so a union holding one is the inferred `sync`.
            let mut arms: Vec<&Audience> = self.arms().map(|(audience, _)| audience).collect();
            arms.sort_by_key(|audience| matches!(audience, Audience::Target { .. }));
            if let Some(Audience::Target { principal, .. }) = arms.iter().find(|audience| matches!(audience, Audience::Target { prefix, .. } if prefix.is_empty())) { format!("sync by .{principal}") }
            else { format!("sync to {}", arms.iter().map(|audience| audience.arm()).collect::<Vec<_>>().join(" or ")) }
        } else {
            match &self.audience {
                Audience::Public => "sync public".to_owned(),
                Audience::Column(column) => format!("sync to .{column}"),
                // A roster with no group columns has no `sync to` spelling: it is the inferred `sync`.
                Audience::Target { prefix, .. } if prefix.is_empty() => "sync".to_owned(),
                Audience::Target { table, prefix, .. } => format!("sync to {table}[{}]", prefix.iter().map(|c| format!(".{c}")).collect::<Vec<_>>().join(", ")),
            }
        };
        match &self.horizon {
            Some(horizon) => format!("{audience} last {} by {}", horizon.last, horizon.by),
            None => audience,
        }
    }

    /// A union has no window; single-audience lines keep their existing spelling.
    pub fn line_in(&self, _table: &Table) -> String { self.line() }

}

#[derive(Clone, Debug, PartialEq, Eq, Serialize, Deserialize)]
pub enum ColumnType {
    Id,
    Principal,
    Text { max: u32 },
    Bytes { max: u32 },
    Image { max: u32, #[serde(default, skip_serializing_if = "std::ops::Not::not")] once: bool },
    Video { max: u32 },
    // @ref LLP 3000.038#1 — an empty allow-list is the default: every verifiable type.
    File { max: u32, #[serde(default, skip_serializing_if = "Vec::is_empty")] types: Vec<String> },
    Int,
    // @ref LLP 3000.040#13-representation — a new tag fails closed on older decoders; `Int` keeps its bytes.
    IntRange { #[serde(default = "crate::schema::int_floor", skip_serializing_if = "crate::schema::is_int_floor")] min: i64, #[serde(default = "crate::schema::int_ceiling", skip_serializing_if = "crate::schema::is_int_ceiling")] max: i64 },
    Enum { variants: Vec<String> },
    Decimal { scale: u8 },
    Money { currency: String, scale: u8 },
    Bool,
    Time,
    Ref { table: String },
    Json { max_bytes: u32 },
    Optional(Box<ColumnType>),
    List { of: Box<ColumnType>, max: u32 },
    // @ref LLP 3000.007#5.4 — a new enum tag makes older engines refuse the minimum.
    ListRange { of: Box<ColumnType>, min: u32, max: u32 },
}
impl ColumnType {
    pub fn required(&self) -> &ColumnType {
        match self {
            Self::Optional(inner) => inner,
            other => other,
        }
    }

    pub fn is_optional(&self) -> bool { matches!(self, Self::Optional(_)) }

    /// An integer slot, bounded or not: the same value, order, key and arithmetic.
    pub fn is_int(&self) -> bool { matches!(self, Self::Int | Self::IntRange { .. }) }

    /// The value's kind without a slot bound: `IntRange` reads as `Int`.
    pub fn unbounded(&self) -> ColumnType {
        match self {
            Self::IntRange { .. } => Self::Int,
            Self::Optional(inner) => Self::Optional(Box::new(inner.unbounded())),
            other => other.clone(),
        }
    }

    pub fn is_indexable(&self) -> bool {
        match self {
            Self::Image { .. } | Self::Video { .. } | Self::File { .. } | Self::List { .. } | Self::ListRange { .. } | Self::Json { .. } => false,
            Self::Optional(inner) => inner.is_indexable(),
            _ => true,
        }
    }

    /// An asset leaf's kind, byte bound and (for a file) allow-list.
    pub fn asset_leaf(&self) -> Option<(crate::assets::AssetKind, u32, &[String])> {
        use crate::assets::AssetKind;
        match self {
            Self::Image { max, .. } => Some((AssetKind::Image, *max, &[])),
            Self::Video { max } => Some((AssetKind::Video, *max, &[])),
            Self::File { max, types } => Some((AssetKind::File, *max, types)),
            _ => None,
        }
    }

    pub fn contains_asset(&self) -> bool {
        match self {
            Self::Image { .. } | Self::Video { .. } | Self::File { .. } => true,
            Self::Optional(inner) | Self::List { of: inner, .. } | Self::ListRange { of: inner, .. } => inner.contains_asset(),
            _ => false,
        }
    }
}

impl fmt::Display for ColumnType {
    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
        match self {
            Self::Id => f.write_str("id"),
            Self::Principal => f.write_str("principal"),
            Self::Text { max } => write!(f, "text <={max}"),
            Self::Bytes { max } => write!(f, "bytes <={max}"),
            Self::Image { max, once: false } => write!(f, "image <={max}"),
            Self::Image { max, once: true } => write!(f, "image <={max} once"),
            Self::Video { max } => write!(f, "video <={max}"),
            Self::File { max, types } if types.is_empty() => write!(f, "file <={max}"),
            Self::File { max, types } => write!(f, "file <={max} ({})", types.iter().map(|t| format!("'{t}'")).collect::<Vec<_>>().join(", ")),
            Self::Int => f.write_str("int"),
            Self::IntRange { min, max } => f.write_str(&int_range_text(int_bound(*min, int_floor()), int_bound(*max, int_ceiling()))),
            Self::Enum { variants } => write!(f, "enum({})", variants.iter().map(|v| format!("'{v}'")).collect::<Vec<_>>().join(", ")),
            Self::Decimal { scale } => write!(f, "decimal {scale}"),
            Self::Money { currency, scale } => write!(f, "money {currency} {scale}"),
            Self::Bool => f.write_str("bool"),
            Self::Time => f.write_str("time"),
            Self::Ref { table } => f.write_str(table),
            Self::Json { max_bytes } => write!(f, "json <={max_bytes}"),
            Self::Optional(inner) => write!(f, "{inner} ?"),
            Self::List { of, max } => write!(f, "[{of}] <={max}"),
            Self::ListRange { of, min, max } => write!(f, "[{of}] {min}..{max}"),
        }
    }
}

/// `int M..N`, `int M..` or `int ..N`, the one spelling of each bound.
pub fn int_range_text(min: Option<i64>, max: Option<i64>) -> String {
    match (min, max) {
        (Some(min), Some(max)) => format!("int {min}..{max}"),
        (Some(min), None) => format!("int {min}.."),
        (None, Some(max)) => format!("int ..{max}"),
        (None, None) => "int".into(),
    }
}

// @ref LLP 3000.040#13-representation — an unwritten end is the i53 limit itself, so the
// variant stays as small as `Int`'s neighbours (no Option) and the bytes omit it.
pub fn int_floor() -> i64 { -crate::value::I53_MAX }
pub fn int_ceiling() -> i64 { crate::value::I53_MAX }
pub fn is_int_floor(min: &i64) -> bool { *min <= int_floor() }
pub fn is_int_ceiling(max: &i64) -> bool { *max >= int_ceiling() }
/// An int range some i53 value satisfies, with both ends inside i53.
pub fn int_range_valid(min: i64, max: i64) -> bool { int_floor() <= min && min <= max && max <= int_ceiling() }
/// A stored end as written: `None` when it is the i53 limit.
pub fn int_bound(end: i64, limit: i64) -> Option<i64> { (end != limit).then_some(end) }

/// Whether `value` lies inside an inclusive optional range.
pub fn int_in_range(value: i64, min: Option<i64>, max: Option<i64>) -> bool {
    min.is_none_or(|min| value >= min) && max.is_none_or(|max| value <= max)
}

#[derive(Clone, Debug, PartialEq, Eq, Serialize, Deserialize)]
pub struct Index {
    pub components: Vec<IndexComponent>,
    pub unique: bool,
}

impl Index {
    pub fn of(columns: &[&str], unique: bool) -> Self {
        Self { components: columns.iter().map(|c| IndexComponent::Column((*c).to_owned())).collect(), unique }
    }

    /// Ordered key names: columns and time-function names. A `words(..)`
    /// component contributes nothing here; `word_component` names it.
    /// Use `IndexComponent::column` for actual row fields (such as rule pins).
    pub fn columns(&self) -> Vec<String> {
        self.components.iter().filter_map(|c| c.key_name().cloned()).collect()
    }

    pub fn is_plain(&self) -> bool {
        self.components.iter().all(|c| matches!(c, IndexComponent::Column(_)))
    }

    pub fn uses_column(&self, name: &str) -> bool {
        self.components.iter().any(|component| match component {
            IndexComponent::Column(column) => column == name,
            IndexComponent::Calendar(key) => key.column == name,
            IndexComponent::WordPrefixes { fields, .. } => fields.iter().any(|field| field == name),
            IndexComponent::Contribution(maintain) => maintain.dependencies().contains(name),
        })
    }

    /// The `words(..)` component, if any: its position among the
    /// components and the text columns it tokenizes.
    pub fn word_component(&self) -> Option<(usize, &[String])> {
        self.components.iter().enumerate().find_map(|(position, component)| match component {
            IndexComponent::WordPrefixes { fields, .. } => Some((position, fields.as_slice())),
            IndexComponent::Column(_) | IndexComponent::Calendar(_) | IndexComponent::Contribution(_) => None,
        })
    }

    /// The columns after the first `components` components: a cursor's
    /// suffix, and the order a page continues in. Counted in components,
    /// so a scan whose prefix ends with a word skips the word too.
    pub fn suffix_after(&self, components: usize) -> Vec<String> {
        self.components.iter().skip(components).filter_map(|c| c.key_name().cloned()).collect()
    }
}

#[derive(Clone, Debug, PartialEq, Eq, Serialize, Deserialize)]
pub enum IndexComponent {
    /// Internal selected physical source rows; an optional following expiry component
    /// orders the group by deadline (null last), then the appended source id.
    Contribution(Box<Maintain>),
    Column(String),
    Calendar(crate::maintain::CalendarKey),
    WordPrefixes { fields: Vec<String>, #[serde(default)] contract: serde_json::Value },
}

impl IndexComponent {
    pub fn key_name(&self) -> Option<&String> {
        match self { Self::Calendar(key) => Some(&key.function), _ => self.column() }
    }
    pub fn key_type(&self, table: &Table) -> Option<ColumnType> {
        match self { Self::Calendar(_) => Some(ColumnType::Int), Self::Column(name) => table.columns.get(name).cloned(), _ => None }
    }
    pub fn value(&self, row: &Value) -> Option<Value> {
        match self { Self::Calendar(key) => Some(key.value(row)), Self::Column(name) => Some(row.field(name).cloned().unwrap_or(Value::Null)), _ => None }
    }
    pub fn column(&self) -> Option<&String> {
        match self {
            Self::Column(column) => Some(column),
            Self::Calendar(_) | Self::WordPrefixes { .. } | Self::Contribution(_) => None,
        }
    }
}

    },
}

impl RuleExpr {
    pub fn is_public(&self) -> bool {
        matches!(self, Self::Public { .. })
    }

    pub fn is_uniformly_readable(&self) -> bool {
        matches!(self, Self::Public { .. } | Self::Allow)
    }

    /// Every `.field` the rule reads, on any probe key or comparison.
    pub fn fields(&self, out: &mut BTreeSet<String>) {
        let term = |term: &RuleTerm, out: &mut BTreeSet<String>| {
            if let RuleTerm::Field(field) | RuleTerm::Reference { column: field, .. } | RuleTerm::ReferenceChain { column: field, .. } = term {
                out.insert(field.clone());
            }
        };
        match self {
            Self::Public { .. } | Self::Allow | Self::Deny => {}
            Self::Eq(left, right) | Self::Ne(left, right) | Self::Compare(_, left, right) => {
                term(left, out);
                term(right, out);
            }
            Self::And(items) | Self::Or(items) => items.iter().for_each(|item| item.fields(out)),
            Self::Not(inner) => inner.fields(out),
            Self::Exists { key, equalities, .. } | Self::Created { key, equalities, .. } => {
                key.iter().chain(equalities.iter().map(|(_, term)| term)).for_each(|item| term(item, out));
            }
        }
    }

    /// The disjuncts of the rule: an `or` flattens, anything else is itself.
    pub fn disjuncts(&self) -> Vec<&RuleExpr> {
        match self {
            Self::Or(items) => items.iter().flat_map(|item| item.disjuncts()).collect(),
            other => vec![other],
        }
    }

    // @ref LLP 3000.007#20.6 — service seats read on the server and never enlarge a device's audience.
    pub fn service_grant(&self) -> Option<&str> {
        match self {
            Self::Eq(RuleTerm::Viewer, RuleTerm::Literal(Value::Id(principal)))
            | Self::Eq(RuleTerm::Literal(Value::Id(principal)), RuleTerm::Viewer)
                if principal.strip_prefix("service:").is_some_and(|name| !name.is_empty()) => Some(principal),
            _ => None,
        }
    }

    /// Remove only entire service disjuncts; a conjunct, negation or ordinary
    /// principal equality must still pass the ordinary exact-audience check.
    pub fn device_audience_rule(&self) -> std::borrow::Cow<'_, RuleExpr> {
        let terms = self.disjuncts();
        if !terms.iter().any(|term| term.service_grant().is_some()) { return std::borrow::Cow::Borrowed(self); }
        let mut audience: Vec<_> = terms.into_iter().filter(|term| term.service_grant().is_none()).cloned().collect();
        std::borrow::Cow::Owned(match audience.len() {
            0 => Self::Deny,
            1 => audience.pop().expect("one audience"),
            _ => Self::Or(audience),
        })
    }

    /// The device audience of `table_name`'s read rule: service grants
    /// removed, then every same-row disjunct a self-probe absorbs (a members
    /// row read under `.principal = viewer or exists members[.group, viewer]`
    /// is its own witness, so the first disjunct adds no reader). What
    /// remains must still be one of the three invertible forms.
    // @ref LLP 3000.036.000#3.1 — absorption is exact equivalence, never a narrowing.
    pub fn sync_audience_rule(&self, table_name: &str, table: &Table) -> std::borrow::Cow<'_, RuleExpr> {
        let audience = self.device_audience_rule();
        let terms = audience.disjuncts();
        if terms.len() < 2 { return audience; }
        let absorbed = |term: &RuleExpr| terms.iter().any(|other| crate::engine::rules::self_probe_absorbs(table_name, table, term, other));
        // @ref LLP 3000.036.000.000#2.2 — a repeated branch (in either equality order) is one audience.
        let mut kept: Vec<RuleExpr> = Vec::new();
        for term in terms.iter().filter(|term| !absorbed(term)) {
            let normal = term.normal_disjunct();
            if !kept.iter().any(|seen| seen.normal_disjunct() == normal) { kept.push((*term).clone()); }
        }
        if kept.len() == terms.len() { return audience; }
        std::borrow::Cow::Owned(if kept.len() == 1 { kept.pop().expect("one audience") } else { Self::Or(kept) })
    }

    /// A disjunct with `viewer = .col` spelled `.col = viewer`, for folding repeats.
    pub fn normal_disjunct(&self) -> RuleExpr {
        match self {
            Self::Eq(RuleTerm::Viewer, other) => Self::Eq(other.clone(), RuleTerm::Viewer),
            other => other.clone(),
        }
    }

    /// The rule as Quarry text, for diagnostics and rewrites.
    pub fn quarry(&self) -> String {
        fn term(term: &RuleTerm) -> String {
            match term {
                RuleTerm::Reference { table, column, field, old } => format!("{table}[{}.{column}].{field}", if *old { "old" } else { "" }),
                RuleTerm::ReferenceChain { column, hops, old } => hops.iter().fold(format!("{}.{column}", if *old { "old" } else { "" }), |key, hop| format!("{}[{key}].{}", hop.table, hop.field)),
                RuleTerm::Field(f) => format!(".{f}"),
                RuleTerm::OldField(f) => format!("old.{f}"),
                RuleTerm::NextField(f) => format!("next.{f}"),
                RuleTerm::Viewer => "viewer".to_owned(),
                RuleTerm::Literal(Value::String(s)) => format!("'{s}'"),
                RuleTerm::Literal(Value::Id(s)) => format!("'{s}'"),
                RuleTerm::Literal(Value::Bool(b)) => b.to_string(), RuleTerm::Literal(Value::Int(n)) => n.to_string(), RuleTerm::Literal(Value::Null) => "null".to_owned(), // Quarry, never `{"Bool":true}`
                RuleTerm::Literal(v) => crate::value::canonical_json(v),
            }
        }
        match self {
            Self::Public { .. } => "public".to_owned(),
            Self::Allow => "allow".to_owned(),
            Self::Deny => "deny".to_owned(),
            Self::Eq(l, r) => format!("{} = {}", term(l), term(r)),
            Self::Compare(op,l,r) => format!("{} {} {}", term(l), match op { crate::ir::BinaryOp::Gte => ">=", crate::ir::BinaryOp::Gt => ">", crate::ir::BinaryOp::Lte => "<=", _ => "<" }, term(r)),
            Self::Ne(l, r) => format!("{} != {}", term(l), term(r)),
            Self::And(items) => items.iter().map(|i| if matches!(i,Self::Or(_)){format!("({})",i.quarry())}else{i.quarry()}).collect::<Vec<_>>().join(" and "),
            Self::Or(items) => items.iter().map(|i| i.quarry()).collect::<Vec<_>>().join(" or "),
            Self::Not(inner) => format!("not ({})", inner.quarry()),
            Self::Exists { table, key, equalities, .. } => format!("exists {table}[{}]", key.iter().map(term).chain(equalities.iter().map(|(col, value)| format!("{col} = {}", term(value)))).collect::<Vec<_>>().join(", ")),
            Self::Created { table, key, equalities, .. } => format!("created {table}[{}]", key.iter().map(term).chain(equalities.iter().map(|(col, value)| format!("{col} = {}", term(value)))).collect::<Vec<_>>().join(", ")),
        }
    }
}

#[derive(Clone, Copy, Debug, PartialEq, Eq, Serialize, Deserialize)]
#[serde(rename_all = "camelCase")]
pub enum TextFormat {
    Handle,
    Url,
    Email,
}

impl TextFormat {
    pub fn matches(self, text: &str) -> bool {
        match self {
            Self::Handle => text.bytes().all(|byte| byte.is_ascii_lowercase() || byte.is_ascii_digit() || byte == b'_'),
            Self::Url => text
                .strip_prefix("https://")
                .or_else(|| text.strip_prefix("http://"))
                .is_some_and(|rest| !rest.is_empty() && rest.chars().all(|c| !c.is_whitespace() && !c.is_control())),
            // @ref LLP 3000.013.001#1-decision — one deterministic ASCII mailbox grammar is shared by arguments, writes and devices.
            Self::Email => {
                if text.len() > 254 { return false; }
                let Some((local, domain)) = text.split_once('@') else { return false };
                if domain.contains('@') || local.is_empty() || local.len() > 64
                    || domain.is_empty() || domain.len() > 253
                    || local.starts_with('.') || local.ends_with('.') || local.contains("..")
                {
                    return false;
                }
                let local_ok = local.bytes().all(|b| b.is_ascii_alphanumeric()
                    || matches!(b, b'.' | b'!' | b'#' | b'$' | b'%' | b'&' | b'\'' | b'*'
                        | b'+' | b'/' | b'=' | b'?' | b'^' | b'_' | b'`' | b'{' | b'|'
                        | b'}' | b'~' | b'-'));
                local_ok && domain.split('.').all(|label| !label.is_empty() && label.len() <= 63
                    && !label.starts_with('-') && !label.ends_with('-')
                    && label.bytes().all(|b| b.is_ascii_alphanumeric() || b == b'-'))
            }
        }
    }
}

/// One problem with a schema, located by a dotted path into it.
#[derive(Clone, Debug, PartialEq, Eq)]
pub struct SchemaIssue {
    pub path: String,
    pub message: String,
}

/// Structural admission of a raw schema: every index names declared,
/// indexable columns; every rule and constraint address exists; every
/// maintain has the closed target shape; every `sync` is invertible.
pub fn validate(schema: &Schema) -> Vec<SchemaIssue> {
    let mut issues = columns::validate(schema);
    issues.extend(column_reads::validate(schema));
    #[cfg(feature = "sqlite")]
    {
    let mut backfill_outputs = BTreeSet::new(); let mut backfill_names = BTreeSet::new();
    for f in &schema.backfills {
        if let Err(message) = f.admit(schema) { issues.push(SchemaIssue { path:format!("backfills.{}.{}",f.table,f.name),message }); }
        if !backfill_names.insert((&f.table,&f.name)) || f.outputs.iter().any(|c|!backfill_outputs.insert((&f.table,c))) {
            issues.push(SchemaIssue { path:format!("backfills.{}.{}",f.table,f.name),message:"backfills require distinct names and disjoint outputs".into() });
        }
    }
    issues.extend(backfill::validate_downs(schema));
    for e in &schema.extractions {if let Err(message)=e.admit(schema){issues.push(SchemaIssue{path:format!("extractions.{}",e.child),message});}}
    }
    issues.extend(ephemeral::validate(schema));
    issues.extend(field_authority::validate(schema));
    issues.extend(through::validate(schema));
    issues.extend(live::validate(schema));
    issues.extend(references::validate(schema));
    timeline::validate(schema, &mut issues);
    time_audience::validate(schema, &mut issues);
    let issue = |issues: &mut Vec<SchemaIssue>, path: String, message: String| issues.push(SchemaIssue { path, message });
    #[cfg(feature="sqlite")]
    {
        let mut sources=BTreeSet::new();
        for (new,old) in &schema.table_renames {
            if new==old || !schema.tables.contains_key(new) || schema.tables.contains_key(old) || !sources.insert(old) {
                issue(&mut issues,format!("table_renames.{new}"),"table rename requires one distinct predecessor-only source".into());
            }
        }
        for (name,renames) in &schema.asset_renames {
            let mut sources=BTreeSet::new();
            let table=schema.table(name);
            for (new,old) in renames {
                let allowed=table.is_some_and(|t|old!=new && !t.columns.contains_key(old) && t.columns.get(new).is_some_and(|k|k.contains_asset()||matches!(k.required(),ColumnType::Ref{..})) && !t.engine_field(new) && !t.written_by.contains_key(new));
                if !allowed || !sources.insert(old) {issue(&mut issues,format!("asset_renames.{name}.{new}"),"asset rename requires a distinct predecessor-only source and an authoritative asset destination".into());}
            }
        }
    }
    let mut route_paths = BTreeSet::new();
    for (name, route) in &schema.routes {
        let path = format!("routes.{name}");
        if name.is_empty() || name.len() > 128 || name.split('.').any(|part| part.is_empty() || !part.bytes().enumerate().all(|(i, b)| b.is_ascii_alphabetic() || b == b'_' || (i > 0 && b.is_ascii_digit()))) {
            issue(&mut issues, path.clone(), "a route name is a qualified identifier of at most 128 bytes".into());
        }
        if let Err(message) = route.validate() { issue(&mut issues, path.clone(), message); }
        if !route_paths.insert(&route.path) { issue(&mut issues, path, "two route declarations publish the same POST path".into()); }
    }
    for (table_name, table) in &schema.tables {
        let path = format!("tables.{table_name}");
        if table_name.starts_with("_dormant_") || table_name.starts_with("_release_") { issue(&mut issues,path.clone(),"table names starting with _dormant_ or _release_ are reserved for release storage".into()); }
        for (column, parent) in &table.sequences {
            let valid = table.columns.get(column) == Some(&ColumnType::Int)
                && matches!(table.columns.get(parent), Some(ColumnType::Ref { table }) if schema.tables.contains_key(table))
                && !table.defaults.contains_key(column) && !table.maintained.contains_key(column)
                && !table.recorded.contains_key(column)
                && [column, parent].iter().all(|field| table.constraints.iter().any(|c| matches!(c, Constraint::Immutable { fields } if fields.contains(field))))
                && table.indexes.get(&sequence_index(column)) == Some(&Index::of(&[parent, column], false));
            if !valid { issue(&mut issues, format!("{path}.sequences.{column}"), "sequence by requires an int column, a required table reference, immutable ordinal and parent, and its sequence index; compile the declaration (guide types)".into()); }
        }
        for (field, capture) in &table.recorded {
            let target = schema.table(&capture.table);
            let valid = !table.defaults.contains_key(field) && field == &format!("{}.{}", capture.reference, capture.column)
                && !capture.reference.contains('.') && !capture.column.contains('.')
                && table.columns.get(&capture.reference).is_some_and(|kind| kind.required() == &ColumnType::Ref { table: capture.table.clone() })
                && table.immutable_fields().any(|name| name == capture.reference)
                && target.is_some_and(|target| target.rules.read.is_uniformly_readable()
                    && (capture.column == "id" || target.immutable_fields().any(|name| name == capture.column))
                    && target.columns.get(&capture.column).is_some_and(|kind| table.columns.get(field) == Some(&ColumnType::Optional(Box::new(kind.required().clone())))));
            if !valid { issue(&mut issues, format!("{path}.recorded.{field}"), "recorded attribution requires one immutable reference to a public target's immutable column or id, with an optional stored type; guide maintains".into()); }
        }
        if table.columns.get("id") != Some(&ColumnType::Id) {
            issue(&mut issues, format!("{path}.columns.id"), "every table has an intrinsic id column".into());
        }
        match table.indexes.get(BY_ID) {
            Some(index) if index.is_plain() && index.unique && index.columns() == ["id"] => {}
            _ => issue(&mut issues, format!("{path}.indexes.{BY_ID}"), "every table has the intrinsic unique byId index".into()),
        }
        for (name, index) in &table.indexes {
            let ipath = format!("{path}.indexes.{name}");
            if index.components.is_empty() || index.components.len() > 8 {
                issue(&mut issues, ipath.clone(), "declare between 1 and 8 components".into());
            }
            let mut seen = BTreeSet::new();
            let mut words = 0;
            for component in &index.components {
                match component {
                    IndexComponent::Column(column) => {
                        if !seen.insert(column) {
                            issue(&mut issues, ipath.clone(), format!("duplicate column {column}"));
                        }
                        match table.columns.get(column) {
                            None => issue(&mut issues, ipath.clone(), format!("undeclared column {column}")),
                            Some(kind) if !kind.is_indexable() => issue(&mut issues, ipath.clone(), format!("column {column} is {kind}, which cannot be indexed")),
                            _ => {}
                        }
                    }
                    IndexComponent::Calendar(key) => {
                        if !seen.insert(&key.function) || table.columns.contains_key(&key.function) {
                            issue(&mut issues, ipath.clone(), format!("computed index key {} is repeated or collides with a column", key.function));
                        }
                        if let Err(message) = key.validate(table) { issue(&mut issues, ipath.clone(), message); }
                    }
                    IndexComponent::Contribution(maintain) => {
                        let format_valid = index.components.len() == 1 || (index.components.len() == 2
                            && table.expiry_column().is_some_and(|expiry| index.components[1].column().map(String::as_str) == Some(expiry)));
                        if index.unique || !format_valid || maintain.source != *table_name || name != &maintain.contribution_index() {
                            issue(&mut issues, ipath.clone(), "invalid internal contribution index".into());
                        }
                        if let Err(message) = maintain.validate_keys(table) { issue(&mut issues, ipath.clone(), message); }
                    }
                    IndexComponent::WordPrefixes { fields, contract } => {
                        // One tokenizer, named: a store built under another
                        // contract would answer a different question.
                        words += 1;
                        if words > 1 {
                            issue(&mut issues, ipath.clone(), "an index has at most one words(..) component".into());
                        }
                        if index.unique {
                            issue(&mut issues, ipath.clone(), "a words(..) index cannot be unique: a row has one entry per word".into());
                        }
                        if fields.is_empty() {
                            issue(&mut issues, ipath.clone(), "words(..) names at least one text column".into());
                        }
                        if index.components.last().is_some_and(|last| last == component) {
                            issue(&mut issues, ipath.clone(), "words(..) is followed by the columns a page is ordered by, ending in id".into());
                        }
                        if contract.as_str() != Some(crate::words::CONTRACT) {
                            issue(&mut issues, ipath.clone(), format!("words(..) contract {contract} is not this engine's {}", crate::words::CONTRACT));
                        }
                        for field in fields {
                            match table.columns.get(field).map(ColumnType::required) {
                                Some(ColumnType::Text { .. } | ColumnType::Enum { .. }) => {}
                                Some(kind) => issue(&mut issues, ipath.clone(), format!("words({field}) needs a text or enum column; {field} is {kind}")),
                                None => issue(&mut issues, ipath.clone(), format!("words names undeclared column {field}")),
                            }
                        }
                    }
                }
            }
        }
        for (side, rule) in [("read", &table.rules.read), ("insert", &table.rules.insert), ("update", &table.rules.update), ("delete", &table.rules.delete)] {
            validate_rule(schema, table, rule, side == "update", side == "insert", &format!("{path}.rules.{side}"), &mut issues);
        }
        for (position, constraint) in table.constraints.iter().enumerate() {
            let cpath = format!("{path}.constraints[{position}]");
            match constraint {
                Constraint::Unique { index } => {
                    if !table.indexes.get(index).is_some_and(|i| i.unique) {
                        issue(&mut issues, cpath, format!("unique constraint names no unique index {index}"));
                    }
                }
                Constraint::Cap { by, max } => {
                    // @ref LLP 3000.007#20.3 — an empty group is the whole table, under the existing cap enforcement.
                    if *max == 0 {
                        issue(&mut issues, cpath.clone(), "cap needs a positive maximum".into());
                    }
                    for column in by {
                        if !table.columns.contains_key(column) {
                            issue(&mut issues, cpath.clone(), format!("cap names undeclared column {column}"));
                        }
                    }
                    if !table.indexes.values().any(|index| index.is_plain() && index.columns().starts_with(by)) {
                        issue(&mut issues, cpath, format!("cap by {} needs an index whose leading columns are exactly those", by.join(", ")));
                    }
                }
                Constraint::Immutable { fields } => {
                    for field in fields {
                        if !table.columns.contains_key(field) {
                            issue(&mut issues, cpath.clone(), format!("immutable names undeclared column {field}"));
                        }
                    }
                }
                Constraint::Text { field, .. } => {
                    if !matches!(table.columns.get(field).map(ColumnType::required), Some(ColumnType::Text { .. })) {
                        issue(&mut issues, cpath, format!("text constraint on non-text column {field}"));
                    }
                }
                Constraint::FinalExistsOnInsert { table: target, index, key } => {
                    for term in key { reference::validate_term(schema, table, term, false, &cpath, &mut issues); }
                    match schema.table(target).and_then(|t| t.index(index)) {
                        Some(target_index) if target_index.is_plain() && target_index.columns().len() == key.len() => {}
                        Some(_) => issue(&mut issues, cpath,
                        "after insert needs a plain index with matching key arity".into()),
                        None => issue(&mut issues, cpath, format!("after insert names unknown {target}.{index}")),
                    }
                }
                Constraint::AppendOnly | Constraint::Capabilities(_) | Constraint::Recorded => {}
            }
        }
        if let Some(sync) = &table.sync {
            validate_sync(schema, table_name, table, sync, &format!("{path}.sync"), &mut issues);
        }
        if let Some(column) = &table.expire {
            let epath = format!("{path}.expire");
            match table.columns.get(column).map(ColumnType::required) {
                Some(ColumnType::Time) => {}
                Some(kind) => issue(&mut issues, epath.clone(), format!("expire at .{column} needs a time column; {column} is {kind}")),
                None => issue(&mut issues, epath.clone(), format!("expire names undeclared column {column}")),
            }
            if table.expiry_index().is_none() {
                issue(&mut issues, epath, format!("expire at .{column} needs an index leading with {column}"));
            }
        }
        if let Some(column) = &table.not_before {
            let opath = format!("{path}.notBefore");
            match table.columns.get(column).map(ColumnType::required) {
                Some(ColumnType::Time) => {}
                Some(kind) => issue(&mut issues, opath.clone(), format!("not before .{column} needs a time column; {column} is {kind}")),
                None => issue(&mut issues, opath.clone(), format!("not before names undeclared column {column}")),
            }
            if table.opening_index().is_none() { issue(&mut issues, opath, format!("not before .{column} needs an index leading with {column}")); }
        }
    }
    for (position, maintain) in schema.maintains.iter().enumerate() {
        validate_maintain(schema, position, maintain, &mut issues);
        if let Err(message) = own_keys::validate(schema, maintain) {
            issues.push(SchemaIssue { path: format!("maintains[{position}]"), message });
        }
    }
    for issue in &mut issues {
        for maintain in schema.maintains.iter().filter(|m|m.private.is_some()) {
            issue.path = issue.path.replace(&maintain.target, &maintain.label());
            issue.message = issue.message.replace(&maintain.target, &maintain.label());
        }
    }
    issues
}

// @ref LLP 3000.036#11 — an insert rule may order the new row's own required
// int/time columns and integer literals: one image, no clock, no hop, no null.
// @ref LLP 3000.036#12 — decimal and money are integer minor units and order the same way.
fn insert_order(table: &Table, left: &RuleTerm, right: &RuleTerm) -> bool {
    let own = |term: &RuleTerm| match term {
        RuleTerm::Field(field) => table.columns.get(field).is_some_and(|kind| matches!(kind, ColumnType::Int | ColumnType::IntRange { .. } | ColumnType::Time | ColumnType::Decimal { .. } | ColumnType::Money { .. })),
        RuleTerm::Literal(Value::Int(_)) => true,
        _ => false,
    };
    let same = match (left, right) {
        (RuleTerm::Field(a), RuleTerm::Field(b)) => table.columns.get(a) == table.columns.get(b),
        _ => true,
    };
    own(left) && own(right) && same && !matches!((left, right), (RuleTerm::Literal(_), RuleTerm::Literal(_)))
}

fn validate_rule(schema: &Schema, table: &Table, rule: &RuleExpr, update: bool, insert: bool, path: &str, issues: &mut Vec<SchemaIssue>) {
    let check_term = |term: &RuleTerm, issues: &mut Vec<SchemaIssue>| reference::validate_term(schema, table, term, update, path, issues);
    match rule {
        RuleExpr::Public { .. } | RuleExpr::Allow | RuleExpr::Deny => {}
        RuleExpr::Eq(l, r) | RuleExpr::Ne(l, r) | RuleExpr::Compare(_, l, r) => {
            if matches!(rule, RuleExpr::Compare(..)) && !(update && table.update_transition) && !(insert && insert_order(table, l, r)) { issues.push(SchemaIssue { path: path.into(), message: "ordered comparisons require a named update transition, or an insert rule over the row's own required int, time, decimal or money columns (two columns of one type)".into() }); }
            check_term(l, issues);
            check_term(r, issues);
        }
        RuleExpr::And(items) | RuleExpr::Or(items) => items.iter().for_each(|item| validate_rule(schema, table, item, update, insert, path, issues)),
        RuleExpr::Not(inner) => validate_rule(schema, table, inner, update, insert, path, issues),
        RuleExpr::Exists { table: target, index, key, equalities, .. } | RuleExpr::Created { table: target, index, key, equalities, .. } => {
            if matches!(rule, RuleExpr::Created { .. }) && !insert {
                issues.push(SchemaIssue { path: path.to_owned(), message: "created probes are valid only in insert rules".into() });
            }
            match schema.table(target).and_then(|t| t.index(index)) {
                Some(target_index) => {
                    if !target_index.is_plain() {
                        issues.push(SchemaIssue {
                            path: path.to_owned(),
                            message: format!("probe of {target}.{index} needs a plain index"),
                        });
                    }
                    if target_index.columns().len() != key.len() {
                        issues.push(SchemaIssue { path: path.to_owned(), message: format!("probe of {target}.{index} has {} terms; the index has {}", key.len(), target_index.columns().len()) });
                    }
                }
                None => issues.push(SchemaIssue { path: path.to_owned(), message: format!("rule probes unknown {target}.{index}") }),
            }
            if !equalities.is_empty() {
                if !schema.table(target).is_some_and(|table| table.index(index).is_some_and(|index| index.unique && index.columns().iter().all(|col| table.columns.get(col).is_some_and(|ty| !ty.is_optional())))) {
                    issues.push(SchemaIssue { path: path.to_owned(), message: "extra probe equalities need a total unique key".into() });
                }
                for (column, term) in equalities {
                    if !schema.table(target).is_some_and(|table| table.columns.contains_key(column)) {
                        issues.push(SchemaIssue { path: path.to_owned(), message: format!("probe checks undeclared column {target}.{column}") });
                    }
                    check_term(term, issues);
                }
            }
            key.iter().for_each(|term| check_term(term, issues));
        }
    }
}

fn validate_maintain(schema: &Schema, position: usize, maintain: &Maintain, issues: &mut Vec<SchemaIssue>) {
    const GLOBAL: &str = "$global";
    let path = format!("maintains[{position}]");
    let issue = |issues: &mut Vec<SchemaIssue>, message: String| issues.push(SchemaIssue { path: path.clone(), message });
    let Some(source) = schema.table(&maintain.source) else {
        issue(issues, format!("maintain names unknown source table {}", maintain.source));
        return;
    };
    let Some(target) = schema.table(&maintain.target) else {
        issue(issues, format!("maintain names unknown target table {}", maintain.target));
        return;
    };
    if let Some(key)=&maintain.row_key {
        if let Err(message)=key.validate(schema,maintain) {issue(issues,message);}
        return;
    }
    // @ref LLP 3000.007#20.9 — stored value order cannot rank live sums between expiry and sweep; enforce at every schema entry point.
    if source.is_expiring() && target.indexes.values().any(|index| index.columns().iter().any(|column| column == "value")) {
        issue(issues, format!("maintain {} cannot order by value because source {} expires", maintain.target, maintain.source));
    }
    // The sweep, and a commit retiring its expired births, delete expired
    // rows as source writes; a maintained row is the engine's to write.
    if target.is_expiring() {
        issue(issues, format!("maintain {} cannot expire; give its source {} the lifetime", maintain.target, maintain.source));
    }
    if let Some(message) = maintain.expiring_read_refusal(schema) { issue(issues, message); }
    if let Err(message) = maintain.validate_keys(source) { issue(issues, message); }
    for (field, target_field) in maintain.by.iter().zip(maintain.group_fields()) {
        let calendar_type = if maintain.computed.get(field).is_some_and(|key| source.recorded.contains_key(&key.column)) { ColumnType::Optional(Box::new(ColumnType::Int)) } else { ColumnType::Int };
        let expected = if maintain.computed.contains_key(field) { Some(&calendar_type) } else { source.columns.get(maintain.aliases.get(field).unwrap_or(field)) };
        match expected {
            None => issue(issues, format!("maintain names unknown source column {}.{field}", maintain.source)),
            Some(kind) if target.columns.get(&target_field) != Some(kind) => issue(issues, format!("maintain target column {field} must match the source type {kind}")),
            _ => {}
        }
    }
    let expected_value = match &maintain.kind {
        MaintainKind::Count => Some(ColumnType::Int),
        MaintainKind::Sum { of } => match source.columns.get(of) {
            Some(kind @ (ColumnType::Int | ColumnType::Decimal { .. } | ColumnType::Money { .. })) => Some(kind.clone()),
            // @ref LLP 3000.040#13-representation — a bound is a fact about the source slot; a total is unbounded.
            Some(ColumnType::IntRange { .. }) => Some(ColumnType::Int),
            Some(kind) => {
                issue(issues, format!("sum over {}.{of} needs int, decimal, or money, not {kind}", maintain.source));
                None
            }
            None => {
                issue(issues, format!("sum names unknown source column {}.{of}", maintain.source));
                None
            }
        },
    };
    if let Some(expected) = expected_value
        && target.columns.get("value") != Some(&expected)
    {
        issue(issues, format!("maintain target {}.value must be {expected}", maintain.target));
    }
    let group_fields = maintain.group_fields();
    // @ref LLP 3000.013.001#3-global-maintained-view-representation — `$global` is admitted only on the exact compiler-owned ungrouped target.
    if maintain.by.is_empty() && target.columns.get(GLOBAL) != Some(&ColumnType::Bool) {
        issue(issues, format!("ungrouped maintain target {} needs its compiler-owned {GLOBAL} bool", maintain.target));
    }
    let allowed: BTreeSet<&str> = ["id", "value"].into_iter()
        .chain(group_fields.iter().map(String::as_str))
        .chain(maintain.by.is_empty().then_some(GLOBAL))
        .collect();
    let extra: Vec<_> = target.columns.keys().filter(|c| !allowed.contains(c.as_str())).cloned().collect();
    if !extra.is_empty() {
        issue(issues, format!("maintain target {} has non-aggregate columns {}", maintain.target, extra.join(", ")));
    }
    let group_index = if maintain.by.is_empty() { vec![GLOBAL.into()] } else { maintain.group_fields() };
    if !target.indexes.values().any(|index| index.unique && index.columns() == group_index) {
        issue(issues, format!("maintain target {} needs a unique index over {}", maintain.target, maintain.by.join(", ")));
    }
    // A delivered aggregate reveals a count: every recipient must be able to
    // read every contributing row, so the aggregate's audience is its
    // source's audience, grouped by the audience columns.
    // @ref LLP 3000.036.000.000#5.4 — a count over a union source is not a count over any one arm's group.
    // A union-synced aggregate would deliver a group's count to column-arm
    // readers who read none of the group's rows: aggregates sync one audience.
    if target.sync.as_ref().is_some_and(Sync::is_union) {
        issue(issues, format!("maintain target {} cannot sync to a union of audiences; it syncs its source's one audience, or is `online only`", maintain.target));
    }
    if target.sync.is_some() && source.sync.as_ref().is_some_and(Sync::is_union) {
        issue(issues, format!("maintain target {} cannot sync: its source {} syncs to a union of audiences, and a device would hold a count over rows only some of its readers hold; make the aggregate `online only`", maintain.target, maintain.source));
    }
    if let Some(target_sync) = &target.sync
        && !source.rules.read.is_uniformly_readable()
    {
        let preserves = |column: &String, source_column: &String| maintain.group_fields().contains(column)
            && if let Some(capture) = maintain.aliases.get(column) { capture == source_column } else { column == source_column && maintain.bare_key(column) };
        let sound = match (&target_sync.audience, source.sync.as_ref().map(|sync| &sync.audience)) {
            (Audience::Public, Some(Audience::Public)) => true,
            (Audience::Column(column), Some(Audience::Column(source_column))) => preserves(column, source_column),
            (Audience::Target { table, prefix, principal }, Some(Audience::Target { table: source_table, prefix: source_prefix, principal: source_principal })) => table == source_table && principal == source_principal && prefix.len() == source_prefix.len() && prefix.iter().zip(source_prefix).all(|(column, source_column)| preserves(column, source_column)),
            _ => false,
        };
        if !sound {
            issue(issues, format!("maintain target {} syncs to an audience its source {} does not deliver to; a device would hold a count over rows it may not read. Sync the source to the same audience and group by its columns, or make the source public", maintain.target, maintain.source));
        }
    }
}

/// Why a `sync` line cannot be honoured, as the term the author must change.
fn validate_sync(schema: &Schema, table_name: &str, table: &Table, sync: &Sync, path: &str, issues: &mut Vec<SchemaIssue>) {
    if sync.is_union() { return union::validate(schema, table_name, table, sync, path, issues); }
    let issue = |issues: &mut Vec<SchemaIssue>, message: String| issues.push(SchemaIssue { path: path.to_owned(), message });
    let audience_rule = table.rules.read.sync_audience_rule(table_name, table);
    // @ref LLP 3000.014#5 — a child admits exactly its substituted audience, never a public subset.
    if table.through.is_some() {
        let exact = match (audience_rule.as_ref(), &sync.audience) {
            (RuleExpr::Allow | RuleExpr::Public { .. }, Audience::Public) => true,
            (RuleExpr::Eq(RuleTerm::Field(field), RuleTerm::Viewer) | RuleExpr::Eq(RuleTerm::Viewer, RuleTerm::Field(field)), Audience::Column(column)) => field == column,
            (RuleExpr::Exists { table, key, equalities, .. }, Audience::Target { table: target, prefix, .. }) => table == target && equalities.is_empty() && key == &prefix.iter().cloned().map(RuleTerm::Field).chain([RuleTerm::Viewer]).collect::<Vec<_>>(),
            _ => false,
        };
        if !exact { issue(issues, "read through sync must exactly equal its inherited audience; use `sync` to infer it, or `online only`".into()); }
    }

    match &sync.audience {
        Audience::Public => {
            if !audience_rule.is_uniformly_readable() {
                issue(issues, format!("sync public needs a public read rule; {table_name} reads under `{}`", table.rules.read.quarry()));
            }
        }
        Audience::Column(column) => {
            match table.columns.get(column) {
                Some(ColumnType::Principal) => {}
                Some(kind) => issue(issues, format!("sync to .{column} needs a principal column, not {kind}")),
                None => issue(issues, format!("sync to .{column} names an undeclared column")),
            }
            let implied = RuleExpr::Eq(RuleTerm::Field(column.clone()), RuleTerm::Viewer);
            let implied_flipped = RuleExpr::Eq(RuleTerm::Viewer, RuleTerm::Field(column.clone()));
            // The rule is exactly the audience: a second disjunct would admit
            // viewers the partition never delivers to; a conjunct would deliver
            // rows the rule denies.
            if !audience_rule.is_uniformly_readable() && *audience_rule != implied && *audience_rule != implied_flipped {
                issue(issues, format!("sync to .{column} is only sound when the read rule is exactly `.{column} = viewer`; it reads under `{}`", table.rules.read.quarry()));
            }
        }
        Audience::Target { table: target_name, prefix, principal } => {
            let Some(target) = schema.table(target_name) else {
                issue(issues, format!("sync to {target_name} names an unknown table"));
                return;
            };
            if target.columns.get(principal) != Some(&ColumnType::Principal) {
                issue(issues, format!("sync to {target_name} reads principals from {target_name}.{principal}, which is not a principal column"));
            }
            for column in prefix {
                if !table.columns.contains_key(column) {
                    issue(issues, format!("sync to {target_name}[.{column}] names an undeclared column of {table_name}"));
                }
            }
            match audience_binding(schema, table_name, table, sync) {
                Err(why) => issue(issues, why),
                Ok(binding) if !audience_rule.is_uniformly_readable() => target_grant_issues(schema, table_name, target_name, target, principal, &binding, &mut |message| issue(issues, message)),
                _ => {}
            }
        }
    }
    sync_index_issues(table_name, table, sync, &sync.audience, sync.horizon.as_ref(), &mut |message| issue(issues, message));
}

/// The grant checks a private group audience needs on its target table: an
/// independent grant probe on every insert branch, and on every update
/// branch unless the audience columns are immutable.
// @ref LLP 3000#7-decisions-during-execution — every audience-granting branch requires prior authority
pub(super) fn target_grant_issues(schema: &Schema, table_name: &str, target_name: &str, target: &Table, principal: &str, binding: &AudienceBinding, issue: &mut dyn FnMut(String)) {
                    if !grant_guard(
                        &target.rules.insert,
                        &binding.columns,
                        &binding.principal,
                        target_name,
                        target,
                        schema,
                    ) {
                        issue(
                            format!(
                                "{target_name} grants the private audience of {table_name}: every insert rule branch needs an independent grant probe in {target_name}, `exists {target_name}[{}]`, binding its recipient to viewer and its group columns to this group's columns{}, or a created probe for the new group; `.{principal} = viewer` alone lets anyone join. This also governs insert if absent and upsert",
                                binding.columns.iter().map(|column| format!(".{column}, ")).collect::<String>() + "viewer",
                                if binding.columns.is_empty() { " (none: the roster is one group, so a caller already in it may add others)".to_owned() } else { format!(" ({})", binding.columns.join(", ")) }
                            ),
                        );
                    }
                    let fixed = binding.columns.iter().chain(std::iter::once(&binding.principal)).all(|column|
                        target.constraints.iter().any(|constraint| matches!(constraint, Constraint::Immutable { fields } if fields.contains(column))));
                    if !fixed
                        && !grant_guard(
                            &target.rules.update,
                            &binding.columns,
                            &binding.principal,
                            target_name,
                            target,
                            schema,
                        )
                    {
                        issue(
                            format!(
                                "{target_name} can retarget the private audience of {table_name}: make the audience columns immutable or guard every update branch with an independent grant probe"
                            ),
                        );
                    }
    let _ = principal;
}

/// One arm's index needs: without a window, an index leading with the
/// arm's columns (a device takes a group's rows through it); with one, a
/// plain window index leading with them.
pub(super) fn sync_index_issues(table_name: &str, table: &Table, sync: &Sync, audience: &Audience, horizon: Option<&Horizon>, issue: &mut dyn FnMut(String)) {
    // Without a horizon, a group's rows are taken from an index leading with
    // the audience columns; `check` says so, or the first sync would.
    let group = audience.columns();
    if horizon.is_none() {
        if !group.is_empty() && !table.indexes.values().any(|index| index.is_plain() && index.columns().starts_with(&group)) {
            issue(format!("{} needs an index on {table_name} leading with {} so a device can take a group's rows: `by by{}: {}, id`", sync.line_in(table), group.join(", "), group.iter().map(|c| { let mut cs = c.chars(); cs.next().map(|f| f.to_uppercase().collect::<String>() + cs.as_str()).unwrap_or_default() }).collect::<String>(), group.join(", ")),
            );
        }
    }
    if let Some(horizon) = horizon {
        match table.index(&horizon.by) {
            None => issue(format!("horizon names unknown index {table_name}.{}", horizon.by),
            ),
            Some(index) => {
                if !index.is_plain() || !index.columns().starts_with(&group) {
                    issue(format!(
                            "horizon index {} must be plain and lead with the audience columns {}", horizon.by, group.join(", ")),
                    );
                }
                if horizon.last == 0 {
                    issue("a horizon keeps at least one row".into());
                }
            }
        }
    }
}

// @ref LLP 3000#7-decisions-during-execution — every audience-granting branch requires prior authority
fn grant_guard(
    rule: &RuleExpr,
    columns: &[String],
    principal: &str,
    audience_name: &str,
    audience: &Table,
    schema: &Schema,
) -> bool {
    let group_pinned = |key: &[RuleTerm]| {
        !columns.is_empty() && columns.iter().all(|column|
        key.iter().any(|term| matches!(term, RuleTerm::Field(field) | RuleTerm::NextField(field) if field == column)))
    };
    // @ref LLP 3000.036.002#2 — a trusted seat grants on the server, never a device audience.
    if rule.service_grant().is_some() { return true; }
    match rule {
        RuleExpr::Deny => true,
        RuleExpr::Exists { table, index, key, .. } => table == audience_name && audience.index(index).is_some_and(|index| {
            let fields = index.columns();
            let term = |column: &str| fields.iter().position(|field| field == column).and_then(|at| key.get(at));
            // With no group columns (a global roster) the one group is the
            // whole table, so any row of the caller's proves the grant.
            index.is_plain() && fields.len() == key.len() && term(principal) == Some(&RuleTerm::Viewer)
                && columns.iter().all(|column|
                    matches!(term(column), Some(RuleTerm::Field(field) | RuleTerm::NextField(field)) if field == column))
        }),
        RuleExpr::Created { table, index, key, .. } => columns.len() == 1 && key.len() == 1 && group_pinned(key)
            && matches!(audience.columns.get(&columns[0]), Some(ColumnType::Ref { table: group }) if group == table)
            && schema.table(table).and_then(|table| table.index(index)).is_some_and(|index| index.unique && index.columns() == ["id"]),
        RuleExpr::And(rules) => rules.iter().any(|rule| grant_guard(rule, columns, principal, audience_name, audience, schema)),
        RuleExpr::Or(rules) => rules.iter().all(|rule| grant_guard(rule, columns, principal, audience_name, audience, schema)),
        _ => false,
    }
}

/// The audience table's insert (or update) rule with the grant probe the
/// sync check asks for conjoined to every branch that lacks one: the paste-
/// ready repair behind "every insert rule branch needs an independent grant
/// probe". Branches that already prove a grant are kept as written. When no
/// branch lets the mutation that creates a group add its first member, the
/// second string names that `created` branch.
#[derive(Clone, Debug, PartialEq, Eq)]
pub struct GrantRepair {
    /// The whole repaired rule line, `insert <- ...` or `update <- ...`.
    pub rule: String,
    /// The branches, as written, that prove no grant.
    pub unguarded: Vec<String>,
    /// `or created group[.column]` when no branch admits a new group's first member.
    pub created: Option<String>,
}

pub fn audience_grant_repair(schema: &Schema, target_name: &str, binding: &AudienceBinding, update: bool) -> Option<GrantRepair> {
    let target = schema.table(target_name)?;
    let rule = if update { &target.rules.update } else { &target.rules.insert };
    let mut key: Vec<RuleTerm> = binding.columns.iter().map(|column| RuleTerm::Field(column.clone())).collect();
    key.push(RuleTerm::Viewer);
    let probe = RuleExpr::Exists { table: target_name.to_owned(), index: binding.group_index.clone(), key, equalities: Vec::new(), provenance: crate::ir::KeyProvenance::Shape };
    let guarded = |rule: &RuleExpr| grant_guard(rule, &binding.columns, &binding.principal, target_name, target, schema);
    let conjoin = |rule: &RuleExpr| -> RuleExpr {
        if guarded(rule) { return rule.clone(); }
        match rule {
            RuleExpr::Allow => probe.clone(),
            RuleExpr::And(items) => RuleExpr::And(items.iter().cloned().chain([probe.clone()]).collect()),
            other => RuleExpr::And(vec![other.clone(), probe.clone()]),
        }
    };
    let mut repaired = match rule {
        RuleExpr::Or(_) => RuleExpr::Or(rule.disjuncts().into_iter().map(conjoin).collect()),
        other => conjoin(other),
    };
    // A standalone membership arm absorbs every extra conjunction of that same arm.
    if let RuleExpr::Or(items) = &mut repaired && items.contains(&probe) {
        items.retain(|item| item == &probe || !matches!(item, RuleExpr::And(parts) if parts.contains(&probe)));
        if items.len() == 1 { repaired = items.pop().unwrap(); }
    }
    let creates = |rule: &RuleExpr| rule.disjuncts().iter().any(|branch| matches!(branch, RuleExpr::Created { .. })
        || matches!(branch, RuleExpr::And(items) if items.iter().any(|item| matches!(item, RuleExpr::Created { .. }))));
    let created = match binding.columns.as_slice() {
        [column] if !update && !creates(rule) => match target.columns.get(column).map(ColumnType::required) {
            Some(ColumnType::Ref { table: group }) => Some(format!("or created {group}[.{column}]")),
            _ => None,
        },
        _ => None,
    };
    let printed = |rule: &RuleExpr| match rule {
        RuleExpr::Or(items) => items.iter().map(|item| match item { RuleExpr::And(_) if items.len() > 1 => format!("({})", item.quarry()), other => other.quarry() }).collect::<Vec<_>>().join(" or "),
        other => other.quarry(),
    };
    let unguarded = rule.disjuncts().into_iter().filter(|branch| !guarded(branch)).map(RuleExpr::quarry).collect();
    Some(GrantRepair { rule: format!("{} <- {}", if update { "update" } else { "insert" }, printed(&repaired)), unguarded, created })
}

/// The target table's columns a content row's `prefix` columns are matched
/// against. `.id` on the content table matches the target's reference to
/// the content table; any other column matches the target's same-named
/// column of the same type, else its sole reference to the same table.
pub fn target_prefix_columns(content_name: &str, content: &Table, target: &Table, prefix: &[String], principal: &str) -> Option<Vec<String>> {
    let mut out = Vec::new();
    for column in prefix {
        let kind = if column == "id" { ColumnType::Ref { table: content_name.to_owned() } } else { content.columns.get(column)?.required().clone() };
        let same_name = target.columns.get(column).filter(|target_kind| column != "id" && target_kind.required() == &kind && column != principal);
        if same_name.is_some() {
            out.push(column.clone());
            continue;
        }
        let by_reference: Vec<&String> = target.columns.iter().filter(|(name, target_kind)| name.as_str() != "id" && name.as_str() != principal && matches!(&kind, ColumnType::Ref { .. }) && target_kind.required() == &kind).map(|(name, _)| name).collect();
        match by_reference.as_slice() {
            [one] => out.push((*one).clone()),
            _ => return None,
        }
    }
    Some(out)
}

/// The proof behind `sync to target[.prefix] by .principal`: the read
/// rule's probe of the target and the target index that probe uses. The
/// partition log delivers by exactly this index; it never reconstructs the
/// mapping from column names or types (a same-typed second reference would
/// deliver one group under the authority of another).
#[derive(Clone, Debug, PartialEq, Eq)]
pub struct AudienceBinding {
    /// The unique target index `columns ++ [principal]`: a group's principals.
    pub group_index: String,
    /// The target's columns the content row's prefix maps to, in order.
    pub columns: Vec<String>,
    /// The target's principal column, at the probe's `viewer` position.
    pub principal: String,
    /// A target index leading with `principal ++ columns`: a principal's groups.
    pub principal_index: String,
}

pub fn audience_binding(schema: &Schema, table_name: &str, table: &Table, sync: &Sync) -> Result<AudienceBinding, String> {
    let Audience::Target { table: target_name, prefix, principal } = &sync.audience else {
        return Err(format!("{table_name} does not sync to a target table"));
    };
    let target = schema.table(target_name).ok_or_else(|| format!("sync to {target_name} names an unknown table"))?;
    let spelled_prefix = prefix.iter().map(|c| format!(".{c}")).collect::<Vec<_>>().join(", ");
    let mut want = prefix.iter().map(|column| RuleTerm::Field(column.clone())).collect::<Vec<_>>();
    want.push(RuleTerm::Viewer);
    let spelled_want = want.iter().map(|t| match t { RuleTerm::Field(f) => format!(".{f}"), _ => "viewer".into() }).collect::<Vec<_>>().join(", ");
    let audience_rule = table.rules.read.sync_audience_rule(table_name, table);
    let columns = if audience_rule.is_uniformly_readable() {
        // A public table may sync to any audience; the mapping is by name and type.
        target_prefix_columns(table_name, table, target, prefix, principal).ok_or_else(|| format!("sync to {target_name}[{spelled_prefix}]: {target_name} has no column matching {} (same name and type, or one reference to the same table)", prefix.iter().map(|c| format!("{table_name}.{c}")).collect::<Vec<_>>().join(", ")))?
    } else {
        // The rule is exactly the probe the sync inverts: a second disjunct
        // would admit viewers the partition never delivers to; a conjunct
        // or a literal in the key would deliver rows the rule denies.
        let exact = format!("sync to {target_name}[{spelled_prefix}] is only sound when the read rule is exactly `exists {target_name}[{spelled_want}]`; it reads under `{}`", table.rules.read.quarry());
        // @ref LLP 3000.036.000.000#2.2 — a union's group arm inverts its own whole disjunct.
        let probe = if sync.is_union() {
            audience_rule.disjuncts().into_iter().find(|disjunct| matches!(disjunct, RuleExpr::Exists { .. })).ok_or_else(|| exact.clone())?
        } else { audience_rule.as_ref() };
        let RuleExpr::Exists { table: probed, index, key, equalities, .. } = probe else { return Err(exact) };
        if probed != target_name || *key != want || !equalities.is_empty() {
            return Err(exact);
        }
        let index_def = target.index(index).ok_or_else(|| format!("the read rule probes unknown index {target_name}.{index}"))?;
        let index_columns = index_def.columns();
        if !index_def.is_plain() || index_columns.len() != want.len() {
            return Err(format!("the read rule's probe uses {target_name}.{index}, which does not cover exactly `{spelled_want}`"));
        }
        let probe_principal = &index_columns[prefix.len()];
        if probe_principal != principal {
            return Err(format!("sync to {target_name}[{spelled_prefix}] by .{principal} reads recipients from {target_name}.{principal}, but the read rule binds viewer to {target_name}.{probe_principal}; say `by .{probe_principal}`"));
        }
        index_columns[..prefix.len()].to_vec()
    };
    for column in prefix {
        if table.columns.get(column).is_some_and(ColumnType::is_optional) && !table.through.as_ref().is_some_and(|t| t.fields.contains_key(column)) {
            return Err(format!("sync to {target_name}[{spelled_prefix}] needs required group keys; {table_name}.{column} is optional. Make every content group column required so online rules and device partitions name the same group"));
        }
    }
    for column in &columns {
        if target.columns.get(column).is_some_and(ColumnType::is_optional) {
            return Err(format!("sync to {target_name}[{spelled_prefix}] needs required group keys; {target_name}.{column} is optional. Make every audience group column required so membership discovery, revocation and content delivery name the same group"));
        }
    }
    let mut by_group = columns.clone();
    by_group.push(principal.clone());
    let (group_index, group_def) = target.indexes.iter().find(|(_, index)| index.is_plain() && index.columns() == by_group).ok_or_else(|| format!("sync to {target_name}[{spelled_prefix}] needs an index on {target_name} over {}", by_group.join(", ")))?;
    if !group_def.unique {
        return Err(format!("the audience index {target_name}.{group_index} must be unique: with two rows for one group and principal, deleting one would revoke a group the other still grants"));
    }
    let mut by_principal = vec![principal.clone()];
    by_principal.extend(columns.iter().cloned());
    let principal_index = target.indexes.iter().find(|(_, index)| index.is_plain() && index.columns().starts_with(&by_principal)).map(|(name, _)| name.clone()).ok_or_else(|| format!("sync to {target_name}[{spelled_prefix}] needs an index on {target_name} whose leading columns are {} so a device can find its groups", by_principal.join(", ")))?;
    Ok(AudienceBinding { group_index: group_index.clone(), columns, principal: principal.clone(), principal_index })
}

/// Missing optional columns read as null; missing required columns with a
/// literal default read that value. Contextual defaults belong to inserts.
pub fn materialize_optional_fields(row: &mut Value, table: &Table) {
    let fields = match row {
        Value::Row(fields) | Value::Map(fields) => fields,
        _ => return,
    };
    for (name, kind) in &table.columns {
        if !fields.contains_key(name) {
            if table.maintained.contains_key(name) {
                fields.insert(name.clone(), Value::Int(0));
            } else if kind.is_optional() {
                fields.insert(name.clone(), Value::Null);
            } else if let Some(ColumnDefault::Literal(value)) = table.defaults.get(name) {
                fields.insert(name.clone(), value.clone());
            }
        }
    }
}

/// Borrow complete images; materialize old images only when needed.
pub fn row_image<'a>(row: &'a Value, table: &Table) -> std::borrow::Cow<'a, Value> {
    if table.columns.keys().all(|column| row.field(column).is_ok()) { return std::borrow::Cow::Borrowed(row); }
    let mut image = row.clone();
    materialize_optional_fields(&mut image, table);
    std::borrow::Cow::Owned(image)
}

// @ref LLP 3000.022#14.3 — one calculation for the compiler and authoritative writer.
use super::*;
use crate::engine::row_copy_ceiling;

// steps, indexes, examined, written rows, transient heap, output, rule probes.
pub type Price = [u64; 7];
pub const COLUMNS: usize = 0;
pub const ASSETS: usize = 1;
pub const AGGREGATES: usize = 2;
pub const ROW_SOURCE: usize = 3;
pub const ROW_MEMBER: usize = 4;
pub const ALLOCATION: usize = 5;
pub const COUNTER: usize = 6;
pub const CAPTURE: usize = 7;
pub const BIRTH: usize = 8;
pub const VIEW_READ: usize = 9;
#[derive(Clone, Debug, PartialEq, Eq, Serialize, Deserialize)]
pub struct ActionEvents {
    pub heap_per_write: u64,
    pub steps_per_write: u64,
    pub tables: Vec<String>,
}
#[derive(Clone, Debug, Default, PartialEq, Eq, Serialize, Deserialize)]
pub struct WriteReservation {
    #[serde(default, skip_serializing_if = "Option::is_none")]
    pub live: Option<super::live::LiveReservation>,
    pub prices: [Price; 8],
    /// v2 adds owner initialization; absence identifies a legacy v1 plan.
    #[serde(default, skip_serializing_if = "Option::is_none")]

    #[serde(default, skip_serializing_if = "Option::is_none")]
    pub view_read: Option<Price>,

    #[serde(default, skip_serializing_if = "Option::is_none")]
    pub action_events: Option<ActionEvents>,
    /// Data-dependent skipped work cannot be admitted from a fixed price.
    pub server_only: bool,
}
impl WriteReservation {
    pub fn price(&self, phase: usize) -> Price {
        if phase == BIRTH { self.birth.unwrap_or_default() } else if phase == VIEW_READ { self.view_read.unwrap_or_default() } else { self.prices[phase] }
    }
    #[cfg(feature = "sqlite")]
    fn is_zero(&self) -> bool {
        self.prices == [[0; 7]; 8] && self.birth.unwrap_or_default() == [0; 7]
            && self.view_read.unwrap_or_default() == [0; 7] && self.action_events.is_none() && self.live.is_none() && !self.server_only
    }
}
fn add(to: &mut Price, price: Price) {
    for (to, price) in to.iter_mut().zip(price) { *to = to.saturating_add(price); }
}
/// Common partition image/metadata envelope; callers add their bookkeeping.
pub fn event_envelope(image: u64, arms: u64, metadata: u64, dependents: u64) -> (u64,u64) {
    (dependents, image.saturating_mul(2*arms).saturating_add(metadata.saturating_mul(2*arms+2*dependents)))
}
pub fn slots(kind: &ColumnType, in_list: bool) -> u64 {
    match kind {
        ColumnType::Optional(inner) => slots(inner, in_list),
        ColumnType::List { of, max } | ColumnType::ListRange { of, max, .. } => slots(of, true).saturating_mul(u64::from(*max)),
        ColumnType::Image { .. } | ColumnType::Video { .. } | ColumnType::File { .. } if in_list => 1,
        _ => 0,
    }
}
pub fn asset_price(table: &Table) -> Price {
    let n = table.columns.values().map(|k| slots(k, false)).fold(0_u64, u64::saturating_add);
    [n.saturating_mul(8), 0, n.saturating_mul(4), n.saturating_mul(2), n.saturating_mul(16_384), 0, 0]
}

}

fn row_price(schema: &Schema, m: &Maintain, recipients: u64) -> Price {
    let k = m.row_key.as_ref().unwrap();
    let target = &schema.tables[&m.target];
    let image = row_copy_ceiling(target);

     row_copy_ceiling(&schema.tables[&k.table]).saturating_mul(4).saturating_add(image.saturating_mul(16)),
     recipients.saturating_mul(image.saturating_mul(2)), 0]

/// Missing references mean an unadmitted schema, never a zero price.
#[cfg(feature = "sqlite")]
pub fn reservation(schema: &Schema, source: &str) -> Option<WriteReservation> {
    let table = schema.table(source)?;
    let mut plan = WriteReservation::default();
    let mut birth = [0; 7];
    for m in schema.maintains.iter().filter(|m| m.private.as_ref().is_some_and(|owner| owner.table == source)) {
        birth[2] += 1;
        birth[4] += row_copy_ceiling(schema.table(&m.target)?);
    }
    if birth[2] != 0 { birth[4] += row_copy_ceiling(table); }
    plan.birth = Some(birth);
    if let Some(m) = schema.maintain_for(source).filter(|m| m.row_key.is_some()) {
        let live = m.row_count_needs_live_basis(schema);
        let bytes = if live {
            let certificate = [m.target.clone(), format!("$basis:{}", m.target)].iter()
                .map(|name| schema.table(name).map(crate::rows::disclosed::decode_reservation)).collect::<Option<Vec<_>>>()?.into_iter().sum::<u64>();
            certificate.max(crate::rows::disclosed::decode_reservation(schema.table(&m.source)?))
        } else { crate::rows::MAX_INDEX_KEY_BYTES as u64 * 4 };
        plan.view_read = Some([u64::from(live),0,1,0,bytes,0,0]);
    }

    if super::live::parent(schema,source) { plan.live=Some(super::live::write_price(schema,source)?); }
    plan.prices[ASSETS] = asset_price(table);
    let mut public = 0;
    for m in &schema.maintains {
        if m.source != source && !m.row_key.as_ref().is_some_and(|k| k.table == source) { continue; }
        let target = schema.table(&m.target)?;
        if let Some(k) = &m.row_key {
            schema.table(&k.table)?;
            let (phase, recipients) = if k.table == source { (ROW_MEMBER, 1) } else { (ROW_SOURCE, k.fanout.saturating_mul(2)) };
            add(&mut plan.prices[phase], row_price(schema, m, recipients));
        } else if let Some(owner) = &m.private {
            let parent = schema.table(&owner.table)?;
            let entries = parent.indexes.values().map(|index| {
                if let Some((_, fields)) = index.word_component() {
                    fields.iter().filter_map(|f| parent.columns.get(f)).map(|kind| match kind.required() {
                        ColumnType::Text { max } => u64::from(*max), _ => 0,
                    }).sum::<u64>().min(crate::words::MAX_TOKENS_PER_ROW as u64)
                } else { 1 }
            }).sum::<u64>();
            let heap = 4*entries.saturating_sub(parent.indexes.len() as u64)*(crate::rows::MAX_INDEX_KEY_BYTES as u64+24)
                + row_copy_ceiling(table)*2 + row_copy_ceiling(target)*(16+4*target.indexes.len() as u64)
                + row_copy_ceiling(parent)*(16+4*parent.indexes.len() as u64);
            add(&mut plan.prices[COLUMNS], [m.evaluation_steps()*2+8, 4*(target.indexes.len() as u64+entries), 4, 4, heap, 0, 0]);
        } else {
            public += 1;
            let evaluations = if table.indexes.contains_key(&m.contribution_index()) { 6 } else { 2 };
            add(&mut plan.prices[AGGREGATES], [m.evaluation_steps()*evaluations+4,
                target.indexes.len() as u64*4+2, 4, 2,
                row_copy_ceiling(target)*(16+4*target.indexes.len() as u64), 0, 0]);

        }
    }
    if public > 0 { plan.prices[AGGREGATES][4] += row_copy_ceiling(table)*4; }
    for f in schema.receipts.iter().filter(|f| f.messages == source) {
        let scope = schema.table(&f.scope)?;
        // The clause admits at most 50 non-expiring members. Charge the full
        // key-only allocation allowance on both hosts, not a live count.
        add(&mut plan.prices[ALLOCATION], [51, 51, 51, 0, 0, 0, 0]);
        add(&mut plan.prices[COUNTER], counter_price(scope));
        // These hooks need more than a simple, fixed counter replacement.
        // Preserve server execution; do not guess their data-dependent work.
        plan.server_only |= !simple_counter(schema, f);
    }
    plan.prices[CAPTURE] = capture_price(schema, table);
    // @ref LLP 3000.039#5.1 — export event pricing and affected tables before projection removes private maintained-column derivations.
    if table.on_delete.values().any(|action| action.writes()) {
        let mut row_tables = BTreeSet::from([source.to_owned()]);
        for m in &schema.maintains {
            if m.source == source || m.row_key.as_ref().is_some_and(|key| key.table == source) {
                row_tables.insert(m.target.clone());
                if let Some(owner) = &m.private { row_tables.insert(owner.table.clone()); }
            }
        }
        let (mut image, mut dependents, mut arms) = (0_u64, 0_u64, 1_u64);
        let mut tables = BTreeSet::new();
        for name in &row_tables {
            image = image.max(row_copy_ceiling(schema.table(name)?));
            arms = arms.max(schema.table(name)?.sync.as_ref().map_or(1, |sync| sync.arms().count() as u64));
            if schema.visible(name) { tables.insert(name.clone()); }
            let mut count = 0;
            for (receiver, _, sync) in schema.synced() {
                // The one group arm is always primary; also-arms are columns.
                if matches!(&sync.audience, Audience::Target { table, .. } if table == name) { tables.insert(receiver.to_owned()); count += 1; }
            }
            dependents = dependents.max(count);
        }
        let metadata = tables.iter().map(|name| 1024 + 4 * crate::rows::MAX_INDEX_KEY_BYTES as u64 + 4 * name.len() as u64).max().unwrap_or(0);
        // @ref LLP 3000.039#5.1 — each union arm can retain its own row event; price schema width rather than live recipients.
        plan.action_events = Some(ActionEvents { steps_per_write: dependents, heap_per_write: event_envelope(image,arms,metadata,dependents).1, tables: tables.into_iter().collect() });
    }

    Some(plan)
}

pub fn simple_counter(schema: &Schema, f: &ReceiptFamily) -> bool {
    let Some(scope) = schema.table(&f.scope) else { return false; };
    !(!scope.written_by.is_empty() || scope.through.is_some()
            || !scope.recorded.is_empty() || !scope.sequences.is_empty()
            || scope.indexes.values().any(|i| i.uses_column(&f.counter))
            || scope.constraints.iter().any(|c| match c {
                Constraint::Unique { .. } | Constraint::FinalExistsOnInsert { .. } | Constraint::Text { .. } => false,
                Constraint::Immutable { fields } => fields.contains(&f.counter),
                _ => true,
            })
            || schema.maintains.iter().any(|m| m.source == f.scope || m.row_key.as_ref().is_some_and(|k| k.table == f.scope))
            || schema.timelines.iter().any(|t| [&t.posts[0], &t.reposts[0], &t.edges[0], &t.authors[0]].contains(&&f.scope)))
}

    #[cfg(feature = "sqlite")]
    fn prepare_native_reservations(&mut self) {
        // Projected columns have no derivation; retain their compiler's complete
        // plan. Full schemas always rebuild, so a forged price is not authority.
        if self.validate_column_owners().is_err() { return; }
        let had_prices = !self.source_reservations.is_empty();
        self.source_reservations = self.tables.keys().filter_map(|source|
            reservation(self, source).map(|price| (source.clone(), price))).collect();
        // Preserve old zero-only store bytes and existing explicit price maps.
        if !had_prices && self.source_reservations.values().all(WriteReservation::is_zero) {
            self.source_reservations.clear();
        }
    }
    /// Export every source if any plan needs a reservation; otherwise preserve
    /// capability-free legacy bytes. Devices already prove structural zero.

    pub fn export_write_reservations(&mut self) {
        self.source_reservations = self.tables.keys().filter_map(|source|
            reservation(self, source).map(|price| (source.clone(), price))).collect();
        if self.source_reservations.values().all(WriteReservation::is_zero) {
            self.source_reservations.clear();

    pub fn write_reservation(&self, source: &str) -> Option<WriteReservation> {
        let table = self.table(source)?;
        if let Some(plan) = self.source_reservations.get(source) {
            // v1 has no owner-birth price. Do not silently interpret it as zero.
            if plan.birth.is_none() && !table.maintained.is_empty() { return None; }
            return Some(plan.clone());
        }
        // A legacy representation may prove *zero*, but a device never rebuilds
        // prices from formulas. Missing private formulas prevent that proof for
        // any source. Ordinary unrelated legacy writers remain available.
        let zero = self.tables.values().all(|t| t.maintained.is_empty())
            && table.through.is_none()
            && table.columns.values().all(|kind| slots(kind, false) == 0)
            && !self.receipts.iter().any(|f| f.messages == source)
            && !self.maintains.iter().any(|m| m.source == source
                || m.row_key.as_ref().is_some_and(|k| k.table == source));
        zero.then(|| WriteReservation { birth: Some([0;7]), ..WriteReservation::default() })
    }
    pub fn prediction_reservation_refusal(&self, program: &crate::ir::Program) -> Option<&'static str> {

            if self.legacy_receipt_prediction && self.receipts.iter().any(|f|
                source == f.messages || source == f.members || source == f.runs
                || (source == f.scope && matches!(kind, "insert" | "upsert"))) {
                return Some("legacy receipt source needs the current backend before predicting");
            }
        }
        // Existing compiler read fences are retained verbatim. Hidden effect
        // slices cannot prove an unaffected footprint from their public body.
        if self.legacy_receipt_prediction && self.operations.get(&program.name).is_some_and(|f|
            !f.prediction_skips.is_empty() || !f.effect_payload_reads.is_empty()) {
            return Some("legacy hidden effect needs current compiler facts before predicting");
        }

    pub column: String,
    pub millis: i64,
    #[serde(default, skip_serializing_if = "zero")]
    pub offset: i64,

impl CalendarKey {
    pub fn value(&self, row: &Value) -> Value {
        if self.column.contains('.') && !matches!(row.field(&self.column), Ok(Value::Int(_))) { return Value::Null; }
        let at = row.field(&self.column).ok().and_then(|v| v.as_int().ok()).unwrap_or(0);

            "month" => crate::calendar::month_of(at),
            "year" => crate::calendar::year_of(at),
            _ => calendar_ordinal(at, self.millis, self.offset),
        })
    }

    /// `month` and `year` count civil periods; the others count fixed durations.
    pub fn civil(&self) -> bool { matches!(self.function.as_str(), "month" | "year") }
    pub fn validate(&self, source: &Table) -> Result<(), String> {
        if !source.columns.get(&self.column).is_some_and(|kind| kind == &ColumnType::Time
            || (source.recorded.contains_key(&self.column) && kind.required() == &ColumnType::Time))
            || (self.civil() && (self.millis != 0 || self.offset != 0))
            || (!self.civil() && (self.millis <= 0 || self.offset < 0 || self.offset >= self.millis))
            || !matches!(self.function.as_str(), "day" | "hour" | "bucket" | "month" | "year")

        }

    }

pub fn calendar_ordinal(at: i64, millis: i64, offset: i64) -> i64 {

#[derive(Clone, Debug, PartialEq, Eq, Serialize, Deserialize)]
pub enum Operand { Field(String), Literal(Value) }
impl Operand {
    fn value<'a>(&'a self, row: &'a Value, member: Option<&'a Value>) -> &'a Value {
        match self {
            Self::Field(field) => match member.zip(field.strip_prefix("$row.")) {
                Some((member, field)) => member.field(field),
                None => row.field(field),
            }.unwrap_or(&Value::Null),
            Self::Literal(value) => value,
        }
    }
}

#[derive(Clone, Debug, PartialEq, Eq, Serialize, Deserialize)]
pub enum Predicate {
    Bool(bool), Field(String), Not(Box<Predicate>), And(Box<Predicate>, Box<Predicate>), Or(Box<Predicate>, Box<Predicate>),
    Compare { op: String, left: Operand, right: Operand },
}
impl Predicate {
    pub fn matches(&self, row: &Value) -> bool {
        self.matches_with_member(row, None)
    }

    // One boolean/comparison walker; only comparison operands can refer to a
    // keyed member. Bare boolean fields keep their source-row interpretation.
    fn matches_with_member(&self, row: &Value, member: Option<&Value>) -> bool {
        match self {
            Self::Bool(value) => *value,
            Self::Field(field) => matches!(row.field(field), Ok(Value::Bool(true))),
            Self::Not(value) => !value.matches_with_member(row, member),
            Self::And(a, b) => a.matches_with_member(row, member) && b.matches_with_member(row, member),
            Self::Or(a, b) => a.matches_with_member(row, member) || b.matches_with_member(row, member),
            Self::Compare { op, left, right } => {
                let (a, b) = (left.value(row, member), right.value(row, member));
                match op.as_str() { "=" => a == b, "!=" => a != b, "<" => a.total_cmp(b).is_lt(), "<=" => !a.total_cmp(b).is_gt(), ">" => a.total_cmp(b).is_gt(), ">=" => !a.total_cmp(b).is_lt(), _ => false }
            }
        }
    }
    pub fn steps(&self) -> u64 {
        match self { Self::Not(p) => 1 + p.steps(), Self::And(a,b) | Self::Or(a,b) => 1 + a.steps() + b.steps(), Self::Compare { .. } => 3, _ => 1 }
    }
    #[inline(never)] // Keep one owned-field collector around the borrowed traversal.
    pub fn fields(&self, out: &mut BTreeSet<String>) {
        self.each_field(&mut |field| { out.insert(field.to_owned()); });
    }
    /// Borrow fields for dependency observation without constructing a copied set.
    pub(crate) fn each_field<'a>(&'a self, out: &mut dyn FnMut(&'a str)) {
        match self {
            Self::Field(field) => out(field),
            Self::Not(p) => p.each_field(out), Self::And(a,b) | Self::Or(a,b) => { a.each_field(out); b.each_field(out); }
            Self::Compare { left, right, .. } => {
                if let Operand::Field(field) = left { out(field); }
                if let Operand::Field(field) = right { out(field); }
            }
            Self::Bool(_) => {}
        }
    }
}

impl Maintain {
    // @ref LLP 3000.015#3 — expired recipients can miss deltas and revive on clock rollback.
    // Effective member expiry includes read-through ancestors, even over a timeless source.
    pub fn row_count_needs_live_basis(&self, schema: &crate::schema::Schema) -> bool {
        self.row_key.as_ref().is_some_and(|key|
            schema.tables[&self.source].is_expiring() || schema.tables[&key.table].is_expiring())
    }

    pub fn selected(&self, image: &Value) -> bool { if self.private.is_some() && self.by.iter().any(|f| matches!(image.field(f), Err(_) | Ok(Value::Null))) { return false; } self.filter.as_ref().is_none_or(|p| p.matches(image)) }
    pub fn source_key(&self, image: &Value) -> Vec<Value> {
        self.by.iter().map(|field| match self.computed.get(field) {
            Some(key) => key.value(image),
            None => image.field(self.aliases.get(field).unwrap_or(field)).cloned().unwrap_or(Value::Null),
        }).collect()
    }
    pub fn bare_key(&self, field: &str) -> bool { self.by.iter().any(|f| f == field) && !self.computed.contains_key(field) }
    pub fn evaluation_steps(&self) -> u64 { self.filter.as_ref().map_or(0, Predicate::steps) + self.by.len() as u64 + self.computed.len() as u64 + self.computed.values().filter(|key| key.offset != 0).count() as u64 }
    pub fn dependencies(&self) -> BTreeSet<String> {
        let mut fields = BTreeSet::new();
        if let Some(k)=&self.row_key { k.predicate.fields(&mut fields); fields.retain(|f|!f.starts_with("$row.")); return fields; }
        if let Some(p) = &self.filter { p.fields(&mut fields); }
        for field in &self.by { fields.insert(self.computed.get(field).map_or(self.aliases.get(field).unwrap_or(field), |key| &key.column).clone()); }
        if let crate::schema::MaintainKind::Sum { of } = &self.kind { fields.insert(of.clone()); }
        fields
    }
    pub fn contribution_index(&self) -> String { format!("$contribution:{}", self.target) }
    pub fn source_scan<'a>(&self, source: &'a Table, group: &[Value]) -> (&'a str, Vec<Value>) {
        let name = self.contribution_index();
        if let Some((name, _)) = source.indexes.get_key_value(&name) { return (name, group.to_vec()); }
        source.indexes.iter().find(|(_, index)| index.is_plain() && index.columns().starts_with(&self.by))
            .map(|(name, _)| (name.as_str(), group.to_vec())).unwrap_or(("byId", Vec::new()))
    }
    /// The group also determines the engine's synthetic id, never its value.
    pub fn read_uses_non_key_fields(&self, schema: &crate::schema::Schema) -> bool {
        let Some(target) = schema.table(&self.target) else { return false; };
        let mut fields = BTreeSet::new();
        target.rules.read.fields(&mut fields);
        fields.iter().any(|field| field != "id" && !self.group_fields().contains(field))
    }

    pub fn expiring_read_refusal(&self, schema: &crate::schema::Schema) -> Option<String> {
        // @ref LLP 3000.015#3 — a row-keyed count is authorized by its member row and certifies its live basis.
        if self.row_key.is_some() { return None; }
        (schema.table(&self.source).is_some_and(Table::is_expiring) && self.read_uses_non_key_fields(schema)).then(||
            format!("{}: expiring aggregate authorization must use only group keys; use a rule over {}, omit read to inherit the source rule, or use online only with an explicit key rule; a cap cannot authorize the live value before recounting", self.target, self.group_fields().join(", ")))
    }

    pub fn groups_by_parent(&self, schema: &crate::schema::Schema) -> bool {
        schema.table(&self.source).and_then(|source| source.through.as_ref()).is_some_and(|parent| self.by.contains(&parent.reference))
    }

    pub fn validate_keys(&self, source: &Table) -> Result<(), String> {
        for (output,capture) in &self.aliases {
            if !self.by.contains(output) || !source.through.as_ref().is_some_and(|t| t.fields.contains_key(capture)) {
                return Err("maintain aliases must reuse an inherited capture".into());
            }
        }
        if let Some(parent)=&self.parent_key && (source.through.as_ref()!=Some(parent) || !self.by.contains(&parent.reference)) {
            return Err("maintain parent key must group by its source's read-through reference".into());
        }
        for key in self.computed.values() {
            key.validate(source)?;
        }
        Ok(())
    }
}
