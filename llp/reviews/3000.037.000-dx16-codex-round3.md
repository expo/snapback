pub(super) fn apply(dir: &std::path::Path, additions: &[(&str, String)]) -> std::io::Result<()> {

        let stage=dir.join(format!(".snapback4-frozen-{}-{nonce}",std::process::id()));
        std::fs::create_dir(&stage)?;
        let result=(|| -> std::io::Result<()> {
            for (name,body) in &frozen {
                let relative=std::path::Path::new(name.strip_prefix("v1/").unwrap());
                if relative.components().any(|c|!matches!(c,std::path::Component::Normal(_))) {return Err(std::io::Error::other("frozen destination escapes v1"));}
                let path=stage.join(relative);
                std::fs::create_dir_all(path.parent().unwrap())?;
                std::fs::write(path,body)?;
            }
            if dir.join("v1").symlink_metadata().is_ok() {return Err(std::io::Error::other("v1 appeared during capture; no historical source replaced"));}
            std::fs::rename(&stage,dir.join("v1"))
        })();
        if result.is_err() {let _=std::fs::remove_dir_all(&stage);}
        result?;
    }
    for (name, body) in additions {
        if name.starts_with("v1/") {continue;}
        let path = dir.join(name);
        std::fs::create_dir_all(path.parent().unwrap())?;
        if *name == "package.json" || *name == "vite.config.mts" || *name == "scripts/frontend-proof.mjs" || *name == "scripts/day2-proof.mjs" { super::replace_file(&path, body)?; }

        }
    }
    Ok(())

#[derive(serde::Deserialize)]
struct RetainedColumn {
    kind: ColumnType,
    default: Option<ColumnDefault>,
    authoritative: bool,
}

impl Inspection {
    pub fn retained_schema(&self, mut schema: Schema) -> Result<Schema> {
        let conn = &self.conn;
        let mut historical = BTreeMap::<String, Table>::new();
        let mut definitions=Vec::new();
        if self.inspection_has_table("_dormant_tables")? { definitions.push("SELECT table_name,definition FROM _dormant_tables"); }
        if self.inspection_has_table("_dormant_narrow_rows")? {definitions.push("SELECT c.name,n.definition FROM _dormant_narrow_rows n JOIN _catalog c ON c.num=n.t");}
        if definitions.is_empty() {definitions.push("SELECT NULL,NULL WHERE 0");}
        let mut statement=conn.prepare(&format!("{} ORDER BY 1",definitions.join(" UNION "))).map_err(store_error)?;
        let rows = statement
            .query_map([], |row| {
                Ok((row.get::<_, String>(0)?, row.get::<_, String>(1)?))
            })
            .map_err(store_error)?;
        for row in rows {
            let (name, definition) = row.map_err(store_error)?;
            let table: Table =
                serde_json::from_str(&definition).map_err(|e| ("E_SCHEMA", e.to_string()))?;
            historical.entry(name).or_insert(table);
        }
        for (name, old) in historical {
            if let Some(table) = schema.tables.get_mut(&name) {
                for (column, kind) in &old.columns {
                    if !old.engine_field(column) {
                        table.columns.entry(column.clone()).or_insert(kind.clone());
                        if let Some(default) = old.defaults.get(column) {
                            table
                                .defaults
                                .entry(column.clone())
                                .or_insert(default.clone());
                        }
                    }
                }
            } else {
                schema.tables.insert(name, old);
            }
        }
        let columns_sql=if self.inspection_has_table("_dormant_columns")? {"SELECT table_name,column_name,definition FROM _dormant_columns ORDER BY table_name,column_name"} else {"SELECT NULL,NULL,NULL WHERE 0"};
        let mut statement=conn.prepare(columns_sql).map_err(store_error)?;
        let rows = statement
            .query_map([], |row| {
                Ok((
                    row.get::<_, String>(0)?,
                    row.get::<_, String>(1)?,
                    row.get::<_, String>(2)?,
                ))
            })
            .map_err(store_error)?;
        for row in rows {
            let (table, column, definition) = row.map_err(store_error)?;
            let retained: RetainedColumn =
                serde_json::from_str(&definition).map_err(|e| ("E_SCHEMA", e.to_string()))?;
            if !retained.authoritative {
                continue;
            }
            if let Some(target) = schema.tables.get_mut(&table) {
                target
                    .columns
                    .entry(column.clone())
                    .or_insert(retained.kind);
                if let Some(default) = retained.default {
                    target.defaults.entry(column).or_insert(default);
                }
            }
        }
        Ok(schema)
    }
