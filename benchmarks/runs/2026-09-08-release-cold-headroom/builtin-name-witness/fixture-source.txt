===== matrix/snapback/witness.q =====
query wordsRead():
  row = words['words-row'] ! else NOT_FOUND
  return { table: 'words', value: row.value }

query tokensWitness():
  row = tokens['tokens-row'] ! else NOT_FOUND
  return { table: 'tokens', value: row.value, builtin: tokens('Hi @alice and @bob!') }

query rangeWitness():
  row = range['range-row'] ! else NOT_FOUND
  return { table: 'range', value: row.value, builtin: range(1, 4) }

query minWitness():
  row = min['min-row'] ! else NOT_FOUND
  return { table: 'min', value: row.value, builtin: min(row.value, 10) }

query maxWitness():
  row = max['max-row'] ! else NOT_FOUND
  return { table: 'max', value: row.value, builtin: max(row.value, 10) }

query countWitness():
  row = count['count-row'] ! else NOT_FOUND
  return { table: 'count', value: row.value, builtin: count(range(1, 4)) }

query divmodWitness():
  row = divmod['divmod-row'] ! else NOT_FOUND
  return { table: 'divmod', value: row.value, builtin: divmod(row.value, 3) }

query shareWitness():
  row = share['share-row'] ! else NOT_FOUND
  recipients = [viewer]
  return { table: 'share', value: row.value, builtin: share(row.value, recipients, viewer) }
