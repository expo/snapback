const users = table("users");       // { id, name }
const shows = table("shows");       // { id, title }
const subs = table("subs");         // { id, userId, showId, lastListenedAt }
                                    // index subs.byUser: (userId)
const episodes = table("episodes"); // { id, showId, publishedAt, title }
                                    // index episodes.byShowPublished: (showId, publishedAt)

export const inboxBadges = query({}, ({ db, ctx }) => {
  const subscriptions = db.scan(subs.byUser, [ctx.viewer], { take: 50 });
  const result: any[] = [];

  for (const subscription of subscriptions) {
    const show = db.get(shows, subscription.showId);
    const recentEpisodes = db.scan(
      episodes.byShowPublished,
      [subscription.showId],
      { order: "asc", after: subscription.lastListenedAt, take: 100 },
    );
    const capped = recentEpisodes.countCapped();

    result.push({
      showTitle: show === null ? null : show.title,
      badge: capped.atCap || capped.count > 99 ? "99+" : capped.count,
    });
  }

  return result;
});
