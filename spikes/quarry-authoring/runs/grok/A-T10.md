A maintained per-ride count is exact at commit, so concurrent joiners cannot slip past a stale scan. Membership is returned first so a second join is idempotent and never raises `RIDE_FULL`.

```ts
const users = table("users");           // { id, name }
const rides = table("rides");           // { id, driverId, seats, departsAt }
const passengers = table("passengers"); // { id, rideId, userId, at }
                                        // unique index passengers.byRideUser: (rideId, userId)
                                        // index passengers.byRideAt: (rideId, at)

const passengerCounts = maintained("passengerCounts", passengers); // keyed by rideId; exact at commit

export const joinRide = mutation({ rideId: id(rides) }, ({ db, ctx, args }) => {
  const existing = db.get(passengers.byRideUser, [args.rideId, ctx.viewer]);
  if (existing) {
    return existing;
  }

  const ride = db.get(rides, args.rideId);
  db.require(ride !== null, "NOT_FOUND");

  const agg = db.get(passengerCounts, args.rideId);
  const taken = agg == null ? 0 : agg.count;
  db.require(taken < ride.seats, "RIDE_FULL");

  const passengerId = ctx.newId();
  const at = ctx.now();
  const passenger = {
    id: passengerId,
    rideId: args.rideId,
    userId: ctx.viewer,
    at,
  };
  db.insert(passengers, passenger);
  return passenger;
});
```
