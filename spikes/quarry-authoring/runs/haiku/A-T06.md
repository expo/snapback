```ts
const employees = table("employees"); // { id, name, title, managerId }

export const chain = query({ employeeId: scalar() }, ({ db, args }) => {
  const employee = db.get(employees, args.employeeId);
  if (!employee) return [];
  
  return recurse(
    employee,
    (emp) => emp.managerId ? db.get(employees, emp.managerId) : null,
    { depth: 12 }
  );
});
```

The query starts with the given employee and uses `recurse` to traverse upward through the management hierarchy by following each employee's `managerId` until reaching the top (where `managerId` is null). The depth limit of 12 prevents unbounded recursion as specified.
