```ts
const users = table("users");
const journals = table("journals");
const entries = privateTable("entries");
const coaches = table("coaches");

export const addEntry = mutation(
  { journalId: id(journals), entryText: text({ max: 4096 }) },
  ({ db, ctx, args }) => {
    const journal = db.get(journals, args.journalId);
    const isOwner = journal && journal.ownerId === ctx.viewer;
    const isCoach = db.exists(coaches.byJournalUser, [args.journalId, ctx.viewer]);

    db.require(isOwner || isCoach, "NOT_ALLOWED");

    const entryId = ctx.newId();
    db.insert(entries, {
      id: entryId,
      journalId: args.journalId,
      at: ctx.now(),
      text: args.entryText,
    });

    return entryId;
  },
);
```

Guards authorization by checking if the caller is the journal owner or an assigned coach, refusing with `NOT_ALLOWED` if neither. Handles null journal gracefully by short-circuiting the owner check. All reads are bounded (point read + index lookup) and the write is guarded before insertion.
