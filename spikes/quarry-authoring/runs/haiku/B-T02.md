```ts
const users = table("users");      // { id, name }
const recipes = table("recipes");  // { id, ownerId, title, createdAt }
                                   // index recipes.byOwnerCreated: (ownerId, createdAt)

export const myRecipes = query({}, ({ db, ctx }) => {
  return db.scan(recipes.byOwnerCreated, [ctx.viewer], {
    order: "desc",
    take: 100,
  });
});
```

The query scans the `byOwnerCreated` index with the viewer as the owner prefix, orders by `createdAt` descending to get newest first, and bounds the scan to a literal 100 rows to satisfy the static checker's boundedness requirement.
