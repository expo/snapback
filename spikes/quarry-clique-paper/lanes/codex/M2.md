```schema
use identity
  -- Built-in, in-system email/password auth with unique emails and password length >=8.
  -- Stores only provider-managed adaptive password verifiers; credentials never enter app tables or logs.
  -- Issues expiring sessions, provides viewer, and rejects revoked sessions after server-side logout.

use assets
  -- Authenticated fixture image/video upload and delivery; provides asset references while bytes remain outside the data plane. -- CHANGED

table users:
  displayName: text <=50
  handle: text <=20
  avatar: asset ?
  bio: text <=200 ?
  location: text <=50 ?
  link: text <=2048 ?
  public
  insert r <- r.id = viewer.id
  update (old, next) <- old.id = viewer.id and next.id = old.id
  unique handle

table follows: -- NEW
  follower: users -- NEW
  followed: users -- NEW
  at: time -- NEW
  public -- NEW
  insert r <- r.follower.id = viewer.id and r.followed.id != viewer.id -- NEW
  delete r <- r.follower.id = viewer.id -- NEW
  unique follower, followed -- NEW

table posts: -- NEW
  author: users -- NEW
  parent: posts ? -- NEW
  kind: enum('text', 'image', 'video') -- NEW
  body: text <=500 ? -- NEW
  image1: asset ? -- NEW
  image2: asset ? -- NEW
  image3: asset ? -- NEW
  image4: asset ? -- NEW
  video: asset ? -- NEW
  at: time -- NEW
  public -- NEW
  insert r <- r.author.id = viewer.id -- NEW
  by replies: parent, at -- NEW

table profilePosts: -- NEW
  post: posts -- NEW
  author: users -- NEW
  at: time -- NEW
  public -- NEW
  insert r <- r.author.id = viewer.id and r.post.author.id = viewer.id -- NEW
  unique post -- NEW
  by authorTime: author, at -- NEW

table likes: -- NEW
  post: posts -- NEW
  user: users -- NEW
  at: time -- NEW
  public -- NEW
  insert r <- r.user.id = viewer.id -- NEW
  delete r <- r.user.id = viewer.id -- NEW
  unique post, user -- NEW

table reposts: -- NEW
  post: posts -- NEW
  user: users -- NEW
  at: time -- NEW
  public -- NEW
  insert r <- r.user.id = viewer.id and exists profilePosts[r.post] -- NEW
  delete r <- r.user.id = viewer.id -- NEW
  unique post, user -- NEW

table globalFeedEvents: -- NEW
  scope: enum('all') -- NEW
  post: posts -- NEW
  actor: users -- NEW
  kind: enum('original', 'repost') -- NEW
  at: time -- NEW
  public -- NEW
  insert r <- r.actor.id = viewer.id and ((r.kind = 'original' and r.post.author.id = viewer.id) or (r.kind = 'repost' and exists reposts[r.post, viewer])) -- NEW
  delete r <- r.actor.id = viewer.id -- NEW
  unique post, actor, kind -- NEW
  by timeline: scope, at -- NEW

table followingFeedEvents: -- NEW
  scope: enum('all') -- NEW
  post: posts -- NEW
  actor: users -- NEW
  kind: enum('original', 'repost') -- NEW
  at: time -- NEW
  read r <- r.actor.id = viewer.id or exists follows[viewer, r.actor] -- NEW
  insert r <- r.actor.id = viewer.id and ((r.kind = 'original' and r.post.author.id = viewer.id) or (r.kind = 'repost' and exists reposts[r.post, viewer])) -- NEW
  delete r <- r.actor.id = viewer.id -- NEW
  unique post, actor, kind -- NEW
  by timeline: scope, at -- NEW

maintain followerCount = count follows by followed (readable: public) -- NEW
maintain followingCount = count follows by follower (readable: public) -- NEW
maintain likeCount = count likes by post (readable: public) -- NEW
maintain repostCount = count reposts by post (readable: public) -- NEW
maintain replyCount = count posts by parent (readable: public) -- NEW
```

  handle: text <=20,
  avatar: asset ?,
  bio: text <=200 ?,

    location,

  displayName: text <=50,
  handle: text <=20,
  avatar: asset ?,
  bio: text <=200 ?,
  location: text <=50 ?,
  link: text <=2048 ?

    location,
    link,
  }
  users[u]! else PROFILE_NOT_FOUND

mutation toggleFollow(u: users) -> bool:
  require u.id != viewer.id else CANNOT_FOLLOW_SELF
  match follows[viewer, u]:
    some f:
      -follows[f]
      false
    none:
      +follows {
        follower: viewer,
        followed: u,
        at: now,
      }
      true

mutation toggleLike(p: posts) -> bool:
  post = posts[p]! else POST_NOT_FOUND
  match likes[post, viewer]:
    some like:
      -likes[like]
      false
    none:
      +likes {
        post,
        user: viewer,
        at: now,
      }
      true

mutation toggleRepost(p: posts) -> bool:
  post = posts[p]! else POST_NOT_FOUND
  require exists profilePosts[post] else REPLIES_CANNOT_BE_REPOSTED
  match reposts[post, viewer]:
    some repost:
      globalEvent = globalFeedEvents[post, viewer, 'repost']! else FEED_EVENT_MISSING
      followingEvent = followingFeedEvents[post, viewer, 'repost']! else FEED_EVENT_MISSING
      -globalFeedEvents[globalEvent]
      -followingFeedEvents[followingEvent]
      -reposts[repost]
      false
    none:
      repost = +reposts {
        post,
        user: viewer,
        at: now,
      }
      +globalFeedEvents {
        scope: 'all',
        post,
        actor: viewer,
        kind: 'repost',
        at: repost.at,
      }
      +followingFeedEvents {
        scope: 'all',
        post,
        actor: viewer,
        kind: 'repost',
        at: repost.at,
      }
      true

```boundary
- Identity — Validates unique email/password accounts, issues and restores expiring sessions, provides `viewer`, and enforces server-side logout and revocation.
- Assets — Accepts authenticated fixture image/video uploads, records media-kind metadata, returns `asset` references, and serves bytes outside Quarry.
- Client session storage — Persists only the opaque identity session token across restarts and cannot extend, restore, or override server validity.
- Live-query transport — Re-evaluates subscribed feed, profile, count, and thread queries after commits; no subscription rows are stored.
- Channels, schedules, and effects — None are declared for M2; durable posts and engagement use ordinary tables, and M2 requires no ephemeral payload, scheduled invocation, or external emission.
```

```notes
- Following uses one current-relationship-filtered chronological event table, so unfollow immediately hides every historical event by that actor while self events remain readable without a self-follow edge.
- Global and Following timelines are separate because Global is public while Following needs viewer-dependent row rules.
- Feed/profile projection rows are duplicated transactionally with post or repost writes; a transaction cannot expose only one side.
- Repost events retain the reposter as `actor` while the nested post retains its original author, enabling correct attribution.
- Only root posts enter feed and profile projections; replies remain direct-thread rows and cannot be reposted into feeds.
- Follower, following, like, repost, and direct-reply numbers are exact maintained aggregates rather than capped page reductions.
- A required `image1` plus three optional image fields represents exactly 1–4 images; video mutations expose exactly one video field and no image fields.
- Quarry cannot inspect an `asset`'s media kind, so image-versus-video validation belongs to the asset provider contract.
- URLs remain bounded post text; tappability and suppression of preview cards are client rendering behavior.
- Pagination uses the language's final-column `time` cursor; Quarry v0.2 cannot express a composite `(time,id)` cursor, so identical intent timestamps cannot be given a formal total-order guarantee.
- Quarry cannot enforce non-empty post text or prevent sparse optional image slots because it has no text-length, absence-test, or cross-field constraint syntax.
```
