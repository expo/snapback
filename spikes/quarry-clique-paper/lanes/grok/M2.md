use identity
  -- email+password credentials, unique email, expiring sessions,
  -- server-side logout/revocation. Provides `viewer`. Credentials
  -- and session tokens never appear in tables.

use assets
  -- upload, storage, and delivery of avatars and post image/video. Provides the -- CHANGED
  -- `asset` column type; bytes never pass through the data plane.

table users:
  name: text <=50              -- display name; max 50; min 1 enforced in programs
  handle: text <=20            -- unique public handle; max 20
  avatar: asset ?              -- set via assets kind, then setAvatar
  bio: text <=200 ?
  location: text <=50 ?
  link: text <=2048 ?          -- single URL, stored as text
  unique handle
  public                       -- profiles are a declared disclosure
  insert r <- r.id = viewer    -- a principal may create only their own row
  update (old, next) <- old.id = viewer and next.id = old.id

table follows:                                                      -- NEW
  follower: users                                                   -- NEW
  followee: users                                                   -- NEW
  at: time                                                          -- NEW
  unique follower, followee                                         -- NEW
  by following: follower, at                                        -- NEW
  by followers: followee, at                                        -- NEW
  public                                                            -- NEW
  insert r <- r.follower = viewer and r.follower != r.followee      -- NEW
  delete r <- r.follower = viewer                                   -- NEW

table posts:                                                        -- NEW
  author: users                                                     -- NEW
  body: text <=500 ?                                                -- NEW
  kind: enum('text', 'image', 'video')                              -- NEW
  parent: posts ?             -- absent = original; set = reply     -- NEW
  at: time                                                          -- NEW
  by thread: parent, at                                             -- NEW
  public                                                            -- NEW
  insert r <- r.author = viewer                                     -- NEW

table postAssets:                                                   -- NEW
  post: posts                                                       -- NEW
  asset: asset                                                      -- NEW
  slot: int                   -- 0..3 images; 0 for the one video   -- NEW
  unique post, slot                                                 -- NEW
  by post, slot                                                     -- NEW
  public                                                            -- NEW
  insert r <- r.post.author = viewer                                -- NEW

table likes:                                                        -- NEW
  user: users                                                       -- NEW
  post: posts                                                       -- NEW
  unique user, post                                                 -- NEW
  read r <- r.user = viewer                                         -- NEW
  insert r <- r.user = viewer                                       -- NEW
  delete r <- r.user = viewer                                       -- NEW

table reposts:                                                      -- NEW
  user: users                                                       -- NEW
  post: posts                                                       -- NEW
  at: time                                                          -- NEW
  unique user, post                                                 -- NEW
  public                      -- feed projection; home rules exist() this -- NEW
  insert r <- r.user = viewer and not r.post.parent                 -- NEW
  delete r <- r.user = viewer                                       -- NEW

table timeline:               -- global + per-actor outbox of originals and reposts -- NEW
  actor: users                -- author (original) or reposter                      -- NEW
  post: posts                 -- always the original post being shown               -- NEW
  kind: enum('original', 'repost')                                                  -- NEW
  at: time                                                                          -- NEW
  unique actor, post, kind                                                          -- NEW
  by global: at                                                                     -- NEW
  by outbox: actor, at                                                              -- NEW
  by profile: actor, kind, at                                                       -- NEW
  public                                                                            -- NEW
  insert r <- r.actor = viewer and (r.kind = 'repost' or r.post.author = viewer)    -- NEW
  delete r <- r.actor = viewer                                                      -- NEW

table home:                   -- viewer's Following tab; fan-out on write           -- NEW
  owner: users                                                                      -- NEW
  actor: users                                                                      -- NEW
  post: posts                                                                       -- NEW
  kind: enum('original', 'repost')                                                  -- NEW
  at: time                                                                          -- NEW
  unique owner, actor, post, kind                                                   -- NEW
  by inbox: owner, at                                                               -- NEW
  read r <- r.owner = viewer and (r.actor = viewer or exists follows[follower = viewer, followee = r.actor]) and (r.kind = 'original' or exists reposts[r.actor, r.post])  -- NEW
  insert r <- (r.owner = viewer and (r.actor = viewer or exists follows[follower = viewer, followee = r.actor])) or (r.actor = viewer and exists follows[follower = r.owner, followee = viewer])  -- NEW
  delete r <- r.owner = viewer or r.actor = viewer                                  -- NEW

maintain followerCount = count follows by followee                                  -- NEW
maintain followingCount = count follows by follower                                 -- NEW
maintain likeCount = count likes by post (readable: public)                         -- NEW
maintain replyCount = count posts by parent                                         -- NEW

-- Removed vs M1: none

mutation createProfile(
  name: text <=50,
  handle: text <=20,
  bio: text <=200 ?,
  location: text <=50 ?,
  link: text <=2048 ?,
  avatar: asset ?
) -> users:
  -- Called once after identity signup, before the app treats the
  -- account as having a profile. Duplicate handle is refused at commit
  -- by `unique handle` (do not check-then-insert).
  require not exists users[viewer] else ALREADY_REGISTERED
  require name != '' else NAME_REQUIRED
  require handle != '' else HANDLE_REQUIRED
  +users { id: viewer, name, handle, bio, location, link, avatar }

mutation updateProfile(
  name: text <=50,
  handle: text <=20,
  bio: text <=200 ?,
  location: text <=50 ?,
  link: text <=2048 ?
) -> users:
  -- Full save of the text profile fields. Avatar is a separate mutation
  -- so a name/bio save cannot clobber it. Duplicate handle refused at commit.
  require exists users[viewer] else NOT_FOUND
  require name != '' else NAME_REQUIRED
  require handle != '' else HANDLE_REQUIRED
  ~users[viewer] { name, handle, bio, location, link }
  users[viewer]! else NOT_FOUND

mutation setAvatar(avatar: asset) -> users:
  -- Replace the avatar with a reference returned by the assets kind.
  require exists users[viewer] else NOT_FOUND
  ~users[viewer] { avatar }
  users[viewer]! else NOT_FOUND

mutation createPost(
  kind: enum('text', 'image', 'video'),
  body: text <=500 ?,
  image1: asset ?,
  image2: asset ?,
  image3: asset ?,
  image4: asset ?,
  video: asset ?,
  parent: posts ?
) -> posts:
  -- Original or reply. Replies skip timeline/home so they never appear in feeds.
  require exists users[viewer] else NOT_FOUND
  require not parent or exists posts[parent] else NOT_FOUND
  if kind = 'text':
    require body != '' else BODY_REQUIRED
    require not image1 and not image2 and not image3 and not image4 and not video else NO_MIX
  else:
    if kind = 'image':
      require image1 else IMAGE_REQUIRED
      require not video else NO_MIX
    else:
      require video else VIDEO_REQUIRED
      require not image1 and not image2 and not image3 and not image4 else NO_MIX
  ts = now
  p = +posts { author: viewer, body, kind, parent, at: ts }
  if image1:
    +postAssets { post: p, asset: image1, slot: 0 }
  if image2:
    +postAssets { post: p, asset: image2, slot: 1 }
  if image3:
    +postAssets { post: p, asset: image3, slot: 2 }
  if image4:
    +postAssets { post: p, asset: image4, slot: 3 }
  if video:
    +postAssets { post: p, asset: video, slot: 0 }
  if not parent:
    +timeline { actor: viewer, post: p, kind: 'original', at: ts }
    +~home[viewer, viewer, p, 'original'] { at: ts }
    for f in follows[followee = viewer] last 100 by followers:
      +~home[f.follower, viewer, p, 'original'] { at: ts }
  p {
    *,
    author: .author { name, handle, avatar },
    likeCount: #likeCount[.] ?? 0,
    liked: exists likes[viewer, .],
    replyCount: #replyCount[.] ?? 0,
    reposted: exists reposts[viewer, .],
    media: postAssets[.] first 4 {
      asset,
      slot,
    },
  }

mutation toggleFollow(u: users) -> users:
  -- Public graph, no requests. Unfollow hides via home's read rule (current
  -- follow set); follow backfills this actor's outbox into the viewer's home.
  require exists users[viewer] else NOT_FOUND
  require u != viewer else CANNOT_FOLLOW_SELF
  match follows[viewer, u]:
    some f:
      -follows[f]
    none:
      +follows { follower: viewer, followee: u, at: now }
      for ev in timeline[u] last 100 by outbox:
        +~home[viewer, u, ev.post, ev.kind] { at: ev.at }
  u {
    *,
    followerCount: #followerCount[.] ?? 0,
    followingCount: #followingCount[.] ?? 0,
    followedByMe: exists follows[viewer, .],
  }

mutation toggleLike(p: posts) -> posts:
  require exists users[viewer] else NOT_FOUND
  require exists posts[p] else NOT_FOUND
  match likes[viewer, p]:
    some l:
      -likes[l]
    none:
      +likes { user: viewer, post: p }
  p {
    *,
    author: .author { name, handle, avatar },
    likeCount: #likeCount[.] ?? 0,
    liked: exists likes[viewer, .],
    replyCount: #replyCount[.] ?? 0,
    reposted: exists reposts[viewer, .],
    media: postAssets[.] first 4 {
      asset,
      slot,
    },
  }

mutation toggleRepost(p: posts) -> posts:
  -- Own repost is a home row with actor=viewer (labeled repost), not a
  -- follow-projection of p.author. Unrepost point-deletes the timeline row;
  -- leftover home rows hide because the read rule exists() public reposts.
  require exists users[viewer] else NOT_FOUND
  require exists posts[p] else NOT_FOUND
  require not p.parent else CANNOT_REPOST_REPLY
  match reposts[viewer, p]:
    some r:
      -reposts[r]
      e = timeline[viewer, p, 'repost']
      if e:
        -timeline[e]
    none:
      ts = now
      +reposts { user: viewer, post: p, at: ts }
      +timeline { actor: viewer, post: p, kind: 'repost', at: ts }
      +~home[viewer, viewer, p, 'repost'] { at: ts }
      for f in follows[followee = viewer] last 100 by followers:
        +~home[f.follower, viewer, p, 'repost'] { at: ts }
  p {
    *,
    author: .author { name, handle, avatar },
    likeCount: #likeCount[.] ?? 0,
    liked: exists likes[viewer, .],
    replyCount: #replyCount[.] ?? 0,
    reposted: exists reposts[viewer, .],
    media: postAssets[.] first 4 {
      asset,
      slot,
    },
  }

```boundary
- identity: in-system email+password (≥8 chars) via the engine's built-in credential path (adaptive KDF — argon2id, scrypt, or bcrypt at published parameters; never plaintext or reversible); unique email (duplicate signup refused); mint expiring session tokens; persist those tokens for app-restart sign-in; logout revokes the token server-side so it cannot call programs; supplies `viewer` to every program; may expose the session email to the client — email is not a `users` column.
- assets: accept fixture/user avatar uploads and post image (1–4) / video (exactly 1) uploads; store bytes; return an `asset` reference; deliver image and video bytes to clients; programs only store and return the reference.
- channels: none this milestone (no typing/presence).
- schedules: none this milestone (no digest, no expiry jobs).
- effects: none this milestone (no verification email, no push).
```

```notes
- Following is write-time fan-out into `home`, not merge(): the follow set is unbounded and merge() is a bounded k-way of explicit prefixes. Global is `timeline` (originals+reposts) so replies can share `posts` without a range `where`.
- No self-follow: own originals/reposts are `home` rows with actor=viewer written at post/repost time; follower/following counts are only `follows` rows.
- Unfollow hides via `home`'s read rule (self or current follow; reposts also exist() the public `reposts` row). Range delete is inexpressible. Assumes unreadable rows are skipped, not slotted, in `last 20`.
- Fan-out and follow-backfill iterate `last 100` follower/outbox pages — unbounded celebrity fan-out and full-history backfill are inexpressible.
- Population facts are `#` maintains. `likeCount` is `(readable: public)` because like rows are owner-only. Feed/thread/profile queries include those fields so live screens subscribe them.
- Pagination cursor is stored intent `at` (never `now`/`commit`). No composite cursor: identical `at` can gap/duplicate.
- Optional presence is used as a boolean in `if cursor:` / `if parent:` / `if image1:` / `require not image1` (same as guards: absent is false).
- Image posts use image1..image4 args (image1 required) because tables have no list columns and `for` cannot mint incrementing slots.
- URL tap / no preview card is client rendering of `body` text; no preview table. Cannot express handle charset, whitespace-only bodies, or complete outbox history >100.
- No post edit/delete, quote-posts, follow requests, or replies-in-feeds this milestone. `posts[parent = .]` is required on the thread; `posts[.]` would be a point read of the parent.
```
